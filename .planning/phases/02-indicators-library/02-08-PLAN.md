<objective>
Implement INDIC-13 — Multi-TF alignment helper. `align(streams: dict[str, list[dict]])` requires keys `'H4'`, `'H1'`, `'M15'` (D-13). Computes EMA50 per stream, derives directional signs (+1/0/-1) using N-bar slope difference (Pitfall 7), and emits a coherence score per M15 bar where M15-direction agreement is the anchor.

Coherence formula (D-14):
- For each M15 index i, compute m15_dir = sign of M15-EMA50 slope at i.
- For H4 and H1, take their dir at the latest available bar with timestamp ≤ M15[i].time.
- Count agreements with m15_dir among {h4_dir, h1_dir, m15_dir}. Score = agreements / 3 ∈ {0/3, 1/3, 2/3, 3/3} = {0.0, 0.33, 0.66, 1.0} (round to 2 decimals; or use exact fractions).

Output: align function + dataclass + synthetic-stream tests.

    ```python
    from dataclasses import dataclass
    from indicators.trend import ema


    @dataclass
    class MTFAlignmentResult:
        score: list[float | None]
        h4_dir: list[int | None]
        h1_dir: list[int | None]
        m15_dir: list[int | None]


    def _compute_dirs(bars: list[dict], ema_period: int = 50, slope_lookback: int = 3,
                      dead_zone: float = 1e-5) -> list[int | None]:
        """Direzione per barra: sign(EMA50[i] - EMA50[i-N]) con dead-zone (Pitfall 7)."""
        if not bars:
            return []
        closes = [float(b["close"]) for b in bars]
        ema_vals = ema(closes, ema_period)
        n = len(bars)
        dirs: list[int | None] = [None]*n
        for i in range(slope_lookback, n):
            cur = ema_vals[i]
            prev = ema_vals[i - slope_lookback]
            if cur is None or prev is None or cur == 0:
                continue
            delta = cur - prev
            if abs(delta) / abs(cur) < dead_zone:
                dirs[i] = 0
            elif delta > 0:
                dirs[i] = 1
            else:
                dirs[i] = -1
        return dirs


    def align(streams: dict[str, list[dict]],
              ema_period: int = 50,
              slope_lookback: int = 3) -> MTFAlignmentResult:
        """INDIC-13: coerenza multi-TF via sign-agreement EMA50-slope su H4/H1/M15 (D-13/D-14).
        Caller MUST passare bars chiuse <= timestamp barra M15 corrente (no future leakage)."""
        for required in ("H4", "H1", "M15"):
            if required not in streams:
                raise ValueError(f"streams deve contenere chiave '{required}' (D-13)")
        h4 = streams["H4"]; h1 = streams["H1"]; m15 = streams["M15"]
        h4_dirs = _compute_dirs(h4, ema_period, slope_lookback)
        h1_dirs = _compute_dirs(h1, ema_period, slope_lookback)
        m15_dirs = _compute_dirs(m15, ema_period, slope_lookback)
        n = len(m15)
        score: list[float | None] = [None]*n
        h4_at_m15: list[int | None] = [None]*n
        h1_at_m15: list[int | None] = [None]*n
        # For each M15 bar, find latest H4/H1 bar with time <= M15[i].time
        # (Caller already sliced; we still defensive-check via timestamp lookup)
        h4_times = [int(b["time"]) for b in h4]
        h1_times = [int(b["time"]) for b in h1]
        for i in range(n):
            t = int(m15[i]["time"])
            # Walk backward through h4 to find latest <= t
            j4 = None
            for j in range(len(h4_times) - 1, -1, -1):
                if h4_times[j] <= t:
                    j4 = j
                    break
            j1 = None
            for j in range(len(h1_times) - 1, -1, -1):
                if h1_times[j] <= t:
                    j1 = j
                    break
            d_m15 = m15_dirs[i]
            d_h4 = h4_dirs[j4] if j4 is not None else None
            d_h1 = h1_dirs[j1] if j1 is not None else None
            h4_at_m15[i] = d_h4
            h1_at_m15[i] = d_h1
            if d_m15 is None or d_h4 is None or d_h1 is None:
                continue
            agree = sum(1 for d in (d_m15, d_h4, d_h1) if d == d_m15)
            score[i] = round(agree / 3.0, 2)
        return MTFAlignmentResult(score=score, h4_dir=h4_at_m15, h1_dir=h1_at_m15, m15_dir=m15_dirs)
    ```
