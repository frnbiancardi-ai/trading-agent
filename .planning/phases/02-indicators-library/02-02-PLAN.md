       ```python
       from dataclasses import dataclass
       from indicators.trend import sma, ema

       @dataclass
       class BollingerResult:
           upper: list[float | None]
           middle: list[float | None]
           lower: list[float | None]
           bbw: list[float | None]
           squeeze: list[bool | None]
           squeeze_ttm: list[bool | None]

       @dataclass
       class KeltnerResult:
           upper: list[float | None]
           middle: list[float | None]
           lower: list[float | None]

       def bollinger_bands(
           closes: list[float],
           length: int = 20,
           std: float = 2.0,
           squeeze_lookback_bars: int = 180,
           squeeze_pct: float = 25.0,
           highs: list[float] | None = None,
           lows: list[float] | None = None,
           keltner_length: int = 20,
           keltner_scalar: float = 2.0,
       ) -> BollingerResult:
           """Bande di Bollinger (length, std) + squeeze percentile + squeeze TTM (BB-inside-Keltner).
           Squeeze percentile: BBW < squeeze_pct-esimo percentile della finestra trailing.
           Squeeze TTM richiede highs/lows; se assenti, squeeze_ttm = [None]*n.
           """
           if length <= 0:
               raise ValueError(f"length deve essere > 0, ricevuto {length}")
           n = len(closes)
           upper = [None]*n; middle = [None]*n; lower = [None]*n; bbw = [None]*n
           squeeze: list[bool | None] = [None]*n
           squeeze_ttm: list[bool | None] = [None]*n
           # Middle = SMA(length); std via cumulative second-moment
           sma_vals = sma(closes, length)
           # Population std with sliding window
           if n >= length:
               # precompute sum of squares
               sq = [c*c for c in closes]
               cum_sq = sum(sq[:length])
               cum = sum(closes[:length])
               for i in range(length-1, n):
                   if i > length-1:
                       cum += closes[i] - closes[i-length]
                       cum_sq += sq[i] - sq[i-length]
                   mean = cum / length
                   var = max(cum_sq/length - mean*mean, 0.0)
                   sd = var ** 0.5
                   middle[i] = mean
                   upper[i] = mean + std * sd
                   lower[i] = mean - std * sd
                   if mean != 0:
                       bbw[i] = (upper[i] - lower[i]) / mean
                   else:
                       bbw[i] = None
           # Percentile-squeeze over trailing squeeze_lookback_bars
           for i in range(n):
               if bbw[i] is None or i < length-1 + squeeze_lookback_bars - 1:
                   continue
               window_vals = [v for v in bbw[i - squeeze_lookback_bars + 1 : i + 1] if v is not None]
               if len(window_vals) < squeeze_lookback_bars:
                   continue
               sorted_vals = sorted(window_vals)
               # 25th percentile via nearest-rank
               cutoff_idx = max(0, int(len(sorted_vals) * squeeze_pct / 100.0) - 1)
               cutoff = sorted_vals[cutoff_idx]
               squeeze[i] = bbw[i] < cutoff
           # TTM squeeze (BB inside Keltner) — only if highs/lows provided
           if highs is not None and lows is not None:
               kc = keltner(highs, lows, closes, length=keltner_length, scalar=keltner_scalar)
               for i in range(n):
                   if upper[i] is None or kc.upper[i] is None:
                       continue
                   squeeze_ttm[i] = (upper[i] < kc.upper[i]) and (lower[i] > kc.lower[i])
           return BollingerResult(upper=upper, middle=middle, lower=lower,
                                  bbw=bbw, squeeze=squeeze, squeeze_ttm=squeeze_ttm)

       def keltner(
           highs: list[float],
           lows: list[float],
           closes: list[float],
           length: int = 20,
           scalar: float = 2.0,
       ) -> KeltnerResult:
           """Canale di Keltner: middle = EMA(length, close); upper/lower = middle ± scalar·ATR(length)."""
           if not (len(highs) == len(lows) == len(closes)):
               raise ValueError("highs, lows, closes devono avere la stessa lunghezza")
           n = len(closes)
           middle = ema(closes, length)
           atr_vals = atr(highs, lows, closes, length)
           upper: list[float | None] = [None]*n
           lower: list[float | None] = [None]*n
           for i in range(n):
               if middle[i] is None or atr_vals[i] is None:
                   continue
               upper[i] = middle[i] + scalar * atr_vals[i]
               lower[i] = middle[i] - scalar * atr_vals[i]
           return KeltnerResult(upper=upper, middle=middle, lower=lower)
       ```
