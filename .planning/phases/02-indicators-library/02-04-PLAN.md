    ```python
    """Esponente di Hurst — analisi R/S su finestra rolling.

    Convenzione: window=100, sub-windows [10, 20, 40, 80], OLS log-log per stimare H.
    H > 0.5 = persistente/trending; H < 0.5 = anti-persistente/mean-reverting; H ≈ 0.5 = random walk.
    """
    from __future__ import annotations
    import math
    from dataclasses import dataclass


    @dataclass
    class HurstResult:
        hurst: list[float | None]
        window: int


    def _rs_for_subwindow(series: list[float], n: int) -> float | None:
        """R/S medio sui chunk di lunghezza n in `series`. None se serie troppo corta o std nulla."""
        N = len(series)
        if N < n or n < 2:
            return None
        chunks = N // n
        rs_values: list[float] = []
        for k in range(chunks):
            chunk = series[k*n:(k+1)*n]
            mean = sum(chunk) / n
            dev = [v - mean for v in chunk]
            cum = []
            running = 0.0
            for d in dev:
                running += d
                cum.append(running)
            R = max(cum) - min(cum)
            var = sum(d*d for d in dev) / n
            S = math.sqrt(var)
            if S == 0:
                continue
            rs_values.append(R / S)
        if not rs_values:
            return None
        return sum(rs_values) / len(rs_values)


    def _hurst_single_window(window_values: list[float], sub_sizes: list[int]) -> float | None:
        """OLS slope of log(R/S) vs log(n) over sub-window sizes."""
        log_n: list[float] = []
        log_rs: list[float] = []
        for n in sub_sizes:
            rs = _rs_for_subwindow(window_values, n)
            if rs is None or rs <= 0:
                continue
            log_n.append(math.log(n))
            log_rs.append(math.log(rs))
        if len(log_n) < 2:
            return None
        # OLS slope
        m = len(log_n)
        mean_x = sum(log_n)/m
        mean_y = sum(log_rs)/m
        num = sum((log_n[i]-mean_x)*(log_rs[i]-mean_y) for i in range(m))
        den = sum((log_n[i]-mean_x)**2 for i in range(m))
        if den == 0:
            return None
        return num/den


    def hurst_rs(values: list[float], window: int = 100) -> HurstResult:
        """INDIC-12: Hurst R/S su finestra rolling. Default window=100 (Pitfall 6: <100 instabile)."""
        if window < 20:
            raise ValueError(f"window deve essere >= 20 per stabilità R/S, ricevuto {window}")
        n = len(values)
        out: list[float | None] = [None] * n
        # Sub-window sizes: powers of 2 fitting in `window`
        sub_sizes = [s for s in [10, 20, 40, 80] if s < window]
        if not sub_sizes:
            return HurstResult(hurst=out, window=window)
        for i in range(window - 1, n):
            wv = values[i - window + 1 : i + 1]
            out[i] = _hurst_single_window(wv, sub_sizes)
        return HurstResult(hurst=out, window=window)
    ```
