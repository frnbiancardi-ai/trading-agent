    ```python
    def draft_to_trade_proposal(
        draft: ProposalDraft,
        symbol: str,
        timeframe: str,
    ) -> "TradeProposal":
        """Converte un ProposalDraft READY in TradeProposal per il path live (D-03).

        Solleva ValueError se draft non è READY o manca direction/prezzi.
        """
        from models import TradeProposal  # import locale per evitare circular import

        if draft.setup_type != "READY":
            raise ValueError(
                f"draft_to_trade_proposal richiede setup_type=READY, ricevuto {draft.setup_type}"
            )
        if draft.direction is None or draft.entry_price is None \
                or draft.stop_loss_price is None or draft.take_profit_price is None:
            raise ValueError("draft READY deve avere direction + entry + sl + tp non-None")

        return TradeProposal(
            symbol=symbol,
            direction=draft.direction,
            entry_price=draft.entry_price,
            stop_loss_price=draft.stop_loss_price,
            take_profit_price=draft.take_profit_price,

            rationale=draft.reason,
        )


    def draft_to_technical_setup(
        draft: ProposalDraft,
        symbol: str,
        timeframe: str,
    ) -> "TechnicalSetup":
        """Converte qualsiasi ProposalDraft (READY/FORMING/NONE) in TechnicalSetup per il path scheduler.

        Mappa i campi del draft sui field names di TechnicalSetup:
          - draft.stop_loss_price  → TechnicalSetup.stop_loss
          - draft.take_profit_price → TechnicalSetup.take_profit
        Il dict `indicators` riceve i meta del draft (factors, grade, rationale_parts).
        """
        from models import TechnicalSetup

        return TechnicalSetup(
            symbol=symbol,
            timeframe=timeframe,
            setup_type=draft.setup_type,
            direction=draft.direction,
            entry_price=draft.entry_price,
            stop_loss=draft.stop_loss_price,
            take_profit=draft.take_profit_price,
            confidence=draft.confidence,
            reason=draft.reason,
            indicators={
                "factors": dict(draft.factors or {}),
                "grade": draft.grade,
                "setup_name": draft.setup_name,
                "rationale_parts": dict(draft.rationale_parts or {}),
            },
            support_resistance=None,
        )


    def rr_meets_profile_floor(
        entry: float,
        sl: float,
        tp: float,
        direction: str,
        profile: str,
        cfg=None,
    ) -> tuple[bool, float]:
        """Verifica se R:R >= profile_filters[profile].min_rr (D-10, STRAT-07).

        Ritorna (passes, rr_value). rr_value è il rapporto calcolato (TP-distance / SL-distance).
        """
        from strategy.confluence import load_strategy_config
        cfg = cfg or load_strategy_config()
        if profile not in cfg.profile_filters:
            raise ValueError(f"profile {profile!r} non in profile_filters: {list(cfg.profile_filters)}")
        min_rr = float(cfg.profile_filters[profile]["min_rr"])

        if direction == "BUY":
            risk = entry - sl
            reward = tp - entry
        elif direction == "SELL":
            risk = sl - entry
            reward = entry - tp
        else:
            raise ValueError(f"direction deve essere BUY/SELL, ricevuto {direction!r}")
        if risk <= 0 or reward <= 0:
            return False, 0.0
        rr = reward / risk
        return (rr >= min_rr), round(rr, 4)


    def compute_levels_with_atr_cap(
        direction: str,
        entry: float,
        structural_level: float,
        atr: float,
        buffer_atr_mult: float,
        cap_atr_mult: float = 1.5,
    ) -> tuple[float, float]:
        """SL universale per tutti i setup (D-10): livello strutturale ± buffer×ATR, capato a cap×ATR.

        Ritorna (entry, sl). TP è specifico per setup → calcolato dal detector.
        """
        if atr is None or atr <= 0:
            raise ValueError(f"atr deve essere > 0, ricevuto {atr!r}")
        buf = buffer_atr_mult * atr
        cap = cap_atr_mult * atr
        if direction == "BUY":
            sl = structural_level - buf
            sl = max(sl, entry - cap)        # cap ATR enforced
        elif direction == "SELL":
            sl = structural_level + buf
            sl = min(sl, entry + cap)
        else:
            raise ValueError(f"direction deve essere BUY/SELL, ricevuto {direction!r}")
        return entry, sl
    ```
