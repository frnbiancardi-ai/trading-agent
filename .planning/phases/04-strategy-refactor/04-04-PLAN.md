    ```python
    """Test AST purity gate — moduli pure-fn non devono importare broker/log/db/print (STRAT-08, D-15).

    Esegue introspezione AST sui 7 moduli pure-fn della strategy package e verifica:
      1. nessun import vietato (mt5, MetaTrader5, requests, sqlite3, subprocess, logging)
      2. nessuna chiamata a logging.getLogger / logger.info / logger.warning / etc.
      3. nessuna chiamata a print() o open() (eccezione: confluence.py usa open() per yaml load)
    """
    from __future__ import annotations
    import ast
    from pathlib import Path

    import pytest

    # Moduli che DEVONO restare puri.
    PURE_MODULES = [
        "strategy/setups/a_breakout.py",
        "strategy/setups/b_reversal.py",
        "strategy/setups/c_compression.py",
        "strategy/setups/d_pullback.py",
        "strategy/setups/__init__.py",
        "strategy/confluence.py",
        "strategy/proposal.py",
        "strategy/context.py",
    ]

    FORBIDDEN_IMPORTS = {
        "mt5", "MetaTrader5", "requests", "sqlite3", "subprocess",
        "urllib", "urllib2", "urllib3", "httpx", "aiohttp",
        "logging",
    }

    # File che hanno eccezioni motivate (loader yaml singleton).
    OPEN_EXEMPT_FILES = {"strategy/confluence.py"}


    def _parse(module_path: str) -> tuple[ast.Module, str]:
        """Carica un file e ritorna (AST tree, source). Solleva FileNotFoundError se mancante."""
        path = Path(module_path)
        if not path.exists():
            pytest.fail(f"modulo pure-fn mancante: {module_path}")
        source = path.read_text(encoding="utf-8")
        try:
            return ast.parse(source, filename=module_path), source
        except SyntaxError as e:
            pytest.fail(f"{module_path}: SyntaxError {e}")


    def _import_root(name: str | None) -> str:
        """Estrae il primo segmento di un nome di import (es. 'urllib.request' → 'urllib')."""
        if name is None:
            return ""
        return name.split(".")[0]


    # ------------------------------------------------------------------------

    def test_strategy_purity_no_forbidden_imports():
        """Nessun import vietato (broker, http, db, logging) nei moduli pure-fn."""
        violations: list[str] = []
        for mod_path in PURE_MODULES:
            tree, _ = _parse(mod_path)
            for node in ast.walk(tree):
                if isinstance(node, ast.Import):
                    for alias in node.names:
                        root = _import_root(alias.name)
                        if root in FORBIDDEN_IMPORTS:
                            violations.append(
                                f"{mod_path}:{node.lineno}: forbidden 'import {alias.name}'"
                            )
                elif isinstance(node, ast.ImportFrom):
                    root = _import_root(node.module)
                    if root in FORBIDDEN_IMPORTS:
                        violations.append(
                            f"{mod_path}:{node.lineno}: forbidden 'from {node.module} import ...'"
                        )
        assert not violations, "Pure modules contengono import vietati:\n  " + "\n  ".join(violations)


    def test_strategy_purity_no_logging_calls():
        """Nessuna chiamata a logging.* o a metodi di un Logger (.info/.warning/.error/.debug)."""
        violations: list[str] = []
        LOG_METHODS = {"info", "warning", "error", "debug", "critical", "exception", "log", "getLogger"}
        for mod_path in PURE_MODULES:
            tree, _ = _parse(mod_path)
            for node in ast.walk(tree):
                if isinstance(node, ast.Call):
                    func = node.func
                    # logging.getLogger(...) o logging.info(...)
                    if isinstance(func, ast.Attribute):
                        attr = func.attr
                        # caso a: logging.<method>(...)
                        if isinstance(func.value, ast.Name) and func.value.id == "logging":
                            violations.append(f"{mod_path}:{node.lineno}: forbidden 'logging.{attr}(...)'")
                        # caso b: <name>.info/warning/error/debug — heuristica nome che contiene 'log' o 'logger'
                        elif (
                            attr in LOG_METHODS
                            and isinstance(func.value, ast.Name)
                            and ("log" in func.value.id.lower())
                        ):
                            violations.append(
                                f"{mod_path}:{node.lineno}: forbidden '{func.value.id}.{attr}(...)' (logger call)"
                            )
        assert not violations, "Pure modules contengono chiamate logging:\n  " + "\n  ".join(violations)


    def test_strategy_purity_no_print_calls():
        """Nessuna chiamata a print() o open() nei moduli puri (eccezione: confluence.py usa open per yaml)."""
        violations: list[str] = []
        for mod_path in PURE_MODULES:
            tree, _ = _parse(mod_path)
            for node in ast.walk(tree):
                if isinstance(node, ast.Call):
                    func = node.func
                    if isinstance(func, ast.Name):
                        if func.id == "print":
                            violations.append(f"{mod_path}:{node.lineno}: forbidden 'print(...)'")
                        elif func.id == "open" and mod_path not in OPEN_EXEMPT_FILES:
                            violations.append(
                                f"{mod_path}:{node.lineno}: forbidden 'open(...)' (not exempt; only {OPEN_EXEMPT_FILES} allowed)"
                            )
        assert not violations, "Pure modules contengono print/open vietati:\n  " + "\n  ".join(violations)


    def test_pure_modules_all_exist():
        """Sanity: tutti i 7 moduli puri devono esistere prima di testare la purity."""
        missing = [m for m in PURE_MODULES if not Path(m).exists()]
        assert not missing, f"moduli puri mancanti: {missing}"
    ```

    Notes:
    - Test 4 (`test_pure_modules_all_exist`) is a sanity check that fails FIRST if any of the 7 modules is missing — this gives a clear error before Test 1/2/3 hit a pytest.fail.
    - The logging detection uses a name-heuristic ("log" in variable name) to catch `self.log.info(...)` patterns. It does NOT cover `getLogger(...)` returns assigned to non-log-named variables — this is acceptable; the bigger gate is the `import logging` block in Test 1.
    - We allow `open()` in confluence.py because the yaml singleton loader needs it (D-08 schema load). All other pure modules MUST NOT touch the filesystem.
