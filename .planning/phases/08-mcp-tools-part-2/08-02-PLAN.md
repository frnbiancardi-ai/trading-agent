def handle_predict_trade_quality(
    args: dict,
    ml_filter_singleton: Any,
    mt5_client: Any,
    cfg: Any,
) -> dict:
    """MCP-05 — D-12 wrapper read-only di MLFilter.predict.

    NON e' un gate decisione (risk_engine resta unico gate, CLAUDE.md).

    Branche:
    1. ENABLE_ML_FILTER=false OR singleton=None: ml_* tutti null + ok:true +
       warning='ml_filter_disabled' (D-07-06-11 zero-impact rollout — graceful,
       no error envelope).
    2. ENABLE_ML_FILTER=true + singleton attivo + predict OK: ml_score raw,
       calibrated_prob, threshold_for_profile, would_pass_gate (D-12 + D-14).
    3. Eccezione durante build_feature_vector OR singleton.predict:
       envelope(INTERNAL_ERROR, str(exc)). Distinto da branca 1 per audit.
    """
    if not getattr(cfg, "ENABLE_ML_FILTER", False) or ml_filter_singleton is None:
        return {
            "ok": True,
            "ml_score": None,
            "calibrated_prob": None,
            "threshold_for_profile": None,
            "would_pass_gate": None,
            "warning": "ml_filter_disabled",
        }

    try:
        # Lazy import per non agganciare ml.* a module-load (Phase 7 execute
        # pre-condizione — durante Wave 0 questa import non viene mai eseguita).
        from ml.feature_extraction import build_feature_vector

        draft_like = _make_draft_from_args(args)
        ctx_like = _make_ctx_from_args(args, cfg)
        indicators_obj = _resolve_indicators(args, mt5_client, cfg)

        X = build_feature_vector(
            draft_like,
            indicators_obj,
            ctx_like,
            feature_list=ml_filter_singleton.features,
            cat_encodings=ml_filter_singleton.categorical_encodings,
        )
        raw, calibrated = ml_filter_singleton.predict(X)

        profile_up = args["profile"].upper()
        threshold = float(
            ml_filter_singleton.threshold_by_profile.get(profile_up, 0.5)
        )
        return {
            "ok": True,
            "ml_score": float(raw),
            "calibrated_prob": float(calibrated),
            "threshold_for_profile": threshold,
            "would_pass_gate": bool(calibrated >= threshold),
        }
    except Exception as exc:
        # D-08 audit: predict failure distinto da disabled.
        # Pattern Phase 6 envelope error_code.
        return envelope(
            ErrorCodes.INTERNAL_ERROR,
            f"predict_trade_quality failed: {type(exc).__name__}: {exc}",
            tool="predict_trade_quality",
        )

def _make_draft_from_args(args: dict) -> Any:
    """Costruisce un ProposalDraft-like SimpleNamespace dagli args MCP.

    Riusa attribute access shape di ProposalDraft (Phase 4 D-03 + Phase 7 D-13)
    per essere compatibile con build_feature_vector che fa attribute lookup.

    standalone (skill chiama predict_trade_quality SENZA aver passato dal
    detector pipeline; D-09-G derivation farà fallback dove possibile).
    """
    from types import SimpleNamespace
    return SimpleNamespace(
        setup_type="READY",
        setup_name=args.get("setup_name", "unknown"),
        pattern_name=args.get("pattern_name", "unknown"),
        bias=args["direction"],
        entry_price=float(args["entry_price"]),
        stop_loss_price=float(args["stop_loss"]),
        take_profit_price=float(args["take_profit"]),

def _compute_rr_from_args(args: dict) -> float:
    """R:R approssimato per il draft_like.

    BUY: rr = (tp - entry) / (entry - sl)
    SELL: rr = (entry - tp) / (sl - entry)
    Returns 0.0 se denominatore <= 0 (input malformed; predict tollera).
    """
    entry = float(args["entry_price"])
    sl = float(args["stop_loss"])
    tp = float(args["take_profit"])
    direction = args["direction"].upper()
    if direction == "BUY":
        risk = entry - sl
        reward = tp - entry
    else:
        risk = sl - entry
        reward = entry - tp
    return float(reward / risk) if risk > 1e-9 else 0.0


def _make_ctx_from_args(args: dict, cfg: Any) -> Any:
    """Costruisce StrategyContext-like dagli args (ctx-side, NON indicators-side).

    B4 revision iter 1 — disambiguazione regime/regime_state:
    - ctx-side: `regime` canonical (post-rename Phase 7 D-07-01-G + Plan 07-03 line 1243).
      build_feature_vector legge `ctx.regime` (NON `ctx.regime_state`).
    - indicators-side: `regime_state` schema-v2 (NON in ctx; vedi _resolve_indicators).

    `regime` canonical per Phase 7: fonte preferita args.context.regime; fallback
    args.context.regime_state (renamed). NON duplicare entrambi in ctx — il
    build_feature_vector contract Phase 7 chiede 1 sola key. Se mancano entrambi
    → "normal" (default safe).

    args.context (optional) puo' contenere anche numeric carry-through (es. atr_pct,
    spread_pips) che ctx espone come attribute per build_feature_vector. NB:
    regime_state se presente NON viene esposto su ctx — vive solo su indicators side.
    """
    from types import SimpleNamespace
    context_in = args.get("context") or {}
    if not isinstance(context_in, dict):
        context_in = {}
    # B4 — regime canonical: prefer context.regime, fallback regime_state (rename).
    # NON duplicare regime_state in ctx (Phase 7 D-07-01-G; vedi 07-03-PLAN.md:1243).
    regime_canonical = (
        context_in.get("regime")
        or context_in.get("regime_state")
        or "normal"
    )
    # Numeric carry-through: tutte le altre keys, ESCLUSE regime/regime_state
    # (regime_state vive solo su indicators side — _resolve_indicators).
    numeric_carry = {
        k: v for k, v in context_in.items()
        if k not in ("regime", "regime_state")
    }
    return SimpleNamespace(
        symbol=args["symbol"],
        timeframe=args["timeframe"],
        profile=args["profile"].upper(),
        regime=regime_canonical,
        **numeric_carry,
    )

    B4 revision iter 1 — separazione indicators-side vs ctx-side:

    Decisione A vs B: presenza di args.context["atr"] (key obbligatoria).
    """
    from types import SimpleNamespace
    context_in = args.get("context") or {}
    if isinstance(context_in, dict) and "atr" in context_in:
        # Branca A: trust skill upstream — valida regime_state present
        indicators_dict = dict(context_in)
        # B4 — assicura regime_state present per build_feature_vector lookup
        # (Phase 5 schema-v2 lock). Fallback rename da legacy `regime`.
        if "regime_state" not in indicators_dict:

        # (Phase 5 dataset_writer.py:28 anti-alias rule).

        return SimpleNamespace(**indicators_dict)
    # Branca B: fetch MT5 fresh (schema-v2 native — get_market_snapshot include
    # regime_state computato da compute_all_extended Phase 2).
    from mcp_tools.handlers.market import handle_get_market_snapshot
    snap = handle_get_market_snapshot(
        {"symbol": args["symbol"], "timeframe": args["timeframe"]},
        mt5_client,
        cfg,
    )
    indicators_extended = snap.get("indicators_extended") or {}
    # Defensive: drop `regime` alias se snapshot legacy lo emette
    if "regime" in indicators_extended and "regime_state" not in indicators_extended:
        indicators_extended = dict(indicators_extended)
        indicators_extended["regime_state"] = indicators_extended.pop("regime")
    return SimpleNamespace(**indicators_extended)
