```python
def handle_evaluate_trade_proposal(
    args: dict,
    mt5_client,
    cfg,
    ml_filter_singleton=None,  # Phase 8 additive: MLFilter | None
) -> dict:
    """Phase 8 D-08 + MCP-R4: extension additive ml_score + calibrated_prob.

    Backward-compat absolute (forex-trader-pro skill consumer):
    - Shape legacy preservata: tutti i fields di RiskDecision.asdict()
    - 4 nuovi fields additive in coda: ml_score, calibrated_prob,
      ml_threshold, ml_filter_active
    - ENABLE_ML_FILTER=false OR singleton=None → 4 fields = None/None/None/False,
      NO error envelope (graceful per Plan 07-06 D-07-06-C pattern try/except swallow)

    Phase 8 NON cambia la decisione (CLAUDE.md "risk_engine = unico gate") —
    espone solo introspection ML allineata al singleton. La decisione reale
    risk_engine.evaluate_trade ha gia' applicato il gate ML (Phase 7 D-12).
    """
    from risk_engine import evaluate_trade

    proposal = _build_proposal_from_args(args, cfg)
    account = mt5_client.get_account_state()
    decision = evaluate_trade(proposal, account, mt5_client, cfg)
    base = dataclasses.asdict(decision)

    # ── Phase 8 additive extension ─────────────────────────────────────
    ml_score = None
    calibrated_prob = None
    ml_threshold = None
    ml_filter_active = False

    if (
        getattr(cfg, "ENABLE_ML_FILTER", False)
        and ml_filter_singleton is not None
    ):
        ml_filter_active = True
        try:
            # Riuso helper Plan 08-02 (mcp_tools/handlers/ml.py)
            from mcp_tools.handlers.ml import (
                _make_draft_from_args,
                _make_ctx_from_args,
                _resolve_indicators,
            )
            from ml.feature_extraction import build_feature_vector

            # MCP-R4 input args differiscono leggermente da predict_trade_quality:
            # qui args.stop_loss_price / take_profit_price (Phase 6 R3 shape).
            # Mapping ai nomi usati dagli helper Plan 08-02.
            args_mapped = {
                "symbol": args["symbol"],
                "timeframe": args.get("timeframe", cfg.TIMEFRAME),
                "profile": getattr(cfg, "RISK_MODE", "MODERATE"),
                "direction": args["direction"],
                "entry_price": args["entry_price"],
                "stop_loss": args["stop_loss_price"],
                "take_profit": args["take_profit_price"],
                "context": args.get("context") or {},
            }

            draft_like = _make_draft_from_args(args_mapped)
            ctx_like = _make_ctx_from_args(args_mapped, cfg)
            indicators_obj = _resolve_indicators(args_mapped, mt5_client, cfg)

            X = build_feature_vector(
                draft_like,
                indicators_obj,
                ctx_like,
                feature_list=ml_filter_singleton.features,
                cat_encodings=ml_filter_singleton.categorical_encodings,
            )
            raw, calibrated = ml_filter_singleton.predict(X)
            ml_score = float(raw)
            calibrated_prob = float(calibrated)
            ml_threshold = float(
                ml_filter_singleton.threshold_by_profile.get(
                    str(getattr(cfg, "RISK_MODE", "MODERATE")).upper(),
                    0.5,
                )
            )
        except Exception:
            # Pattern Plan 07-06 D-07-06-C: swallow eccezione,
            # ml_* restano None, ml_filter_active resta True per audit
            # (handler ha provato ma fallito → distinto da "disabled").
            # NB: NON ritorna envelope error — extension graceful absolute.
            pass

    base.update({
        "ml_score": ml_score,
        "calibrated_prob": calibrated_prob,
        "ml_threshold": ml_threshold,
        "ml_filter_active": ml_filter_active,
    })
    return base
```
