```python
def handle_get_ml_calibration(
    args: dict,
    cfg: Any,
) -> dict:
    """MCP-06 — D-08-C1 default audit-first per-fold; D-08-C2 summary_only opt-out.

    Default response (`summary_only=False`, ~30-50KB):
    {
        "ok": True,
        "model_version": "v1",
        "trained_at": "2026-05-12",
        "threshold_by_profile": {"CONSERVATIVE": 0.55, ...},
        "folds": [
            {fold_n, brier, ece, reliability_data, calibrator_winner,
             val_size, val_too_small_flag, n_train, n_val}, ... x10
        ],
        "aggregate": {"brier": ..., "ece": ..., "reliability_data": [...]}
    }

    summary_only=True response (~3-5KB):
    {
        "ok": True,
        "model_version": ...,
        "trained_at": ...,
        "threshold_by_profile": {...},
        "aggregate": {...}
    }

    Error paths:
    - metadata.json mancante: envelope(NOT_FOUND, f"metadata.json mancante: {path}")
    - JSON parse fail: envelope(INTERNAL_ERROR, f"metadata.json corrupt: {exc}")
    """
    from pathlib import Path
    import json

    bundle_path = Path(getattr(cfg, "ML_MODEL_PATH", "models/classifier_v1_latest.pkl"))
    meta_path = bundle_path.with_suffix(".metadata.json")
    if not meta_path.exists():
        return envelope(
            ErrorCodes.NOT_FOUND,
            f"metadata.json mancante: {meta_path} (modello non addestrato?)",
            tool="get_ml_calibration",
        )

    try:
        metadata = json.loads(meta_path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        return envelope(
            ErrorCodes.INTERNAL_ERROR,
            f"metadata.json corrupt: {exc}",
            tool="get_ml_calibration",
        )

    summary_only = bool(args.get("summary_only", False))

    # Common fields (D-08-C2 minimo)
    common = {
        "ok": True,
        "model_version": metadata.get("version"),
        "trained_at": metadata.get("train_date"),
        "threshold_by_profile": metadata.get("threshold_by_profile", {}),
        "aggregate": metadata.get("aggregate_reliability", {}),
    }

    if summary_only:
        return common

    # Default D-08-C1: per-fold breakdown completo (audit-first)
    folds_raw = metadata.get("fold_metrics", [])
    common["folds"] = [_fold_to_payload(f) for f in folds_raw]
    return common


def _fold_to_payload(fold: dict) -> dict:
    """D-08-C3 — Estrae 9 fields per-fold dal metadata.json sidecar.

    Fields enumerati:
        fold_n: int 1-10
        brier: float
        ece: float
        reliability_data: list[{prob_bin_mid, observed_freq, count}]
        calibrator_winner: str enum {"sigmoid", "isotonic"}
        val_size: int
        val_too_small_flag: bool (true iff val_size < 50 → Platt-only forzato
            HANDOFF blocker #3 Phase 7)
        n_train: int
        n_val: int

    Defensive: se metadata fold schema ha key 'fold' invece di 'fold_n'
    (Phase 7 Plan 07-03 lines 350-359 produce 'fold' int 1-10), accept both
    aliases per backward-compat.
    """
    fold_n = fold.get("fold_n", fold.get("fold"))
    val_size = int(fold.get("val_size", fold.get("n_val", 0)))
    return {
        "fold_n": int(fold_n) if fold_n is not None else None,
        "brier": float(fold.get("brier", 0.0)),
        "ece": float(fold.get("ece", 0.0)),
        "reliability_data": fold.get("reliability_data", []),
        "calibrator_winner": fold.get("calibrator_winner", "unknown"),
        "val_size": val_size,
        "val_too_small_flag": bool(fold.get(
            "val_too_small_flag",
            val_size < 50,
        )),
        "n_train": int(fold.get("n_train", 0)),
        "n_val": int(fold.get("n_val", val_size)),
    }
```
