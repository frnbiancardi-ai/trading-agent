```python
def _sortable_utc_ts() -> str:
    """Timestamp UTC sortable per run_id (analog backtest.py:36-42).

    Returns YYYYMMDDTHHMMSSZ — string ordinabile lexicograficamente.
    """
    from datetime import datetime, timezone
    return datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")


def _build_ml_run_id() -> str:
    """Build run_id univoco per ML training job.

    Format: mcp_ml_<utc_ts>
    Analog backtest.py:45-51 ma senza symbol/tf/profile (training e' globale).
    """
    return f"mcp_ml_{_sortable_utc_ts()}"


def _validate_parquet_schema_v2(parquet_path) -> tuple[bool, str, str]:
    """D-08-B3 strict-fail pre-training validation + sha256 audit trail.

    Verifica:
    1. parquet_path esiste come file
    2. T-8-04-04 security: parquet_path.resolve().is_relative_to(
       Path('data/training').resolve()) — blocco path traversal env-driven
       (MCP_TRAINING_DATA_PATH potrebbe puntare a path arbitrario)
    3. pyarrow load successful (corruption check)
    4. n_rows >= MIN_TRAINING_ROWS (1000, HANDOFF SC#3 hard gate)
    5. _SCHEMA_V2_REQUIRED_KEYS subset di dataset.schema.names
    6. sha256 computed via file_sha256(parquet_path)

    Returns:
        (ok: bool, message: str, sha256_hex: str)
        ok=False → message contiene reason; sha256_hex="" su fail
        ok=True → message="OK schema-v2: N rows, sha256=...", sha256_hex 64-char

    NB: import lazy interni per non agganciare pyarrow/dataset_writer a
    module-load (handler caricato anche quando non chiamato).
    """
    from pathlib import Path
    import pyarrow.dataset as ds

    parquet_path = Path(parquet_path)
    if not parquet_path.exists():
        return False, f"parquet mancante: {parquet_path}", ""

    # T-8-04-04 security: path-traversal mitigation
    try:
        resolved = parquet_path.resolve()
        training_root = Path("data/training").resolve()
        if not resolved.is_relative_to(training_root):
            return (
                False,
                f"path security violation: {resolved} outside data/training/",
                "",
            )
    except (ValueError, OSError) as exc:
        return False, f"path resolution failed: {exc}", ""

    # pyarrow load + schema check
    try:
        dataset = ds.dataset(str(parquet_path), format="parquet")
    except Exception as exc:
        return False, f"pyarrow load fallito: {exc}", ""

    n_rows = dataset.count_rows()
    # MIN_TRAINING_ROWS = 1000 (HANDOFF SC#3 hard gate, Plan 05-09)
    if n_rows < 1000:
        return (
            False,
            f"n_rows {n_rows} < MIN_TRAINING_ROWS 1000 (SC#3 hard gate)",
            "",
        )

    # Schema v2 check
    from backtest.baseline.dataset_writer import _SCHEMA_V2_REQUIRED_KEYS
    schema_names = set(dataset.schema.names)
    missing = _SCHEMA_V2_REQUIRED_KEYS - schema_names
    if missing:
        # Truncate per evitare message gigantesco
        sample = sorted(missing)[:10]
        return (
            False,
            f"missing cols schema-v2: {sample} (total {len(missing)})",
            "",
        )

    # D-08-B3 audit: sha256 del file fisico
    from backtest.baseline.determinism import file_sha256
    sha = file_sha256(parquet_path)
    return (
        True,
        f"OK schema-v2: {n_rows} rows, sha256={sha[:12]}...",
        sha,
    )
```

```python
def _train_ml_filter_worker(
    run_id: str,
    parquet_path: str,
    output_dir: str,
    ml_yaml_path: str,
    parquet_sha256: str,
) -> dict:
    """Worker process: training LightGBM, NO mt5.

    TOP-LEVEL function (picklable per ProcessPoolExecutor).

    Carry-forward constraints (Phase 6 Plan 06-03 + Phase 5 D-15):
    - Pitfall 1: NON importare MetaTrader5 / mt5_client (lib non fork-safe)
    - Pitfall 8: NON usare stdout (corromperebbe MCP JSON-RPC framing).
      Logger su file via init_logger (RotatingFileHandler — safe).
    - D-08-A3: fire-and-forget, NO cancel handling (training va in fondo).

    Args:
        run_id: identificatore univoco mcp_ml_<utc_ts>
        parquet_path: data/training/baseline_decisions/part-0.parquet (validated upstream)
        output_dir: directory destinazione bundle.pkl (es. "models/")
        ml_yaml_path: data/configs/ml.yaml (Phase 7 Plan 07-01 config)
        parquet_sha256: hash 64-char audit trail (validated upstream)

    Returns:
        dict con keys: run_id, bundle_path, metadata_path, n_folds, parquet_sha256.
        Su eccezione: JobQueue cattura tramite future.exception() e marca
        backtest_runs.status='failed' + error_message (Plan 06-03 _on_done).
    """
    # Lazy imports interni (Pitfall 1 + module-load perf)
    import dataclasses
    from datetime import date
    from pathlib import Path

    from logger import init_logger
    from config import Config

    cfg = Config()
    log = init_logger(cfg)  # RotatingFileHandler file-only — Pitfall 8 safe

    log.info(
        "ml worker start: run_id=%s parquet=%s sha256=%s...",
        run_id, parquet_path, parquet_sha256[:12],
    )

    # Lazy ml import (Phase 7 surface NON disponibile a module-load
    # finche' Phase 7 plan-execute non e' completo).
    from ml.train import train_classifier, load_ml_training_config

    ml_cfg = load_ml_training_config(Path(ml_yaml_path))
    # Override dataset_path con quello passato (D-08-B1 baseline hardcoded)
    # NB: MLTrainingConfig e' frozen dataclass (Plan 07-03 line 280) → use replace.
    ml_cfg = dataclasses.replace(ml_cfg, dataset_path=parquet_path)

    fold_results = train_classifier(ml_cfg)
    # train_classifier salva bundle a output_dir/classifier_v1_{date}.pkl
    # + sidecar .metadata.json (Plan 07-05 D-15 path convention).
    today = date.today().isoformat()
    bundle_path = Path(output_dir) / f"classifier_v1_{today}.pkl"
    meta_path = bundle_path.with_suffix(".metadata.json")

    # B5 revision iter 1 — audit trail D-08-B3: inject training_data_sha256
    # nel metadata.json REALE scritto da train_classifier. Phase 7 Plan 07-05
    # metadata schema include `dataset_hash` (calcolato internamente da
    # train_classifier dal proprio parquet load), NON expose hook per
    # external sha256 injection. Plan 08-04 audit-first richiede che il sha
    # calcolato in _validate_parquet_schema_v2 (pre-flight) sia REGISTRATO
    # in metadata.json: garanzia di rispecchiabilità "il bundle e' stato
    # addestrato su questo specifico parquet, verificato pre-training".
    #
    # Write atomico: scrivi su .tmp poi os.replace per evitare corruzione
    # in caso di crash mid-write (POSIX rename atomic, Windows analog).
    import json
    import os
    if meta_path.exists():
        try:
            meta = json.loads(meta_path.read_text(encoding="utf-8"))
            # Inject anche se la key gia' presente (sovrascrivi con il sha
            # passato da pre-flight — fonte di verita' del Plan 08-04 audit).
            meta["training_data_sha256"] = parquet_sha256
            tmp_path = meta_path.with_suffix(".metadata.json.tmp")
            tmp_path.write_text(
                json.dumps(meta, indent=2, ensure_ascii=False),
                encoding="utf-8",
            )
            os.replace(tmp_path, meta_path)  # atomic
            log.info(
                "metadata.json updated con training_data_sha256=%s...",
                parquet_sha256[:12],
            )
        except (json.JSONDecodeError, OSError) as exc:
            # Defensive: NON propagare l'errore (train_classifier ha gia' avuto
            # successo; questo e' audit-trail add-on). Log errore e continua.
            log.error(
                "metadata.json sha injection fallito (audit trail incomplete): %s",
                exc,
            )
    else:
        log.warning(
            "metadata.json mancante post-train (%s) — sha injection skippato",
            meta_path,
        )

    log.info(
        "ml worker done: bundle=%s n_folds=%d sha=%s...",
        bundle_path, len(fold_results), parquet_sha256[:12],
    )
    return {
        "run_id": run_id,
        "bundle_path": str(bundle_path),
        "metadata_path": str(meta_path),
        "n_folds": len(fold_results),
        "parquet_sha256": parquet_sha256,
    }
```

```python
def handle_train_ml_filter(
    args: dict,
    job_queue: Any,
    cfg: Any,
) -> dict:
    """MCP-04 — D-08-A1 async submit training.

    Pre-flight validation D-08-B3 (strict-fail):
    1. data_source=="baseline" (D-08-B2 enum only)
    2. _validate_parquet_schema_v2: path traversal + schema-v2 + n_rows>=1000 + sha256
    3. JobQueue.submit cap=1 condiviso → run_in_progress envelope se backtest gira

    Returns:
        Successo: {ok:true, run_id, status:'started', started_at}
        Validation fail: envelope(VALIDATION_FAILED, reason)
        JobQueue cap: envelope(RUN_IN_PROGRESS, active_run_id)
    """
    from pathlib import Path

    data_source = args.get("data_source", "baseline")
    if data_source != "baseline":
        return envelope(
            ErrorCodes.VALIDATION_FAILED,
            f"data_source='{data_source}' non supportato "
            f"(D-08-B2 enum: ['baseline'])",
            tool="train_ml_filter",
        )

    # Pre-flight schema-v2 + sha256 + path-traversal
    parquet_path = Path(getattr(
        cfg, "MCP_TRAINING_DATA_PATH",
        "data/training/baseline_decisions/part-0.parquet",
    ))
    ok, msg, parquet_sha = _validate_parquet_schema_v2(parquet_path)
    if not ok:
        return envelope(
            ErrorCodes.VALIDATION_FAILED,
            f"parquet schema check fallito: {msg}",
            tool="train_ml_filter",
        )

    # Output dir = parent del bundle path
    bundle_path = Path(getattr(
        cfg, "ML_MODEL_PATH", "models/classifier_v1_latest.pkl",
    ))
    output_dir = str(bundle_path.parent)
    ml_yaml_path = "data/configs/ml.yaml"

    run_id = _build_ml_run_id()
    # JobQueue.submit ritorna {ok:true, run_id, status:'started', started_at}
    # OR {ok:false, error:'run_in_progress', active_run_id} se cap raggiunto.
    return job_queue.submit(
        run_id,
        _train_ml_filter_worker,
        run_id,
        str(parquet_path),
        output_dir,
        ml_yaml_path,
        parquet_sha,
    )
```
