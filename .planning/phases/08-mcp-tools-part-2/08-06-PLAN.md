def _g(obj: Any, key: str, default: Any = 0.0) -> Any:
    """Dual-mode access: dict.get OR getattr (analog report_writer.py:49-54)."""
    if obj is None:
        return default
    if isinstance(obj, dict):
        return obj.get(key, default)
    return getattr(obj, key, default)


def _row_side_by_side(r_ml: dict, r_base: dict | None) -> str:
    """Una riga side-by-side per slice (D-08-D4 part 3).

    Se r_base None (slice non presente nel baseline): mostra solo ml side, base="-".
    """
    sym = r_ml["symbol"]
    tf = r_ml["timeframe"]
    prof = r_ml["profile"]
    ml_n = r_ml.get("n_trades", 0)
    m_ml = r_ml.get("metrics")
    ml_sharpe = _g(m_ml, "sharpe", 0.0)
    ml_hit = _g(m_ml, "hit_rate", 0.0)
    ml_pf = _g(m_ml, "profit_factor", 0.0)

    if r_base is None:
        return (f"| {sym} | {tf} | {prof} | - | {ml_n} | - | {ml_sharpe:.3f} "
                f"| - | {ml_hit:.3f} | - | {ml_pf:.2f} | - |")

    base_n = r_base.get("n_trades", 0)
    m_base = r_base.get("metrics")
    base_sharpe = _g(m_base, "sharpe", 0.0)
    base_hit = _g(m_base, "hit_rate", 0.0)
    base_pf = _g(m_base, "profit_factor", 0.0)
    rejection = (1.0 - ml_n / base_n) if base_n > 0 else 0.0

    return (f"| {sym} | {tf} | {prof} | {base_n} | {ml_n} "
            f"| {base_sharpe:.3f} | {ml_sharpe:.3f} "
            f"| {base_hit:.3f} | {ml_hit:.3f} "
            f"| {base_pf:.2f} | {ml_pf:.2f} | {rejection:.2%} |")


# ── Aggregate Verdict (D-08-D5) ─────────────────────────────────────────

def _aggregate_metrics(results: list[dict], baseline_results: list[dict]) -> dict:
    """Computa aggregate delta tra ml e baseline."""
    base_by_key = {(r["symbol"], r["timeframe"], r["profile"]): r for r in baseline_results}
    ml_total_n = sum(r.get("n_trades", 0) for r in results if r.get("status") == "OK")
    base_total_n = sum(r.get("n_trades", 0) for r in baseline_results if r.get("status") == "OK")

    # Weighted average per trades count (più trade = più peso)
    def _wavg(metric: str, items: list[dict]) -> float:
        num = 0.0
        den = 0
        for r in items:
            if r.get("status") != "OK":
                continue
            n = r.get("n_trades", 0)
            v = _g(r.get("metrics"), metric, 0.0)
            num += v * n
            den += n
        return num / den if den > 0 else 0.0

    return {
        "ml_total_n": ml_total_n,
        "base_total_n": base_total_n,
        "trade_retention": ml_total_n / base_total_n if base_total_n > 0 else 0.0,
        "sharpe_delta": _wavg("sharpe", results) - _wavg("sharpe", baseline_results),
        "hit_rate_delta": _wavg("hit_rate", results) - _wavg("hit_rate", baseline_results),
        "expectancy_base": _wavg("expectancy_usd", baseline_results),
        "expectancy_ml": _wavg("expectancy_usd", results),
        "expectancy_delta": _wavg("expectancy_usd", results) - _wavg("expectancy_usd", baseline_results),
        "max_dd_delta": _wavg("max_drawdown_pct", results) - _wavg("max_drawdown_pct", baseline_results),
    }


def _compute_verdict(aggregate: dict) -> dict:
    """5 SC#4 criteria automated (D-08-D5).

    Criteria:
    - sharpe_delta >= +0.50
    - hit_rate_delta >= +0.05
    - trade_retention >= 30% (ml_n / base_n aggregato)
    - expectancy sign flip from neg→pos AND delta >= +5
    - max_dd_delta <= -1.0 (W3 revision iter 1 — unit clarification: max_drawdown_pct
      e' POSITIVO percentage point in metrics.py:156-170 — drawdown del 2.5% = 2.5
      NON -2.5. delta = ml_avg - base_avg. delta NEGATIVO significa "ml ha drawdown
      INFERIORE a base" = improvement. Threshold -1.0 pp = "ml almeno 1 pp meglio
      del baseline" — coerente con ROADMAP SC#4 hint).

    Status: NON_TRIVIAL_IMPROVEMENT se >=4/5; NEGATIVE_RESULT_DOCUMENTED altrimenti.
    """
    criteria = {}
    passes = 0

    # 1. sharpe_delta
    cond = aggregate["sharpe_delta"] >= 0.50
    criteria["sharpe_delta"] = {"value": aggregate["sharpe_delta"], "threshold": 0.50, "pass": cond}
    if cond:
        passes += 1

    # 2. hit_rate_delta
    cond = aggregate["hit_rate_delta"] >= 0.05
    criteria["hit_rate_delta"] = {"value": aggregate["hit_rate_delta"], "threshold": 0.05, "pass": cond}
    if cond:
        passes += 1

    # 3. trade_retention >= 30%
    cond = aggregate["trade_retention"] >= 0.30
    criteria["trade_retention"] = {"value": aggregate["trade_retention"], "threshold": 0.30, "pass": cond}
    if cond:
        passes += 1

    # 4. expectancy sign flip neg->pos + delta >= +5
    sign_flip = aggregate["expectancy_base"] < 0 and aggregate["expectancy_ml"] > 0
    delta_ok = aggregate["expectancy_delta"] >= 5.0
    cond = sign_flip and delta_ok
    criteria["expectancy_sign_flip"] = {
        "value": aggregate["expectancy_delta"],
        "threshold": 5.0,
        "sign_flip": sign_flip,
        "pass": cond,
    }
    if cond:
        passes += 1

    # 5. max_dd_delta <= -1.0 (W3 revision iter 1 — improvement)
    # max_drawdown_pct e' POSITIVO percentage points (metrics.py:156-170);
    # delta = ml - base; delta NEGATIVO = ml ha drawdown INFERIORE = improvement.
    # Threshold -1.0 pp coerente con ROADMAP SC#4 hint.
    cond = aggregate["max_dd_delta"] <= -1.0
    criteria["max_dd_improvement"] = {
        "value": aggregate["max_dd_delta"],
        "threshold": -1.0,
        "unit": "percentage_points (max_drawdown_pct >= 0)",
        "pass": cond,
    }
    if cond:
        passes += 1

    status = "NON_TRIVIAL_IMPROVEMENT" if passes >= 4 else "NEGATIVE_RESULT_DOCUMENTED"
    return {
        "status": status,
        "passes": passes,
        "total": 5,
        "criteria": criteria,
        "aggregate": aggregate,
    }


# ── Degraded Slices Analysis (D-08-D4 part 5) ───────────────────────────

def _degraded_slices_section(
    results: list[dict],
    baseline_results: list[dict],
    threshold_margin_pct: float,
) -> str:
    """Per ogni slice con ml_sharpe < base_sharpe:
    - identifica slice degraded
    - lista n_rejected (base_n - ml_n)
    - suggested threshold tuning (qualitative — Phase 9 implementa quantitative)
    - recommendation Phase 9 calibration retune

    threshold_margin_pct: da cfg.MCP_ML_THRESHOLD_MARGIN_PCT (default 0.05).
    """
    base_by_key = {(r["symbol"], r["timeframe"], r["profile"]): r for r in baseline_results}
    degraded = []
    for r in results:
        if r.get("status") != "OK":
            continue
        key = (r["symbol"], r["timeframe"], r["profile"])
        base = base_by_key.get(key)
        if base is None or base.get("status") != "OK":
            continue
        m_ml = r.get("metrics")
        m_base = base.get("metrics")
        sharpe_ml = _g(m_ml, "sharpe", 0.0)
        sharpe_base = _g(m_base, "sharpe", 0.0)
        if sharpe_ml < sharpe_base:
            degraded.append((key, r, base, sharpe_ml, sharpe_base))

    if not degraded:
        return ("## Degraded Slices Analysis\n\n"
                "Nessuna slice presenta ml_sharpe < base_sharpe. Phase 9 monitoring "
                "non rileva slice da retune.\n")

    lines = ["## Degraded Slices Analysis\n"]
    lines.append(
        f"**{len(degraded)} slice degraded** (ml_sharpe < base_sharpe). "
        f"Margine analisi: ±{threshold_margin_pct*100:.0f}% del threshold di profilo.\n"
    )
    lines.append("| symbol | tf | profile | base_n | ml_n | n_rejected | base_sharpe | ml_sharpe | sharpe_drop | suggested_action |\n"
                 "|---|---|---|---|---|---|---|---|---|---|")
    for key, r, base, sharpe_ml, sharpe_base in degraded:
        sym, tf, prof = key
        base_n = base.get("n_trades", 0)
        ml_n = r.get("n_trades", 0)
        n_rejected = base_n - ml_n
        sharpe_drop = sharpe_base - sharpe_ml
        action = (
            f"Phase 9 retune calibration {prof} threshold (margine ±{threshold_margin_pct*100:.0f}%); "
            f"Investigate {n_rejected} rejected marginal trades"
        )
        lines.append(
            f"| {sym} | {tf} | {prof} | {base_n} | {ml_n} | {n_rejected} "
            f"| {sharpe_base:.3f} | {sharpe_ml:.3f} | {sharpe_drop:.3f} | {action} |"
        )
    lines.append("")
    lines.append(
        "**Phase 9 recommendation**: rerunning calibration con dataset esteso "
        "(post paper deploy live trades) OR per-slice threshold dict in metadata.json "
        "(deferred Phase 9 calibration retune)."
    )
    return "\n".join(lines)


# ── Rejection rate per profile (D-08-D4 part 4) ─────────────────────────

def _rejection_rate_section(results: list[dict], baseline_results: list[dict]) -> str:
    """Aggrega rejection rate per profilo (CONSERVATIVE / MODERATE / AGGRESSIVE)."""
    base_n_by_prof = {}
    ml_n_by_prof = {}
    for r in baseline_results:
        if r.get("status") == "OK":
            p = r["profile"]
            base_n_by_prof[p] = base_n_by_prof.get(p, 0) + r.get("n_trades", 0)
    for r in results:
        if r.get("status") == "OK":
            p = r["profile"]
            ml_n_by_prof[p] = ml_n_by_prof.get(p, 0) + r.get("n_trades", 0)

    lines = ["## Rejection Rate Per Profile\n"]
    lines.append("| profile | base_n | ml_n | rejection_rate |\n|---|---|---|---|")
    for prof in ["CONSERVATIVE", "MODERATE", "AGGRESSIVE"]:
        base_n = base_n_by_prof.get(prof, 0)
        ml_n = ml_n_by_prof.get(prof, 0)
        rej = (1.0 - ml_n / base_n) if base_n > 0 else 0.0
        lines.append(f"| {prof} | {base_n} | {ml_n} | {rej:.2%} |")
    lines.append("")
    return "\n".join(lines)

def _validate_bundle_pre_run(bundle_path: Path) -> tuple[bool, str, dict]:
    """D-08-D3 pre-flight: bundle existence + sha256 + metadata.json parseable.

    W5 revision iter 1 — SEMANTICS chiarite: questo pre-flight calcola il
    bundle sha256 a SCOPO AUDIT-TRAIL (registrato in meta dict per il report
    header — l'operator puo' rispecchiare il bundle effettivamente usato).
    NON e' un gate di VERIFICATION: il sha non viene confrontato contro un
    expected_hash. Pickle deserialization signature gate (T-8-06-01) e'
    DEFERRED a Phase 9 (Pickle Spoofing Mitigation).

    Il log emette training_data_sha256 (post B5 revision iter 1 injection)
    per traceability dataset+bundle: operator audit log mostra entrambi gli sha.

    Ritorna (ok, msg, meta_dict).
    """
    if not bundle_path.exists():
        return False, f"bundle mancante: {bundle_path}", {}
    try:
        from backtest.baseline.determinism import file_sha256
        bundle_sha = file_sha256(bundle_path)
    except Exception as exc:
        return False, f"sha256 fallito: {exc}", {}

    meta_path = bundle_path.with_suffix(".metadata.json")
    if not meta_path.exists():
        return False, f"metadata.json mancante: {meta_path}", {}
    try:
        meta = json.loads(meta_path.read_text(encoding="utf-8"))
    except json.JSONDecodeError as exc:
        return False, f"metadata.json corrupt: {exc}", {}

    # W5 revision iter 1 — log audit-trail: bundle_sha + training_data_sha256
    # (post B5 injection). NON gate, solo traceability operator.
    training_sha = meta.get("training_data_sha256", "<not-injected>")
    _log.warning(
        "AUDIT-ONLY (non-verification gate): bundle_sha=%s training_data_sha=%s — "
        "pickle deserialization signature gate DEFERRED Phase 9 (T-8-06-01)",
        bundle_sha[:12], training_sha[:12] if training_sha != "<not-injected>" else "MISSING",
    )

    # Smoke MLFilter.load + 1-sample predict (sanity check)
    try:
        from ml.inference import MLFilter
        flt = MLFilter.load(bundle_path)
        # Smoke: predict 1 sample via fake DataFrame se features disponibili
        if hasattr(flt, "features") and flt.features:
            import pandas as pd
            X = pd.DataFrame([{f: 0.0 for f in flt.features}])
            raw, calibrated = flt.predict(X)
            _log.info("MLFilter smoke predict: raw=%.4f calibrated=%.4f", raw, calibrated)
    except Exception as exc:
        return False, f"MLFilter load/predict smoke fallito: {exc}", {}

    return (
        True,
        f"OK bundle (audit-only): sha256={bundle_sha[:12]}... training_sha={training_sha[:12] if training_sha != '<not-injected>' else 'MISSING'} + metadata.json + smoke predict",
        {**meta, "bundle_sha256": bundle_sha},
    )


def _load_baseline_results() -> list[dict]:
    """Carica baseline metrics da SQLite ledger (backtest_runs + backtest_trades)
    OR dal report MD parsing.

    Strategia semplice: ricarica i metrics via runner.run_baseline con flag
    cached/read-only mode. Per Plan 08-06: usa Baseline ledger SQL query diretta.

    Alternative implementation: per ridurre complessità, leggi
    .planning/research/baseline-2026-05-12.md parsing della Slice Metrics
    table. Plan 08-06 deferisce parsing complesso e usa SQL diretta.
    """
    from logger import _trades_db_path
    from config import Config
    import sqlite3

    cfg = Config()
    db_path = _trades_db_path(cfg)
    if not Path(db_path).exists():
        _log.warning("baseline ledger DB mancante: %s — return empty", db_path)
        return []
    rows = []
    with sqlite3.connect(db_path) as c:
        cur = c.execute("""
            SELECT run_id, symbol, timeframe, profile, status,
                   COALESCE(sharpe, 0.0), COALESCE(hit_rate, 0.0),
                   COALESCE(profit_factor, 0.0), COALESCE(expectancy_usd, 0.0),
                   COALESCE(max_drawdown_pct, 0.0), COALESCE(trade_count, 0)
            FROM backtest_runs
            WHERE run_id LIKE 'baseline_%'
        """)
        for row in cur.fetchall():
            (run_id, symbol, tf, profile, status, sharpe, hit, pf, exp_usd, dd, n_trades) = row
            rows.append({
                "run_id": run_id, "symbol": symbol, "timeframe": tf,
                "profile": profile, "status": "OK" if status == "done" else status,
                "n_trades": n_trades,
                "metrics": {
                    "sharpe": sharpe, "hit_rate": hit, "profit_factor": pf,
                    "expectancy_usd": exp_usd, "max_drawdown_pct": dd,
                },
            })
    _log.info("baseline_results caricati: %d run", len(rows))
    return rows
