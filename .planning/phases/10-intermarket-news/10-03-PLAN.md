    ```python
    class CalendarRSSClient:
        """FF RSS fetch+cache+filter client (D-10-B1..B6).

        Cache su disk JSON con TTL 60min (D-10-B2). Refresh on-demand (no scheduler).
        """

        def __init__(
            self,
            cfg=None,
            logger: logging.Logger | None = None,
            *,
            url: str | None = None,
            cache_path: Path | str | None = None,
            ttl_minutes: int | None = None,
            timeout_seconds: int = _DEFAULT_TIMEOUT_SECONDS,
        ):
            self.cfg = cfg
            self.log = logger or logging.getLogger(__name__)
            self.url = url or getattr(cfg, "FF_RSS_URL", None) or _DEFAULT_FF_URL
            cache_p = cache_path or getattr(cfg, "CALENDAR_CACHE_PATH", None) or _DEFAULT_CACHE_PATH
            self.cache_path = Path(cache_p)
            self.ttl_minutes = ttl_minutes or getattr(cfg, "CALENDAR_CACHE_TTL_MINUTES", None) or _DEFAULT_CACHE_TTL_MINUTES
            self.timeout_seconds = timeout_seconds

        # ---------- Public API ----------

        def fetch_events(
            self,
            *,
            force: bool = False,
            now: datetime | None = None,
        ) -> list[CalendarEvent]:
            """Fetch live FF feed o cache TTL <60min. Fallback cache stale on failure.

            Pattern riusato da news_aggregator.fetch_recent_news:34-54.
            """
            now = now or datetime.now(tz=timezone.utc)
            cached = self._load_cache()
            if not force and cached is not None:
                age = now - cached["fetched_at"]
                if age < timedelta(minutes=self.ttl_minutes):
                    return cached["events"]  # cache hit
            # Cache miss o force: fetch remote
            try:
                events = self._fetch_remote()
                self._save_cache(now, events)
                return events
            except Exception as exc:
                self.log.warning("FF RSS fetch fallito (%s) — fallback cache stale", exc)
                return cached["events"] if cached else []

        def filter_window(
            self,
            events: list[CalendarEvent],
            now: datetime,
            window_minutes: int,
            min_impact: str = "High",
            currencies: list[str] | None = None,
        ) -> list[CalendarEvent]:
            """Half-open lookback filter (D-10-B3, B4, B6).

            cutoff_start <= ev.event_time_utc < cutoff_end (half-open).
            Holiday events sempre inclusi se nel window (D-10-B6).
            """
            cutoff_start = now - timedelta(minutes=window_minutes)
            cutoff_end = now + timedelta(minutes=window_minutes)
            min_rank = _IMPACT_RANK.get(min_impact, 3)
            currencies_set = set(currencies) if currencies else None
            result = []
            for ev in events:
                # Half-open interval: include cutoff_start, exclude cutoff_end
                if not (cutoff_start <= ev.event_time_utc < cutoff_end):
                    continue
                # D-10-B6: Holiday sempre incluso (bypassa min_impact)
                if ev.impact != "Holiday":
                    rank = _IMPACT_RANK.get(ev.impact, 0)
                    if rank < min_rank:
                        continue
                if currencies_set and ev.country not in currencies_set:
                    continue
                result.append(ev)
            return sorted(result, key=lambda e: e.event_time_utc)

        # ---------- Internal (fetch + parse + cache) ----------

        def _fetch_remote(self) -> list[CalendarEvent]:
            """Fetch FF feed with socket timeout try/finally restore (Pitfall 7)."""
            old_to = socket.getdefaulttimeout()
            socket.setdefaulttimeout(self.timeout_seconds)
            try:
                parsed = feedparser.parse(self.url)
            finally:
                socket.setdefaulttimeout(old_to)
            entries = getattr(parsed, "entries", []) or []
            if getattr(parsed, "bozo", False) and not entries:
                raise RuntimeError(
                    f"FF feed bozo: {getattr(parsed, 'bozo_exception', None)}"
                )
            events: list[CalendarEvent] = []
            for entry in entries:
                ev = self._parse_event(entry)
                if ev is not None:
                    events.append(ev)
            return events

        def _parse_event(self, entry) -> CalendarEvent | None:
            """Parse FF custom schema (Pitfall 4) con ET->UTC zoneinfo (Pitfall 3).

            Returns None se:
            - campi obbligatori mancanti
            - impact == 'Tentative' (D-10-B6 skip)
            - time='All Day' e impact != 'Holiday' (D-10-B6)
            - time imaginary durante DST start (Pitfall 3)

            Ambiguous fall-back time (DST end, es. 11-01 01:30am) usa default zoneinfo
            fold=0 = EDT pre-switch (Warning #7). Comportamento documentato + testato.
            """
            title = (entry.get("title") or "").strip()
            country = (entry.get("country") or "").strip().upper()
            date_str = (entry.get("date") or "").strip()      # MM-DD-YYYY
            time_str = (entry.get("time") or "").strip()      # H:MMam/pm OR "All Day"
            impact = (entry.get("impact") or "").strip()

            if not title or not country or not date_str:
                return None

            # D-10-B6: skip Tentative
            if impact == "Tentative":
                return None

            all_day = time_str.lower() in ("all day", "")
            if all_day:
                # D-10-B6: All Day -> include solo se Holiday, nominal 00:00 UTC
                if impact != "Holiday":
                    return None
                try:
                    dt_et = datetime.strptime(date_str, "%m-%d-%Y").replace(tzinfo=_NY)
                except ValueError:
                    return None
                return CalendarEvent(
                    title=title, country=country,
                    event_time_utc=dt_et.astimezone(timezone.utc).replace(hour=0, minute=0, second=0, microsecond=0),
                    impact=impact, all_day=True,
                )

            # Time format normale: "H:MMam/pm" (D-10-B5)
            try:
                dt_naive = datetime.strptime(f"{date_str} {time_str}", "%m-%d-%Y %I:%M%p")
            except ValueError:
                return None

            # Pitfall 3 + Warning #7: zoneinfo replace + DST imaginary detection via round-trip.
            # Ambiguous fall-back (DST end): default fold=0 = EDT pre-switch — documentato in
            # test_dst_boundary_march_november con assertion 11-01 01:30am -> UTC 05:30.
            dt_et = dt_naive.replace(tzinfo=_NY)
            dt_utc = dt_et.astimezone(timezone.utc)
            # Round-trip check: se dt_et era imaginary (skip-hour DST start),
            # astimezone(utc).astimezone(_NY) ritorna ora DIVERSA da dt_et
            dt_back = dt_utc.astimezone(_NY).replace(tzinfo=_NY)
            if dt_back.hour != dt_et.hour or dt_back.minute != dt_et.minute:
                self.log.warning(
                    "Evento FF '%s' a %s imaginary (DST start skip-hour), skippato",
                    title, dt_et,
                )
                return None

            return CalendarEvent(
                title=title, country=country, event_time_utc=dt_utc, impact=impact,
                forecast=(entry.get("forecast") or None),
                previous=(entry.get("previous") or None),
                url=(entry.get("url") or None),
            )

        # ---------- Cache JSON (Pitfall 6) ----------

        def _load_cache(self) -> Optional[dict]:
            """Load cache JSON deserialize CalendarEvent. Tutti gli error -> None safely."""
            if not self.cache_path.exists():
                return None
            try:
                raw = json.loads(self.cache_path.read_text(encoding="utf-8"))
                fetched_at = datetime.fromisoformat(raw["fetched_at"])
                events = [
                    CalendarEvent(
                        title=e["title"], country=e["country"],
                        event_time_utc=datetime.fromisoformat(e["event_time_utc"]),
                        impact=e["impact"],
                        forecast=e.get("forecast"),
                        previous=e.get("previous"),
                        url=e.get("url"),
                        all_day=e.get("all_day", False),
                    )
                    for e in raw["events"]
                ]
                return {"fetched_at": fetched_at, "events": events}
            except (KeyError, ValueError, TypeError, json.JSONDecodeError) as exc:
                # Pitfall 6: TypeError aggiunto (datetime.fromisoformat su None/int crash)
                self.log.debug("Cache calendar_rss invalida (%s) — ignored", exc)
                return None

        def _save_cache(self, fetched_at: datetime, events: list[CalendarEvent]) -> None:
            """Save cache JSON con datetime->isoformat (Pitfall 6)."""
            self.cache_path.parent.mkdir(parents=True, exist_ok=True)
            raw = {
                "fetched_at": fetched_at.isoformat(),
                "events": [
                    {
                        "title": e.title, "country": e.country,
                        "event_time_utc": e.event_time_utc.isoformat(),
                        "impact": e.impact,
                        "forecast": e.forecast,
                        "previous": e.previous,
                        "url": e.url,
                        "all_day": e.all_day,
                    }
                    for e in events
                ],
            }
            self.cache_path.write_text(json.dumps(raw, indent=2), encoding="utf-8")

        # ---------- Helper public ----------

        @staticmethod
        def derive_currencies_from_pair(pair: str | None) -> list[str]:
            """D-10-B3: derive currencies from FX pair, fallback default 4-major."""
    ```
