    """Phase 10 D-10-A4: Refresh manuale on-demand Stooq macro CSV.

    Pitfall 1 (Stooq CAPTCHA Dec 2020+):
    - sleep(30) tra simboli per rate-limit.
    - HTML detection: response body inizia con '<!DOCTYPE' o '<html' -> abort.
    - Row count > 5000 obbligatorio post-download (23.6y daily ~6000 bar).
    - Manual fallback documentato: browser https://stooq.com/q/d/?s=<ticker>&i=d.

    Warning #6 (ASC ordering):
    - Stooq native order non garantito (può essere DESC newest-first OR ASC oldest-first).
    - Post pd.to_datetime: sort_values(\"Data\", ascending=True) + reset_index esplicito.
    - date_min = df[\"Data\"].iloc[0] (oldest, post-sort), date_max = df[\"Data\"].iloc[-1] (newest).
    - ASC ordering enforced locally — Stooq native order irrilevante.

    Pitfall 4 (FF RSS schema watch):
    - --probe-ff-rss dump live FF feed keys + compara vs tests/fixtures/ff_rss_smoke.xml.
    - Operator manual settimanale (NON CI: richiede live network).

    D-10-A4 sha256 explicit-intent:
    - Pre-write: leggi sha256 vecchio da metadata.json.
    - Post-write: calcola sha256 nuovo, confronta, log WARNING se cambiato.
    - metadata.json aggiornato con nuovo sha256 + downloaded_at ISO8601 UTC.

    CLI usage:
        python scripts/refresh_macro_csv.py                       # refresh all 4 simboli
        python scripts/refresh_macro_csv.py --symbols DXY US10Y   # subset
        python scripts/refresh_macro_csv.py --no-sleep            # skip sleep (testing only, NOT prod)
        python scripts/refresh_macro_csv.py --probe-ff-rss        # FF schema drift check
        python scripts/refresh_macro_csv.py --dry-run             # solo download + sample, no commit
    """
    from __future__ import annotations

    import argparse
    import hashlib
    import json
    import logging
    import sys
    import time
    from datetime import datetime, timezone
    from io import StringIO
    from pathlib import Path

    import pandas as pd
    import requests

    # Stooq ticker mapping (D-10-A1 locked + Wave 0 spike verified)
    _STOOQ_TICKERS = {
        "DXY":    "^dxy",
        "US10Y":  "^tnx",
        "XAUUSD": "xauusd",
        "WTI":    "cl.f",
    }

    _STOOQ_CSV_URL = "https://stooq.com/q/d/l/?s={ticker}&i=d"
    _SLEEP_BETWEEN_SYMBOLS_SEC = 30  # Pitfall 1 rate-limit
    _MIN_ROW_COUNT = 5000             # 23.6y daily ~6000 bar -> 5000 minimum
    _USER_AGENT = "trading-agent/2.0 (refresh_macro_csv)"
    _MACRO_ROOT = Path("data/historical/macro")
    _METADATA_PATH = _MACRO_ROOT / "metadata.json"
    _FF_RSS_URL = "https://nfs.faireconomy.media/ff_calendar_thisweek.xml"
    _FF_FIXTURE_PATH = Path("tests/fixtures/ff_rss_smoke.xml")
    _FF_EXPECTED_TAGS = {"title", "country", "date", "time", "impact", "forecast", "previous", "url"}


    # ============================================================================
    # Stooq refresh
    # ============================================================================

    def fetch_stooq_csv(ticker: str, *, log: logging.Logger) -> str:
        """Fetch Stooq daily CSV. Pitfall 1: HTML detection + warning + raise."""
        url = _STOOQ_CSV_URL.format(ticker=ticker)
        log.info("Fetch %s -> %s", ticker, url)
        resp = requests.get(url, timeout=15, headers={"User-Agent": _USER_AGENT})
        if resp.status_code != 200:
            raise RuntimeError(f"Stooq HTTP {resp.status_code} per {ticker}: {resp.text[:200]}")
        body = resp.text
        # Pitfall 1: HTML detection (Stooq serve HTML CAPTCHA page invece di CSV)
        if body.lstrip().lower().startswith(("<!doctype", "<html")):
            raise RuntimeError(
                f"Stooq ha ritornato HTML per {ticker} (probabile CAPTCHA gate). "
                f"Fallback manuale: scaricare da https://stooq.com/q/d/?s={ticker}&i=d via browser."
            )
        return body


    def normalize_csv_to_back01(raw_csv: str, *, log: logging.Logger) -> pd.DataFrame:
        """Normalizza Stooq US format -> BACK-01 semicolon Italian-format.

        Stooq default: Date,Open,High,Low,Close,Volume comma-separated YYYY-MM-DD dot decimal.
        BACK-01: Data;Ora;Open;High;Low;Close;Volume semicolon DD/MM/YYYY dot decimal.

        Warning #6: ASC ordering enforced locally (sort_values ascending=True + reset_index)
        post pd.to_datetime — Stooq native order irrilevante.
        """
        df = pd.read_csv(StringIO(raw_csv))
        expected_cols = {"Date", "Open", "High", "Low", "Close", "Volume"}
        if not expected_cols.issubset(set(df.columns)):
            raise RuntimeError(f"Stooq CSV header inatteso: {list(df.columns)} (atteso {expected_cols})")
        # Warning #6: parse Date come datetime PRIMA del sort
        df["Date"] = pd.to_datetime(df["Date"])
        # ASC ordering enforced locally — Stooq native order irrilevante
        df = df.sort_values("Date", ascending=True).reset_index(drop=True)
        # Trasformazione formato BACK-01
        df["Data"] = df["Date"].dt.strftime("%d/%m/%Y")
        df["Ora"] = "00:00:00"
        out = df[["Data", "Ora", "Open", "High", "Low", "Close", "Volume"]].copy()
        log.info("Normalizzato a BACK-01 format: %d row (ASC ordering enforced, oldest=%s newest=%s)",
                 len(out), df["Date"].iloc[0].strftime("%Y-%m-%d"), df["Date"].iloc[-1].strftime("%Y-%m-%d"))
        return out


    def refresh_symbol(symbol: str, *, dry_run: bool, log: logging.Logger) -> dict:
        """Refresh singolo symbol. Returns metadata dict per metadata.json update.

        Warning #6: date_min/date_max derivati da `df["Data"].iloc[0]` (oldest, post-ASC-sort)
        e `df["Data"].iloc[-1]` (newest). NESSUNA ipotesi sull'ordering nativo Stooq.
        """
        ticker = _STOOQ_TICKERS[symbol]
        raw = fetch_stooq_csv(ticker, log=log)
        df = normalize_csv_to_back01(raw, log=log)
        if len(df) < _MIN_ROW_COUNT:
            raise RuntimeError(
                f"Row count {len(df)} < {_MIN_ROW_COUNT} per {symbol}. "
                f"Probabile CAPTCHA partial response. Aborto (manual fallback consigliato)."
            )
        csv_path = _MACRO_ROOT / symbol / "daily.csv"
        # Old sha256 (pre-overwrite) per warning explicit-intent (D-10-A4)
        old_sha = hashlib.sha256(csv_path.read_bytes()).hexdigest() if csv_path.exists() else None
        if dry_run:
            log.info("[DRY-RUN] %s: %d row, sample close=%.4f (NO write)", symbol, len(df), float(df["Close"].iloc[0]))
            return {"symbol": symbol, "dry_run": True, "row_count": len(df), "old_sha256": old_sha}
        # Write atomic-ish
        csv_path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(csv_path, sep=";", index=False, encoding="utf-8")
        new_sha = hashlib.sha256(csv_path.read_bytes()).hexdigest()
        if old_sha and old_sha != new_sha:
            log.warning(
                "D-10-A4 sha256 CAMBIATO per %s: %s... -> %s... (intent operator richiesto)",
                symbol, old_sha[:16], new_sha[:16],
            )
        # Warning #6: ASC ordering enforced → iloc[0]=oldest, iloc[-1]=newest
        date_min = df["Data"].iloc[0]    # oldest (post-ASC-sort)
        date_max = df["Data"].iloc[-1]   # newest (post-ASC-sort)
        return {
            "path": str(csv_path),
            "sha256": new_sha,
            "row_count": len(df),
            "date_range": f"{date_min}..{date_max}",
            "old_sha256": old_sha,
        }


    def write_metadata(updates: dict[str, dict], *, log: logging.Logger) -> None:
        """Atomic update metadata.json preservando struttura D-10-A1."""
        if _METADATA_PATH.exists():
            meta = json.loads(_METADATA_PATH.read_text(encoding="utf-8"))
        else:
            meta = {
                "phase": 10,
                "decision": "D-10-A1, D-10-A4",
                "stooq_tickers": _STOOQ_TICKERS,
                "files": {},
                "notes_italiano": "Refresh manuale via scripts/refresh_macro_csv.py.",
            }
        meta["downloaded_at"] = datetime.now(tz=timezone.utc).isoformat()
        meta.setdefault("files", {})
        for sym, info in updates.items():
            if info.get("dry_run"):
                continue
            meta["files"][sym] = {
                "path": info["path"],
                "sha256": info["sha256"],
                "row_count": info["row_count"],
                "date_range": info["date_range"],
            }
        _METADATA_PATH.parent.mkdir(parents=True, exist_ok=True)
        _METADATA_PATH.write_text(json.dumps(meta, indent=2), encoding="utf-8")
        log.info("metadata.json aggiornato: %s", _METADATA_PATH)


    # ============================================================================
    # FF RSS schema probe (Pitfall 4 weekly regression)
    # ============================================================================

    def probe_ff_rss(*, log: logging.Logger) -> int:
        """Fetch live FF feed + dump keys + compare vs fixture. Returns exit code 0/1."""
        log.info("Probe FF RSS live: %s", _FF_RSS_URL)
        try:
            resp = requests.get(_FF_RSS_URL, timeout=10, headers={"User-Agent": _USER_AGENT})
        except Exception as exc:
            log.error("FF feed network error: %s", exc)
            return 1
        if resp.status_code != 200:
            log.error("FF feed HTTP %d", resp.status_code)
            return 1
        try:
            import feedparser
            parsed = feedparser.parse(resp.content)
        except Exception as exc:
            log.error("FF feed parse error: %s", exc)
            return 1
        if not parsed.entries:
            log.error("FF feed entries vuote (bozo=%s)", getattr(parsed, "bozo", None))
            return 1
        first = parsed.entries[0]
        keys_present = set(dict(first).keys())
        log.info("FF live entry[0] keys: %s", sorted(keys_present))
        missing = _FF_EXPECTED_TAGS - keys_present
        if missing:
            log.warning("FF schema DRIFT rilevato — tag mancanti: %s", missing)
            log.warning("Aggiorna tests/fixtures/ff_rss_smoke.xml + verifica parser calendar_rss.client._parse_event")
            return 1
        log.info("FF schema OK (8 tag canonical presenti)")
        return 0


    # ============================================================================
    # CLI main
    # ============================================================================

    def main(argv: list[str] | None = None) -> int:
        parser = argparse.ArgumentParser(
            description="Phase 10 D-10-A4 refresh Stooq macro CSV + FF RSS schema probe.",
        )
        parser.add_argument(
            "--symbols", nargs="*", default=None,
            choices=list(_STOOQ_TICKERS.keys()),
            help="Subset simboli (default tutti 4).",
        )
        parser.add_argument(
            "--no-sleep", action="store_true",
            help="Disabilita sleep(30) tra simboli. SOLO per testing, NON in prod (CAPTCHA risk).",
        )
        parser.add_argument(
            "--probe-ff-rss", action="store_true",
            help="Dump FF RSS live feed schema + compare vs fixture (Pitfall 4 weekly check).",
        )
        parser.add_argument(
            "--dry-run", action="store_true",
            help="Download + validate ma non scrive CSV/metadata.json.",
        )
        args = parser.parse_args(argv)

        logging.basicConfig(
            level=logging.INFO,
            format="%(asctime)s [%(levelname)s] %(message)s",
        )
        log = logging.getLogger("refresh_macro_csv")

        if args.probe_ff_rss:
            return probe_ff_rss(log=log)

        symbols = args.symbols or list(_STOOQ_TICKERS.keys())
        log.info("Refresh symbols: %s (dry_run=%s, no_sleep=%s)", symbols, args.dry_run, args.no_sleep)
        updates: dict[str, dict] = {}
        for i, sym in enumerate(symbols):
            if i > 0 and not args.no_sleep:
                log.info("Sleep %ds (Pitfall 1 rate-limit)...", _SLEEP_BETWEEN_SYMBOLS_SEC)
                time.sleep(_SLEEP_BETWEEN_SYMBOLS_SEC)
            try:
                updates[sym] = refresh_symbol(sym, dry_run=args.dry_run, log=log)
            except Exception as exc:
                log.error("Refresh %s FALLITO: %s", sym, exc)
                log.error("MANUAL FALLBACK: scaricare da https://stooq.com/q/d/?s=%s&i=d e normalizzare manualmente.", _STOOQ_TICKERS[sym])
                return 2

        if not args.dry_run:
            write_metadata(updates, log=log)
        log.info("Refresh completato: %d simboli", len(updates))
        return 0


    if __name__ == "__main__":
        sys.exit(main())
