    ```python
    class MacroLoader:
        """Bootstrap loader 4 macro CSV con strict-< close lookup.

        D-10-A2: lookup `close(D-1)` STRICT-< per evitare future leakage.
        D-10-A4: sha256 anchor calcolato a bootstrap, consumato da Phase 7 retrain
                 abort-on-drift via metadata.json.
        """

        def __init__(self, series_by_symbol: dict[str, MacroSeries]):
            self._series = series_by_symbol

        @classmethod
        def from_repo(cls, root: Path | str = DEFAULT_MACRO_ROOT) -> "MacroLoader":
            """Carica i 4 CSV da `root/{SYMBOL}/daily.csv`.

            Raises:
                FileNotFoundError: se anche solo uno dei 4 CSV manca.
            """
            root = Path(root)
            series: dict[str, MacroSeries] = {}
            for sym in MACRO_SYMBOLS:
                csv_path = root / sym / "daily.csv"
                if not csv_path.exists():
                    raise FileNotFoundError(
                        f"Macro CSV mancante per {sym}: {csv_path}. "
                        f"Verifica Plan 10-01 Task 1 (download manuale Stooq)."
                    )
                df = cls._read_csv(csv_path)
                sha = file_sha256(csv_path)
                series[sym] = MacroSeries(symbol=sym, df=df, sha256=sha)
            return cls(series)

        @staticmethod
        def _read_csv(path: Path) -> pd.DataFrame:
            """Parse BACK-01 semicolon Italian-format DD/MM/YYYY.

            Returns DataFrame con solo colonne ['ts_utc', 'close'] (Open/High/Low/Volume scartate).
            ts_utc = nominal date 00:00 UTC (daily-close convention).
            """
            df = pd.read_csv(path, sep=";", encoding="utf-8", dtype=str)
            df.columns = [c.strip() for c in df.columns]
            # Difensiva: header Stooq potrebbe variare; supporto sia "Data;Ora;Close" minimal
            # sia "Data;Ora;Open;High;Low;Close;Volume" full BACK-01.
            if "Data" not in df.columns or "Close" not in df.columns:
                raise ValueError(
                    f"CSV {path}: header inatteso {list(df.columns)}. "
                    f"Atteso: Data;Ora;Close (min) o Data;Ora;Open;High;Low;Close;Volume (full BACK-01)."
                )
            ora = df["Ora"] if "Ora" in df.columns else pd.Series(["00:00:00"] * len(df))
            ts_source = pd.to_datetime(df["Data"] + " " + ora, format="%d/%m/%Y %H:%M:%S")
            # Daily-close: nominal date 00:00 UTC (NO timezone offset come BACK-01 GMT-6)
            df["ts_utc"] = ts_source.dt.tz_localize(timezone.utc).dt.normalize()
            df["close"] = df["Close"].astype(float)
            df = (df[["ts_utc", "close"]]
                  .sort_values("ts_utc")
                  .drop_duplicates("ts_utc")
                  .reset_index(drop=True))
            return df

        def close_at(self, symbol: str, ts_utc: datetime) -> float | None:
            """Strict-< lookup close(D-1).

            D-10-A2: NO future leakage. Ritorna close del primo bar
            PRIMA di ts_utc (NON quello @ ts_utc esatto).

            Returns None se symbol non mappato o nessun bar antecedente disponibile.
            """
            series = self._series.get(symbol)
            if series is None:
                return None
            df = series.df
            ts_pd = pd.Timestamp(ts_utc).tz_convert("UTC") if pd.Timestamp(ts_utc).tzinfo else pd.Timestamp(ts_utc, tz="UTC")
            mask = df["ts_utc"] < ts_pd  # STRICT-< (NON <=) — fondamentale
            if not mask.any():
                return None
            return float(df.loc[mask, "close"].iloc[-1])

        def sha256(self, symbol: str) -> str | None:
            """Hex digest sha256 (64-char) del CSV sorgente per symbol."""
            s = self._series.get(symbol)
            return s.sha256 if s else None

        def symbols(self) -> tuple[str, ...]:
            """Tupla simboli caricati (subset di MACRO_SYMBOLS in caso di partial load future)."""
            return tuple(self._series.keys())
    ```

    ```python
    """Build intermarket score function (D-10-D1, Claude's Discretion).

    Pure-fn factory: chiude su MacroLoader, ritorna `Callable[[str, Literal['long','short']], float]`.

    Score > 0 ⇒ macro CONFERMA direction.
    Score < 0 ⇒ macro CONTRARIA direction.
    Output clampato a [-1.0, +1.0].

    _PAIR_WEIGHTS: hardcoded per Phase 10 (Open Question 4 — YAML promotion deferred a Phase 11+
    se paper trading rivela need-to-tune). Weights basati su Murphy intermarket analysis:
      - EURUSD/GBPUSD: DXY weak (-0.5) confermano BUY (USD inverso); risk-on (US10Y up +0.2) lieve
        confirm; gold up (-0.2) risk-off contraria; WTI up (+0.1) risk-on lieve confirm.
      - USDJPY: DXY strong (+0.5) confirms BUY; US10Y up (+0.3) carry-trade confirm; gold up (-0.2)
        risk-off (JPY safe-haven) contraria.

    Pitfall 5: callsite confluence.py:266 ha try/except no-op safety — qualunque
    TypeError/KeyError viene inghiottito. NESSUN logging interno qui (pure-fn).
    """
    from __future__ import annotations

    from datetime import datetime, timedelta, timezone
    from typing import Callable, Literal

    # Pair -> macro symbol weight (Murphy intermarket reference, hardcoded D-10-D1)
    _PAIR_WEIGHTS: dict[str, dict[str, float]] = {
        "EURUSD": {"DXY": -0.5, "US10Y": +0.2, "XAUUSD": -0.2, "WTI": +0.1},
        "GBPUSD": {"DXY": -0.5, "US10Y": +0.2, "XAUUSD": -0.2, "WTI": +0.1},
        "USDJPY": {"DXY": +0.5, "US10Y": +0.3, "XAUUSD": -0.2, "WTI": 0.0},
    }

    # Lookback window: prev close = today's strict-< (D-1), then prev-prev close (D-2) per delta
    _DELTA_LOOKBACK_DAYS = 1


    def build_intermarket_score(
        loader,
        *,
        weights: dict[str, dict[str, float]] | None = None,
        now_fn: Callable[[], datetime] | None = None,
    ) -> Callable[[str, Literal["long", "short"]], float]:
        """Factory: closure su loader. Ritorna fn(symbol, direction) -> float [-1, +1].

        Args:
            loader: MacroLoader istanza (deve implementare close_at(symbol, ts_utc)).
            weights: override dei _PAIR_WEIGHTS (default None = hardcoded). Phase 11+ retrofit
                     può iniettare YAML config.
            now_fn: callable that ritorna current UTC datetime; default datetime.now(tz=utc).
                    Permette test deterministico con clock injection.

        Returns:
            _score(symbol, direction) -> float clampato [-1, +1]. Direction-aware sign flip.
        """
        active_weights = weights if weights is not None else _PAIR_WEIGHTS
        clock = now_fn if now_fn is not None else (lambda: datetime.now(tz=timezone.utc))

        def _score(symbol: str, direction: Literal["long", "short"]) -> float:
            """Closure: scoreggia il macro signal aggregato direction-aware."""
            pair_weights = active_weights.get(symbol)
            if pair_weights is None:
                return 0.0  # symbol non mappato: neutral safe fallback (test_unmapped_symbol_returns_zero)
            now_utc = clock()
            # D-1 = strict-< now_utc; D-2 = strict-< (D-1 - 1 day) — i.e., 2-bar delta
            d1_ts = now_utc
            d2_ts = now_utc - timedelta(days=_DELTA_LOOKBACK_DAYS + 1)
            total_weight = 0.0
            weighted_signal = 0.0
            for macro_sym, w in pair_weights.items():
                c_d1 = loader.close_at(macro_sym, d1_ts)
                c_d2 = loader.close_at(macro_sym, d2_ts)
                if c_d1 is None or c_d2 is None or c_d2 == 0:
                    continue  # contribuente skippato (loader senza dato)
                delta_pct = (c_d1 - c_d2) / c_d2  # daily ret approx
                # Scaled x100 per portare in range comparabile [-0.1..+0.1] tipico daily ret macro
                weighted_signal += w * delta_pct * 100.0
                total_weight += abs(w)
            if total_weight == 0:
                return 0.0  # tutti i contributenti None: neutral (test_no_data_returns_zero)
            raw = weighted_signal / total_weight
            # D-10-D1: direction-aware sign flip
            if direction == "short":
                raw = -raw
            # Clamp [-1.0, +1.0]
            return max(-1.0, min(1.0, raw))

        return _score
    ```
