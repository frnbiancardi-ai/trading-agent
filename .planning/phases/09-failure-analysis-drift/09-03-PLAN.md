```python
"""drift.compute — Phase 9 Area B D-09-B1 3-tier drift compute.

3 metriche:
1. Prediction KS (scipy.stats.ks_2samp method='auto')
2. Calibration ECE — W6: manual np.bincount (NOT sklearn.calibration_curve, che droppa empty bins)
3. Feature drift top-K (top 15 da bundle.feature_importance_, P12 filter > 0 + drop categorical)
"""
from __future__ import annotations

import numpy as np
import pandas as pd
from scipy.stats import ks_2samp


def compute_prediction_ks(pred_reference, pred_current) -> dict:
    pred_ref = np.asarray(pred_reference, dtype=np.float64)
    pred_cur = np.asarray(pred_current, dtype=np.float64)
    stat, pvalue = ks_2samp(pred_ref, pred_cur, method="auto")
    return {
        "ks_stat": float(stat),
        "ks_pvalue": float(pvalue),
        "n_reference": int(len(pred_ref)),
        "n_current": int(len(pred_cur)),
    }


def compute_ece(y_true, y_prob, n_bins: int = 10) -> float:
    """W6 ECE manual implementation via np.bincount (NOT sklearn.calibration_curve).

    Rationale: sklearn.calibration_curve droppa empty bins silently → ECE
    sotto-stimato + shape mismatch su sparse-bin scenario. Manual np.bincount
    preserva bin-edge invariant + applica weights = 0 a empty bins.

    P10: n_bins MUST match Phase 7 training (default 10 da cfg.DRIFT_ECE_N_BINS).
    """
    y_true = np.asarray(y_true, dtype=np.int64)
    y_prob = np.asarray(y_prob, dtype=np.float64)
    if len(y_true) < n_bins:
        return float("nan")

    bin_edges = np.linspace(0.0, 1.0, n_bins + 1)
    bin_idx = np.clip(np.digitize(y_prob, bin_edges) - 1, 0, n_bins - 1)
    counts = np.bincount(bin_idx, minlength=n_bins)
    sum_pred = np.bincount(bin_idx, weights=y_prob, minlength=n_bins)
    sum_outcome = np.bincount(bin_idx, weights=y_true.astype(np.float64), minlength=n_bins)
    nonempty = counts > 0
    if not nonempty.any():
        return float("nan")
    mean_pred = np.where(nonempty, sum_pred / np.maximum(counts, 1), 0.0)
    mean_outcome = np.where(nonempty, sum_outcome / np.maximum(counts, 1), 0.0)
    weights = counts / counts.sum()
    return float(np.sum(np.abs(mean_outcome - mean_pred) * weights))


def compute_feature_drift_topk(df_reference, df_current, top_k_features: list[str],
                                ks_threshold: float = 0.01) -> list[dict]:
    out: list[dict] = []
    for feat in top_k_features:
        if feat not in df_reference.columns or feat not in df_current.columns:
            continue
        ref = df_reference[feat].dropna().values
        cur = df_current[feat].dropna().values
        if len(ref) < 10 or len(cur) < 10:
            continue
        stat, pvalue = ks_2samp(ref, cur, method="auto")
        out.append({
            "feature": feat,
            "ks_stat": float(stat),
            "ks_pvalue": float(pvalue),
            "breached": bool(pvalue < ks_threshold),
        })
    return out
```

def handle_get_drift_metrics(args: dict, cfg: Any, drift_log_db_path: str) -> dict:
    """MCP-08 — B1 pre-flight + compute + persist + return polymorphic shape."""
    # B1 cross-phase pre-flight (declared by Phase 8 Plan 08-01)
    for _attr in _REQUIRED_PHASE8_CFG_ATTRS:
        if not hasattr(cfg, _attr):
            return envelope(
                ErrorCodes.INTERNAL_ERROR,
                f"Phase 8 prerequisite missing: cfg.{_attr} not declared. "
                f"Run Phase 8 Plan 08-01 scaffolding first.",
            )

    try:
        window_days = int(args.get("window_days", cfg.DRIFT_DEFAULT_WINDOW_DAYS))
    except (TypeError, ValueError):
        return envelope(ErrorCodes.VALIDATION_FAILED, "window_days deve essere int")
    if window_days < 1 or window_days > 365:
        return envelope(ErrorCodes.VALIDATION_FAILED, f"window_days fuori range [1,365]: {window_days}")

    from drift.reference import load_reference
    from drift.compute import compute_prediction_ks, compute_ece, compute_feature_drift_topk
    from drift.alarm import check_alarm
    from drift.log_db import ensure_drift_log_db, append_drift_row

    ensure_drift_log_db(drift_log_db_path)

    try:
        ref = load_reference(Path(cfg.ML_MODEL_PATH))
    except FileNotFoundError as exc:
        return envelope(ErrorCodes.NOT_FOUND, f"reference mancante: {exc}")
    except RuntimeError as exc:
        return envelope(ErrorCodes.INTERNAL_ERROR, f"reference integrity violation: {exc}")

    try:
        pred_cur, y_cur, df_cur = _load_current_window(window_days, cfg)
        pred_ref, y_ref, df_ref = _load_reference_window(ref)
    except Exception as exc:
        log.exception("get_drift_metrics: errore caricamento window")
        return envelope(ErrorCodes.INTERNAL_ERROR, str(exc))

    n_current = int(len(pred_cur))
    is_cold_start = n_current < cfg.DRIFT_MIN_SAMPLES_CURRENT

    pred_drift = compute_prediction_ks(pred_ref, pred_cur) if n_current > 0 else {
        "ks_stat": float("nan"), "ks_pvalue": float("nan"),
        "n_reference": len(pred_ref), "n_current": 0,
    }
    pred_drift["threshold"] = cfg.DRIFT_KS_PVALUE_THRESHOLD
    pred_drift["breached"] = bool(
        pred_drift.get("ks_pvalue") is not None
        and not _is_nan(pred_drift["ks_pvalue"])
        and pred_drift["ks_pvalue"] < cfg.DRIFT_KS_PVALUE_THRESHOLD
    )

    current_ece = compute_ece(y_cur, pred_cur, n_bins=cfg.DRIFT_ECE_N_BINS) if not is_cold_start else float("nan")
    reference_ece = float(ref["metadata"].get("aggregate", {}).get("ece", 0.0))
    delta_ece = (current_ece - reference_ece) if not _is_nan(current_ece) else float("nan")
    calibration_drift = {
        "current_ece": float(current_ece) if not _is_nan(current_ece) else None,
        "reference_ece": reference_ece,
        "delta_ece": float(delta_ece) if not _is_nan(delta_ece) else None,
        "threshold": cfg.DRIFT_ECE_DELTA_THRESHOLD,
        "breached": bool(not _is_nan(delta_ece) and delta_ece > cfg.DRIFT_ECE_DELTA_THRESHOLD),
        "n_with_outcome": int(len(y_cur)),
        "n_pending_outcome": int(max(0, n_current - len(y_cur))),
        "cold_start": is_cold_start,
    }

    feature_importance = ref["metadata"].get("feature_importance_", {})
    CATEGORICAL = {"symbol", "timeframe", "profile", "setup_name", "regime"}
    top_k_features = [
        f for f, w in sorted(feature_importance.items(), key=lambda kv: -kv[1])
        if w > 0 and f not in CATEGORICAL
    ][:cfg.DRIFT_TOP_K_FEATURES]
    feature_drift = compute_feature_drift_topk(df_ref, df_cur, top_k_features,
                                                ks_threshold=cfg.DRIFT_KS_PVALUE_THRESHOLD)

    metrics = {
        "window_days": window_days,
        "reference_period": ref["metadata"].get("training_period", "n/a"),
        "current_period": f"last {window_days} days",
        "reference_hash": ref["sha256"],
        "model_version": ref["metadata"].get("model_version"),
        "n_predictions_current": n_current,
        "n_predictions_reference": int(len(pred_ref)),
        "prediction_drift": pred_drift,
        "calibration_drift": calibration_drift,
        "feature_drift": feature_drift,
    }
    metrics.update(check_alarm(metrics, cfg))

    # Persist audit trail D-09-B4
    ts_utc = datetime.now(timezone.utc).isoformat()
    for m_name, m_value, m_thresh, m_breach in _flatten_to_rows(metrics):
        try:
            append_drift_row(drift_log_db_path, {
                "ts_utc": ts_utc,
                "model_version": metrics["model_version"],
                "reference_hash": metrics["reference_hash"],
                "metric_name": m_name,
                "metric_value": float(m_value) if m_value is not None and not _is_nan(m_value) else 0.0,
                "threshold": float(m_thresh) if m_thresh is not None else 0.0,
                "breached": bool(m_breach),
                "n_samples_current": metrics.get("n_predictions_current"),
                "n_samples_reference": metrics.get("n_predictions_reference"),
                "window_days": window_days,
                "computed_by": "mcp_get_drift_metrics",
            })
        except Exception:
            log.exception("get_drift_metrics: errore append_drift_row %s", m_name)
    return metrics


def _flatten_to_rows(metrics: dict) -> list[tuple]:
    rows = []
    pd_drift = metrics.get("prediction_drift") or {}
    rows.append(("prediction_ks", pd_drift.get("ks_pvalue"), pd_drift.get("threshold"), pd_drift.get("breached")))
    cd = metrics.get("calibration_drift") or {}
    rows.append(("calibration_ece", cd.get("delta_ece"), cd.get("threshold"), cd.get("breached")))
    for f in metrics.get("feature_drift") or []:
        rows.append((f"feature_drift_{f['feature']}", f["ks_stat"], pd_drift.get("threshold"), f["breached"]))
    return rows

def _load_current_window(window_days: int, cfg: Any) -> tuple:
    """B6 Option A: legge current-window prediction + outcomes per drift compute.

    Sources:
    - prediction: trades.db trades_log.ml_calibrated_prob (D-09-D4 column post Wave 0)
    - outcomes y_true: trades_log.pnl_realized > 0 → 1 (TP_HIT-like) else 0 (Phase 7 D-01 align)
    - df_features: decision_context_json indicators (parsed JSON)

    Cold-start safety: rows con ml_calibrated_prob IS NULL escluse (legacy pre-Phase-9).
    Returns (pred_cur: np.ndarray, y_cur: np.ndarray, df_cur: pd.DataFrame).
    """
    from logger import _trades_db_path
    cutoff = datetime.now(timezone.utc) - timedelta(days=window_days)
    cutoff_iso = cutoff.isoformat()
    db_path = _trades_db_path(cfg)

    pred_list: list[float] = []
    y_list: list[int] = []
    feat_rows: list[dict] = []
    with sqlite3.connect(str(db_path)) as c:
        c.execute("PRAGMA journal_mode=WAL")
        rows = c.execute(
            "SELECT id, timestamp, ml_calibrated_prob, decision_context_json, "
            "pnl_realized FROM trades_log "
            "WHERE timestamp >= ? AND ml_calibrated_prob IS NOT NULL",
            (cutoff_iso,),
        ).fetchall()

    for tid, _ts, prob, ctx_json, pnl in rows:
        if prob is None:
            continue
        pred_list.append(float(prob))
        outcome = 1 if (pnl is not None and float(pnl) > 0) else 0
        y_list.append(outcome)
        try:
            ctx = json.loads(ctx_json or "{}")
        except (TypeError, json.JSONDecodeError):
            ctx = {}
        ind = ctx.get("indicators", {}) if isinstance(ctx, dict) else {}
        feat_rows.append({"trade_id": tid, **ind})

    pred_cur = np.asarray(pred_list, dtype=np.float64)
    y_cur = np.asarray(y_list, dtype=np.int64)
    df_cur = pd.DataFrame(feat_rows)
    return pred_cur, y_cur, df_cur


def _load_reference_window(ref: dict) -> tuple:
    """B6 Option A: legge reference da baseline_decisions parquet (Phase 5 cert.).

    Returns (pred_ref: np.ndarray, y_ref: np.ndarray, df_ref: pd.DataFrame).
    Cold-start safety: se parquet legacy senza calibrated_prob column → array vuoto
    (handler gestisce n_predictions_reference=0; alarm-side gestisce cold-start).
    """
    parquet_path = Path(ref["parquet_path"])
    df = pd.read_parquet(parquet_path)

    if "calibrated_prob" in df.columns:
        pred_ref = df["calibrated_prob"].dropna().values.astype("float64")
    elif "ml_calibrated_prob" in df.columns:
        pred_ref = df["ml_calibrated_prob"].dropna().values.astype("float64")
    else:
        pred_ref = np.array([], dtype=np.float64)

    if "exit_reason" in df.columns:
        y_ref = (df["exit_reason"] == "TP_HIT").astype("int64").values
    else:
        y_ref = np.array([], dtype=np.int64)

    return pred_ref, y_ref, df
