```python
"""drift.reference — Phase 9 Area B D-09-B2 static training distribution loader.

Audit-anchored via sha256(part-0.parquet) strict-fail (P14 protezione training data integrity).
"""
from __future__ import annotations

import hashlib
import json
from pathlib import Path


def _file_sha256(path: Path) -> str:
    """sha256 hex digest del file (mirror backtest.baseline.determinism.file_sha256)."""
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def load_reference(model_bundle_path: str | Path) -> dict:
    """Carica metadata.json sidecar Phase 7 + verifica sha256 baseline parquet.

    D-09-B2: reference = static training distribution. Audit-anchored via
    metadata.dataset_hash == sha256(data/training/baseline_decisions/part-0.parquet).
    P14: strict-fail se hash mismatch → never silently use stale reference.

    Args:
        model_bundle_path: percorso bundle .pkl Phase 7 (sidecar .metadata.json sibling).

    Returns:
        {"metadata": dict, "parquet_path": str, "sha256": str}.

    Raises:
        FileNotFoundError: metadata.json o parquet mancante.
        RuntimeError: hash mismatch (training data integrity violation).
    """
    bundle_path = Path(model_bundle_path)
    meta_path = bundle_path.with_suffix(".metadata.json")
    if not meta_path.exists():
        raise FileNotFoundError(f"metadata.json mancante: {meta_path}")
    meta = json.loads(meta_path.read_text(encoding="utf-8"))

    parquet_path = Path("data/training/baseline_decisions/part-0.parquet")
    if not parquet_path.exists():
        raise FileNotFoundError(f"baseline parquet mancante: {parquet_path}")

    actual_sha = _file_sha256(parquet_path)
    expected_sha = meta.get("dataset_hash")
    if expected_sha and actual_sha != expected_sha:
        raise RuntimeError(
            f"reference baseline parquet drift (P14 strict-fail): "
            f"expected {expected_sha[:12]}... actual {actual_sha[:12]}..."
        )
    return {"metadata": meta, "parquet_path": str(parquet_path), "sha256": actual_sha}
```

```python
"""drift.compute — Phase 9 Area B D-09-B1 3-tier drift compute.

3 metriche:
1. Prediction KS (scipy.stats.ks_2samp method='auto')
2. Calibration ECE (sklearn.calibration_curve, n_bins=cfg.DRIFT_ECE_N_BINS, P10 invariant)
3. Feature drift top-K (top 15 da bundle.feature_importance_, P12 filter > 0 + drop categorical)
"""
from __future__ import annotations

import numpy as np
import pandas as pd
from scipy.stats import ks_2samp
from sklearn.calibration import calibration_curve


def compute_prediction_ks(
    pred_reference: np.ndarray,
    pred_current: np.ndarray,
) -> dict:
    """KS-test su distribuzione prediction (calibrated_prob).

    Returns:
        {ks_stat, ks_pvalue, n_reference, n_current}.
    """
    pred_ref = np.asarray(pred_reference, dtype=np.float64)
    pred_cur = np.asarray(pred_current, dtype=np.float64)
    stat, pvalue = ks_2samp(pred_ref, pred_cur, method="auto")
    return {
        "ks_stat": float(stat),
        "ks_pvalue": float(pvalue),
        "n_reference": int(len(pred_ref)),
        "n_current": int(len(pred_cur)),
    }


def compute_ece(
    y_true: np.ndarray,
    y_prob: np.ndarray,
    n_bins: int = 10,
) -> float:
    """ECE via sklearn.calibration_curve + numpy weighted sum.

    P10 invariant: n_bins MUST match Phase 7 training (default 10 da cfg.DRIFT_ECE_N_BINS).

    Returns:
        ECE float; NaN se len(y_true) < n_bins (cold-start protection).
    """
    y_true = np.asarray(y_true, dtype=np.int64)
    y_prob = np.asarray(y_prob, dtype=np.float64)
    if len(y_true) < n_bins:
        return float("nan")
    prob_true, prob_pred = calibration_curve(y_true, y_prob, n_bins=n_bins, strategy="uniform")
    # Bin assignment per sample
    bin_edges = np.linspace(0, 1, n_bins + 1)
    bin_idx = np.clip(np.searchsorted(bin_edges, y_prob, side="right") - 1, 0, n_bins - 1)
    bin_counts = np.bincount(bin_idx, minlength=n_bins)
    nonzero = bin_counts > 0
    weights = bin_counts[nonzero] / len(y_true)
    return float(np.sum(np.abs(prob_true - prob_pred) * weights))


def compute_feature_drift_topk(
    df_reference: pd.DataFrame,
    df_current: pd.DataFrame,
    top_k_features: list[str],
    ks_threshold: float = 0.01,
) -> list[dict]:
    """Per-feature KS top-K (D-09-B1).

    P12: caller upstream filtra feature_importance_ > 0 + esclude categorical
    (Phase 7 D-10 symbol/timeframe/profile/setup_name/regime → chi-square Phase 11+).
    """
    out: list[dict] = []
    for feat in top_k_features:
        if feat not in df_reference.columns or feat not in df_current.columns:
            continue  # P12 lower bound: missing feature skip
        ref = df_reference[feat].dropna().values
        cur = df_current[feat].dropna().values
        if len(ref) < 10 or len(cur) < 10:
            continue
        stat, pvalue = ks_2samp(ref, cur, method="auto")
        out.append({
            "feature": feat,
            "ks_stat": float(stat),
            "ks_pvalue": float(pvalue),
            "breached": bool(pvalue < ks_threshold),
        })
    return out
```

```python
"""drift.alarm — Phase 9 Area B D-09-B3 3-tier alarm aggregation.

Global = ANY 1 di 3 breach: prediction_KS pvalue, calibration ECE delta, feature_drift fraction.
"""
from __future__ import annotations

from typing import Any


def check_alarm(metrics: dict, cfg: Any) -> dict:
    """3-tier alarm check (D-09-B3): pred_KS + ECE_delta + feature_fraction breach.

    Returns:
        {breached: bool, breaches: list[str], alarm_status: "breach"|"ok"}.
    """
    breaches: list[str] = []

    # Tier 1 — Prediction KS (pvalue < threshold → breached)
    pred_drift = metrics.get("prediction_drift") or {}
    pred_pvalue = pred_drift.get("ks_pvalue")
    if pred_pvalue is not None and pred_pvalue < cfg.DRIFT_KS_PVALUE_THRESHOLD:
        breaches.append("prediction_ks")

    # Tier 2 — Calibration ECE delta absolute > threshold → breached
    calib = metrics.get("calibration_drift") or {}
    delta = calib.get("delta_ece")
    # NaN guard: cold-start ECE NaN → no breach
    if delta is not None and not _is_nan(delta) and delta > cfg.DRIFT_ECE_DELTA_THRESHOLD:
        breaches.append("calibration_ece")

    # Tier 3 — Feature drift fraction ≥ threshold → breached
    feature_drift = metrics.get("feature_drift") or []
    if feature_drift:
        feature_breached = [f for f in feature_drift if f.get("breached")]
        frac = len(feature_breached) / max(1, len(feature_drift))
        if frac >= cfg.DRIFT_FEATURE_FRACTION_THRESHOLD:
            breaches.append("feature_drift")

    return {
        "breached": len(breaches) > 0,
        "breaches": breaches,
        "alarm_status": "breach" if breaches else "ok",
    }
```

```python
"""mcp_tools.handlers.drift — MCP-08 get_drift_metrics handler.

Compose pipeline: drift.reference + drift.compute + drift.alarm + drift.log_db.
Response shape verbatim CONTEXT.md § specifics MCP-08 lines 316-345.
"""
from __future__ import annotations

import logging
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from mcp.types import Tool

from mcp_tools.errors import ErrorCodes, envelope


log = logging.getLogger(__name__)


GET_DRIFT_METRICS_TOOL = Tool(
    name="get_drift_metrics",
    description=(
        "3-tier drift monitor: prediction-distribution KS (scipy), calibration ECE "
        "(sklearn calibration_curve), feature drift top-K (LightGBM importance). "
        "Reference = static training distribution (D-09-B2). Alarm fires when ANY "
        "1 of 3 breached (D-09-B3 .env thresholds). Audit-anchored via "
        "reference_hash == bundle dataset_hash."
    ),
    inputSchema={
        "type": "object",
        "properties": {
            "window_days": {"type": "integer", "minimum": 1, "maximum": 365, "default": 30},
        },
        "required": [],
    },
)


def handle_get_drift_metrics(args: dict, cfg: Any, drift_log_db_path: str) -> dict:
    """MCP-08 — compute current-window drift + persist in drift_log.db + ritorna polymorphic."""
    try:
        window_days = int(args.get("window_days", cfg.DRIFT_DEFAULT_WINDOW_DAYS))
    except (TypeError, ValueError):
        return envelope(ErrorCodes.VALIDATION_FAILED, "window_days deve essere int")
    if window_days < 1 or window_days > 365:
        return envelope(ErrorCodes.VALIDATION_FAILED, f"window_days fuori range [1,365]: {window_days}")

    from drift.reference import load_reference
    from drift.compute import compute_prediction_ks, compute_ece, compute_feature_drift_topk
    from drift.alarm import check_alarm
    from drift.log_db import ensure_drift_log_db, append_drift_row

    ensure_drift_log_db(drift_log_db_path)

    try:
        ref = load_reference(Path(cfg.ML_MODEL_PATH))
    except FileNotFoundError as exc:
        return envelope(ErrorCodes.NOT_FOUND, f"reference mancante: {exc}")
    except RuntimeError as exc:
        # P14: sha256 mismatch
        return envelope(ErrorCodes.INTERNAL_ERROR, f"reference integrity violation: {exc}")

    try:
        pred_cur, y_cur, df_cur = _load_current_window(window_days, cfg)
        pred_ref, y_ref, df_ref = _load_reference_window(ref)
    except Exception as exc:
        log.exception("get_drift_metrics: errore caricamento window")
        return envelope(ErrorCodes.INTERNAL_ERROR, str(exc))

    # Cold-start: n_current < MIN_DRIFT_SAMPLES → NaN ECE + soft warning
    n_current = int(len(pred_cur))
    is_cold_start = n_current < cfg.DRIFT_MIN_SAMPLES_CURRENT

    pred_drift = compute_prediction_ks(pred_ref, pred_cur) if n_current > 0 else {
        "ks_stat": float("nan"), "ks_pvalue": float("nan"),
        "n_reference": len(pred_ref), "n_current": 0,
    }
    pred_drift["threshold"] = cfg.DRIFT_KS_PVALUE_THRESHOLD
    pred_drift["breached"] = bool(pred_drift.get("ks_pvalue") is not None
                                   and not _is_nan(pred_drift["ks_pvalue"])
                                   and pred_drift["ks_pvalue"] < cfg.DRIFT_KS_PVALUE_THRESHOLD)

    current_ece = compute_ece(y_cur, pred_cur, n_bins=cfg.DRIFT_ECE_N_BINS) if not is_cold_start else float("nan")
    reference_ece = float(ref["metadata"].get("aggregate", {}).get("ece", 0.0))
    delta_ece = (current_ece - reference_ece) if not _is_nan(current_ece) else float("nan")
    calibration_drift = {
        "current_ece": float(current_ece) if not _is_nan(current_ece) else None,
        "reference_ece": reference_ece,
        "delta_ece": float(delta_ece) if not _is_nan(delta_ece) else None,
        "threshold": cfg.DRIFT_ECE_DELTA_THRESHOLD,
        "breached": bool(not _is_nan(delta_ece) and delta_ece > cfg.DRIFT_ECE_DELTA_THRESHOLD),
        "n_with_outcome": int(len(y_cur)),
        "n_pending_outcome": int(max(0, n_current - len(y_cur))),
        "cold_start": is_cold_start,
    }

    # Top-K features (P12 guard: filter > 0 importance + drop categorical)
    feature_importance = ref["metadata"].get("feature_importance_", {})
    CATEGORICAL = {"symbol", "timeframe", "profile", "setup_name", "regime"}
    top_k_features = [
        f for f, w in sorted(feature_importance.items(), key=lambda kv: -kv[1])
        if w > 0 and f not in CATEGORICAL
    ][:cfg.DRIFT_TOP_K_FEATURES]
    feature_drift = compute_feature_drift_topk(df_ref, df_cur, top_k_features,
                                                ks_threshold=cfg.DRIFT_KS_PVALUE_THRESHOLD)

    metrics = {
        "window_days": window_days,
        "reference_period": ref["metadata"].get("training_period", "n/a"),
        "current_period": f"last {window_days} days",
        "reference_hash": ref["sha256"],
        "model_version": ref["metadata"].get("model_version"),
        "n_predictions_current": n_current,
        "n_predictions_reference": int(len(pred_ref)),
        "prediction_drift": pred_drift,
        "calibration_drift": calibration_drift,
        "feature_drift": feature_drift,
    }
    metrics.update(check_alarm(metrics, cfg))

    # Persist each metric as row (audit trail D-09-B4)
    ts_utc = datetime.now(timezone.utc).isoformat()
    for m_name, m_value, m_thresh, m_breach in _flatten_to_rows(metrics):
        try:
            append_drift_row(drift_log_db_path, {
                "ts_utc": ts_utc,
                "model_version": metrics["model_version"],
                "reference_hash": metrics["reference_hash"],
                "metric_name": m_name,
                "metric_value": float(m_value) if m_value is not None and not _is_nan(m_value) else 0.0,
                "threshold": float(m_thresh) if m_thresh is not None else 0.0,
                "breached": bool(m_breach),
                "n_samples_current": metrics.get("n_predictions_current"),
                "n_samples_reference": metrics.get("n_predictions_reference"),
                "window_days": window_days,
                "computed_by": "mcp_get_drift_metrics",
            })
        except Exception:
            log.exception("get_drift_metrics: errore append_drift_row %s", m_name)
    return metrics


def _flatten_to_rows(metrics: dict) -> list[tuple]:
    """Trasforma metrics nested dict → flat rows per drift_log persistence."""
    rows = []
    pd = metrics.get("prediction_drift") or {}
    rows.append(("prediction_ks", pd.get("ks_pvalue"), pd.get("threshold"), pd.get("breached")))
    cd = metrics.get("calibration_drift") or {}
    rows.append(("calibration_ece", cd.get("delta_ece"), cd.get("threshold"), cd.get("breached")))
    for f in metrics.get("feature_drift") or []:
        rows.append((f"feature_drift_{f['feature']}", f["ks_stat"], pd.get("threshold"), f["breached"]))
    return rows
```
