def _load_current_window(window_days: int, cfg: Any) -> tuple:
    """Carica current-window prediction + outcomes per drift compute.

    Sources:
    - prediction array: ml_inference.log (Phase 7 D-07-05 rotativo, JSON lines)
    - outcomes (y_true): trades.db trades_log dove exit_reason resolved
    - df_features: trades.db trades_log + JSON decision_context_json columns

    Returns:
        (pred_cur: np.ndarray, y_cur: np.ndarray, df_cur: pd.DataFrame).
    """
    from datetime import datetime, timezone, timedelta
    from pathlib import Path
    import sqlite3, json as _json
    import numpy as _np
    import pandas as _pd
    from logger import _trades_db_path

    cutoff = datetime.now(timezone.utc) - timedelta(days=window_days)
    cutoff_iso = cutoff.isoformat()
    db_path = _trades_db_path(cfg)

    with sqlite3.connect(str(db_path)) as c:
        c.execute("PRAGMA journal_mode=WAL")
        rows = c.execute(
            "SELECT id, timestamp, ml_calibrated_prob, decision_context_json, "
            "pnl_realized FROM trades_log WHERE timestamp >= ? AND ml_calibrated_prob IS NOT NULL",
            (cutoff_iso,),
        ).fetchall()

    pred_list: list[float] = []
    y_list: list[int] = []
    feat_rows: list[dict] = []
    for tid, ts, prob, ctx_json, pnl in rows:
        if prob is None:
            continue
        pred_list.append(float(prob))
        # outcome derivation: pnl > 0 ⇒ assume TP_HIT (Phase 7 D-01 binary alignment)
        outcome = 1 if (pnl is not None and float(pnl) > 0) else 0
        y_list.append(outcome)
        ctx = _json.loads(ctx_json or "{}")
        ind = ctx.get("indicators", {})
        feat_rows.append({"trade_id": tid, **ind})

    pred_cur = _np.asarray(pred_list, dtype=_np.float64)
    y_cur = _np.asarray(y_list, dtype=_np.int64)
    df_cur = _pd.DataFrame(feat_rows)
    return pred_cur, y_cur, df_cur


def _load_reference_window(ref: dict) -> tuple:
    """Carica reference-window da baseline_decisions parquet.

    Returns:
        (pred_ref: np.ndarray, y_ref: np.ndarray, df_ref: pd.DataFrame).
    """
    from pathlib import Path
    import pandas as _pd
    import numpy as _np

    parquet_path = Path(ref["parquet_path"])
    df = _pd.read_parquet(parquet_path)
    # Reference prediction: usa colonna calibrated_prob se presente; altrimenti
    # cold-start: array vuoto (D-09-B2 nota — può accadere per legacy parquet
    # pre-Phase-8 ml-on run).
    if "calibrated_prob" in df.columns:
        pred_ref = df["calibrated_prob"].dropna().values.astype("float64")
    elif "ml_calibrated_prob" in df.columns:
        pred_ref = df["ml_calibrated_prob"].dropna().values.astype("float64")
    else:
        pred_ref = _np.array([], dtype=_np.float64)

    if "exit_reason" in df.columns:
        y_ref = (df["exit_reason"] == "TP_HIT").astype("int64").values
    else:
        y_ref = _np.array([], dtype=_np.int64)

    return pred_ref, y_ref, df
