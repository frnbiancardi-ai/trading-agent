"""position_action.rules — Phase 9 Area D D-09-D3 4 rule families + profile-aware lookup.

W10 PRECEDENCE (restructured per plan-checker iter-2 WARNING #10):
  1. Time-stop check FIRST — standalone override (bars > max AND r ≤ time_stop_r)
  2. R-multiple ladder elif chain — highest first (full → partial → BE → hold)
  3. Default "hold"

Documented order: time-stop SUBSUMES r-ladder. State con bars=120 + r=2.0 → action="full_close"
rule_name="time_stop" (NOT partial_close from r-ladder). Test guard:
test_time_stop_overrides_r_multiple_partial.
"""
from __future__ import annotations

import logging
from typing import Any


log = logging.getLogger(__name__)


def _max_hold_bars(profile: str, cfg: Any) -> int:
    return int(getattr(cfg, f"MAX_HOLD_BARS_{profile}"))


def _be_r(profile: str, cfg: Any) -> float:
    return float(getattr(cfg, f"BE_R_THRESHOLD_{profile}"))


def _partial_r(profile: str, cfg: Any) -> float:
    return float(getattr(cfg, f"PARTIAL_R_THRESHOLD_{profile}"))


def _full_r(profile: str, cfg: Any) -> float:
    return float(getattr(cfg, f"FULL_R_THRESHOLD_{profile}"))


def evaluate_rules(state: dict, cfg: Any) -> dict:
    """W10 PRECEDENCE: time-stop FIRST, then r-multiple elif chain (highest first)."""
    profile = str(state.get("profile", "MODERATE")).upper()
    if profile not in ("CONSERVATIVE", "MODERATE", "AGGRESSIVE"):
        log.warning("evaluate_rules: profile sconosciuto '%s', fallback MODERATE", profile)
        profile = "MODERATE"

    r_mult = float(state.get("r_multiple", 0.0) or 0.0)
    bars = int(state.get("bars_in_trade", 0) or 0)
    entry_price = state.get("entry_price")

    max_bars = _max_hold_bars(profile, cfg)
    time_stop_r = float(cfg.TIME_STOP_R_THRESHOLD)
    partial_fraction = float(cfg.PARTIAL_CLOSE_FRACTION)

    rules_triggered: list[dict] = []

    # === W10 step 1 — Time-stop check FIRST (standalone override) ===
    if bars > max_bars and r_mult <= time_stop_r:
        rules_triggered.append({
            "name": "time_stop",
            "condition": f"bars>{max_bars} AND r<={time_stop_r} (profile={profile})",
            "value": float(bars),
        })
        return {"action": "full_close", "params": {}, "rules_triggered": rules_triggered}

    # === W10 step 2 — R-multiple elif chain (highest first) ===
    full_threshold = _full_r(profile, cfg)
    partial_threshold = _partial_r(profile, cfg)
    be_threshold = _be_r(profile, cfg)

    if r_mult >= full_threshold:
        rules_triggered.append({
            "name": "r_multiple_full",
            "condition": f"r>={full_threshold} (profile={profile})",
            "value": r_mult,
        })
        return {"action": "full_close", "params": {}, "rules_triggered": rules_triggered}
    elif r_mult >= partial_threshold:
        rules_triggered.append({
            "name": "r_multiple_partial",
            "condition": f"r>={partial_threshold} (profile={profile})",
            "value": r_mult,
        })
        return {
            "action": "partial_close",
            "params": {"close_fraction": partial_fraction},
            "rules_triggered": rules_triggered,
        }
    elif r_mult >= be_threshold:
        rules_triggered.append({
            "name": "r_multiple_to_be",
            "condition": f"r>={be_threshold} (profile={profile})",
            "value": r_mult,
        })
        return {
            "action": "move_sl",
            "params": {"method": "to_breakeven", "new_sl_price": entry_price},
            "rules_triggered": rules_triggered,
        }

    # === W10 step 3 — Default ===
    return {"action": "hold", "params": {}, "rules_triggered": rules_triggered}


def modulate_with_ml(rule_outcome: dict, ml_check: dict, cfg: Any) -> tuple[str, dict, float]:
    """D-09-D1 hybrid: ML modula severity rule action.

    Logica:
    - below_threshold (con o senza _no_entry_ref) AND rule_action ∈ {hold, move_sl} → upgrade full_close
    - above_but_decreasing → no upgrade, confidence diminuisce
    - above_threshold (con o senza _no_entry_ref) → no upgrade, confidence boost
    - n/a (ENABLE_ML_FILTER=false) → no upgrade
    """
    action = rule_outcome["action"]
    params = dict(rule_outcome["params"])
    rules_triggered = rule_outcome["rules_triggered"]
    signal = (ml_check or {}).get("signal", "n/a")

    # W11: "below_threshold" matches both pure and "_no_entry_ref" variants
    is_below = signal.startswith("below_threshold")
    is_above = signal.startswith("above_threshold")

    if is_below and action in ("hold", "move_sl"):
        action = "full_close"
        params = {}
        rules_triggered.append({
            "name": "ml_thesis_invalidated_upgrade",
            "condition": "ml.signal startswith below_threshold AND rule.action in {hold, move_sl}",
            "value": float(ml_check.get("calibrated_prob_now") or 0.0),
        })

    base = 0.5 + 0.2 * len(rules_triggered)
    if is_above and signal != "above_but_decreasing":
        base += 0.1
    elif is_below:
        base += 0.05
    elif signal == "above_but_decreasing":
        base -= 0.05
    confidence = max(0.0, min(1.0, base))

    return action, params, float(confidence)

"""position_action.ml_recheck — Phase 9 Area D D-09-D1 + D-09-D4 mid-trade ML re-eval.

W11 (5 signal values): `prob_at_entry=None` non genera silent classification.
Branch dedicato → signal="above_threshold_no_entry_ref" o "below_threshold_no_entry_ref".

Tutti i 5 signal:
- "above_threshold"            : prob_at_entry available, prob_now > threshold, no ratio drop
- "above_but_decreasing"       : prob_at_entry available, prob_now > threshold, prob_now < prob_at_entry * 0.9
- "below_threshold"            : prob_at_entry available, prob_now <= threshold
- "above_threshold_no_entry_ref": prob_at_entry=None, prob_now > threshold (W11 EXPLICIT)
- "below_threshold_no_entry_ref": prob_at_entry=None, prob_now <= threshold (W11 EXPLICIT)
- "n/a"                        : ENABLE_ML_FILTER=false fallback

D-09-D1: hybrid rule + ML re-evaluation. OOD per costruzione (P9).
D-09-D4: prefer cached ml_calibrated_prob da trades_log; fallback recompute.
"""
from __future__ import annotations

import json

log = logging.getLogger(__name__)


def evaluate_ml_thesis(state: dict, position_id: int, cfg: Any) -> dict:
    """Mid-trade ML re-eval con W11 explicit prob_at_entry=None branch."""
    if not getattr(cfg, "ENABLE_ML_FILTER", False):
        return {
            "calibrated_prob_at_entry": None,
            "calibrated_prob_now": None,
            "threshold_profile": None,
            "signal": "n/a",
            "ood_caveat": False,
            "ml_filter_disabled": True,
        }

    prob_at_entry = _lookup_cached_prob(position_id, cfg)
    if prob_at_entry is None:
        prob_at_entry = _recompute_prob_at_entry(position_id, cfg)

    try:
        from ml.inference import get_ml_filter
        from ml.feature_extraction import build_feature_vector

        draft_now, ind_now, ctx_now = _build_inputs_now(state)
        feature_list = _read_feature_list_from_metadata(cfg)
        cat_encodings = _read_cat_encodings_from_metadata(cfg)
        fv = build_feature_vector(draft_now, ind_now, ctx_now, feature_list, cat_encodings)
        ml_filter = get_ml_filter(cfg.ML_MODEL_PATH)
        _, prob_now = ml_filter.predict(fv)
    except (ImportError, FileNotFoundError) as exc:
        log.warning("evaluate_ml_thesis: ML stack non disponibile (%s) → fallback n/a", exc)
        return {
            "calibrated_prob_at_entry": float(prob_at_entry) if prob_at_entry is not None else None,
            "calibrated_prob_now": None,
            "threshold_profile": None,
            "signal": "n/a",
            "ood_caveat": False,
            "ml_filter_disabled": False,
            "warning": "ml_stack_unavailable",
        }

    profile = str(state.get("profile", "MODERATE")).upper()
    threshold = _threshold_for_profile(profile, cfg)

    # W11: EXPLICIT branch for prob_at_entry=None — NO silent classification
    if prob_at_entry is None:
        signal = "above_threshold_no_entry_ref" if prob_now > threshold else "below_threshold_no_entry_ref"
    elif prob_now <= threshold:
        signal = "below_threshold"
    elif prob_now < prob_at_entry * 0.9:
        signal = "above_but_decreasing"
    else:
        signal = "above_threshold"

    return {
        "calibrated_prob_at_entry": float(prob_at_entry) if prob_at_entry is not None else None,
        "calibrated_prob_now": float(prob_now),
        "threshold_profile": float(threshold),
        "signal": signal,
        "ood_caveat": True,            # P9 MANDATORY when ML active
        "ml_filter_disabled": False,
    }


def _lookup_cached_prob(position_id: int, cfg: Any) -> float | None:
    """SELECT ml_calibrated_prob FROM trades_log WHERE position_id=? (D-09-D4)."""
    try:
        from logger import _trades_db_path
        db_path = _trades_db_path(cfg)
        with sqlite3.connect(str(db_path)) as c:
            c.execute("PRAGMA journal_mode=WAL")
            row = c.execute(
                "SELECT ml_calibrated_prob FROM trades_log WHERE id = ?",
                (int(position_id),),
            ).fetchone()
        if row is None or row[0] is None:
            return None
        return float(row[0])
    except Exception as exc:
        log.warning("_lookup_cached_prob: errore lookup pid=%s: %s", position_id, exc)
        return None

        db_path = _trades_db_path(cfg)
        with sqlite3.connect(str(db_path)) as c:
            c.execute("PRAGMA journal_mode=WAL")
            row = c.execute(
                "SELECT decision_context_json FROM trades_log WHERE id = ?",
                (int(position_id),),
            ).fetchone()
        if row is None or row[0] is None:
            return None
        decision_ctx = json.loads(row[0])
        draft = decision_ctx.get("draft", {})
        ind = decision_ctx.get("indicators", {})
        ctx = decision_ctx.get("ctx", {})

        from ml.inference import get_ml_filter
        from ml.feature_extraction import build_feature_vector

        feature_list = _read_feature_list_from_metadata(cfg)
        cat_encodings = _read_cat_encodings_from_metadata(cfg)
        fv = build_feature_vector(draft, ind, ctx, feature_list, cat_encodings)
        ml_filter = get_ml_filter(cfg.ML_MODEL_PATH)
        _, prob = ml_filter.predict(fv)
        return float(prob)
    except Exception as exc:
        log.warning("_recompute_prob_at_entry: pid=%s errore %s", position_id, exc)
        return None


def _build_inputs_now(state: dict) -> tuple[dict, dict, dict]:
    """Costruisce (draft, indicators, ctx) NOW da position state corrente."""
    draft = state.get("draft_now", state.get("draft", {}))
    indicators = state.get("indicators_now", state.get("indicators", {}))
    ctx = state.get("ctx_now", {
        "symbol": state.get("symbol"),
        "timeframe": state.get("timeframe"),
        "profile": state.get("profile"),
    })
    return draft, indicators, ctx


def _threshold_for_profile(profile: str, cfg: Any) -> float:
    threshold_map = getattr(cfg, "ML_THRESHOLD_BY_PROFILE", None)
    if isinstance(threshold_map, dict):
        return float(threshold_map.get(profile, threshold_map.get("MODERATE", 0.55)))
    return 0.55


def _read_feature_list_from_metadata(cfg: Any) -> list[str]:
    try:
        from pathlib import Path
        meta_path = Path(cfg.ML_MODEL_PATH).with_suffix(".metadata.json")
        meta = json.loads(meta_path.read_text(encoding="utf-8"))
        return list(meta.get("feature_list", []))
    except Exception:
        return []


def _read_cat_encodings_from_metadata(cfg: Any) -> dict:
    try:
        from pathlib import Path
        meta_path = Path(cfg.ML_MODEL_PATH).with_suffix(".metadata.json")
        meta = json.loads(meta_path.read_text(encoding="utf-8"))
        return dict(meta.get("categorical_encodings", {}))
    except Exception:
        return {}

def _format_summary_it(rule_outcome: dict, ml_check: dict, final_action: str) -> str:
    """Italiano operator-readable summary — CLAUDE.md + P15 + W11.

    Required markers (≥1): suggerisco|confermata|invalidata|raggiunto|caveat.
    Forbidden phrases (P9 OOD guard): "probabilità che vinca", "P(TP_HIT)", "odds of TP".

    elif signal.startswith("below_threshold"):
        prob_now = ml_check.get("calibrated_prob_now") or 0.0
        threshold = ml_check.get("threshold_profile") or 0.0

    return f"Suggerisco {final_action} ({rules_part}). {ml_part}."
