```python
"""position_action.rules — Phase 9 Area D D-09-D3 4 rule families + profile-aware lookup.

4 rule families:
1. R-multiple BE move (T_be_profile, default 1.0/1.5/2.0)
2. R-multiple partial close (T_partial_profile, default 1.5/2.0/2.5, fraction = PARTIAL_CLOSE_FRACTION)
3. R-multiple full close (T_full_profile, default 2.5/3.0/3.5)
4. Time stop (bars > MAX_HOLD_BARS_profile AND r ≤ TIME_STOP_R_THRESHOLD)
"""
from __future__ import annotations

import logging
from typing import Any


log = logging.getLogger(__name__)


def evaluate_rules(state: dict, cfg: Any) -> dict:
    """4 rule families con threshold profile-aware via .env getattr dinamico.

    Args:
        state: position state dict da Phase 6 handle_get_position_state.
               Required keys: r_multiple, bars_in_trade, profile, entry_price.
        cfg: Config con BE_R_THRESHOLD_{PROFILE}, PARTIAL_R_THRESHOLD_{PROFILE},
             FULL_R_THRESHOLD_{PROFILE}, MAX_HOLD_BARS_{PROFILE},
             PARTIAL_CLOSE_FRACTION, TIME_STOP_R_THRESHOLD.

    Returns:
        {action: str, params: dict, rules_triggered: list[dict]}.
    """
    profile = str(state.get("profile", "MODERATE")).upper()
    if profile not in ("CONSERVATIVE", "MODERATE", "AGGRESSIVE"):
        log.warning("evaluate_rules: profile sconosciuto '%s', fallback MODERATE", profile)
        profile = "MODERATE"

    r_mult = float(state.get("r_multiple", 0.0) or 0.0)
    bars = int(state.get("bars_in_trade", 0) or 0)
    entry_price = state.get("entry_price")

    be_threshold = float(getattr(cfg, f"BE_R_THRESHOLD_{profile}"))
    partial_threshold = float(getattr(cfg, f"PARTIAL_R_THRESHOLD_{profile}"))
    full_threshold = float(getattr(cfg, f"FULL_R_THRESHOLD_{profile}"))
    max_bars = int(getattr(cfg, f"MAX_HOLD_BARS_{profile}"))
    partial_fraction = float(cfg.PARTIAL_CLOSE_FRACTION)
    time_stop_r = float(cfg.TIME_STOP_R_THRESHOLD)

    rules_triggered: list[dict] = []
    action = "hold"
    params: dict = {}

    # Rule 3 — Full close (highest R, evaluated first)
    if r_mult >= full_threshold:
        action = "full_close"
        rules_triggered.append({
            "name": "r_multiple_full",
            "condition": f"r>={full_threshold} (profile={profile})",
            "value": r_mult,
        })
    # Rule 2 — Partial close
    elif r_mult >= partial_threshold:
        action = "partial_close"
        params = {"close_fraction": partial_fraction}
        rules_triggered.append({
            "name": "r_multiple_partial",
            "condition": f"r>={partial_threshold} (profile={profile})",
            "value": r_mult,
        })
    # Rule 1 — BE move
    elif r_mult >= be_threshold:
        action = "move_sl"
        params = {"method": "to_breakeven", "new_sl_price": entry_price}
        rules_triggered.append({
            "name": "r_multiple_to_be",
            "condition": f"r>={be_threshold} (profile={profile})",
            "value": r_mult,
        })
    # Rule 4 — Time stop (independent / cumulative with above)
    if bars > max_bars and r_mult <= time_stop_r and action == "hold":
        action = "full_close"
        rules_triggered.append({
            "name": "time_stop",
            "condition": f"bars>{max_bars} AND r<={time_stop_r} (profile={profile})",
            "value": float(bars),
        })

    return {"action": action, "params": params, "rules_triggered": rules_triggered}


def modulate_with_ml(rule_outcome: dict, ml_check: dict, cfg: Any) -> tuple[str, dict, float]:
    """D-09-D1 hybrid: ML modula severity rule action.

    Logica:
    - below_threshold (ML invalida thesis) AND rule_action ∈ {hold, move_sl} → upgrade full_close
    - above_but_decreasing → no upgrade, ma confidence diminuisce
    - above_threshold → no upgrade, confidence boost
    - n/a (ENABLE_ML_FILTER=false) → no upgrade
    """
    action = rule_outcome["action"]
    params = dict(rule_outcome["params"])
    rules_triggered = rule_outcome["rules_triggered"]
    signal = (ml_check or {}).get("signal", "n/a")

    if signal == "below_threshold" and action in ("hold", "move_sl"):
        action = "full_close"
        params = {}  # full_close no params
        rules_triggered.append({
            "name": "ml_thesis_invalidated_upgrade",
            "condition": "ml.signal == below_threshold AND rule.action in {hold, move_sl}",
            "value": float(ml_check.get("calibrated_prob_now") or 0.0),
        })

    # Confidence: base 0.5 + 0.2 per rule + ±0.1 ML
    base = 0.5 + 0.2 * len(rules_triggered)
    if signal == "above_threshold":
        base += 0.1
    elif signal == "below_threshold":
        base += 0.05  # net negative since action upgraded
    elif signal == "above_but_decreasing":
        base -= 0.05
    confidence = max(0.0, min(1.0, base))

    return action, params, float(confidence)
```

```python
"""position_action.ml_recheck — Phase 9 Area D D-09-D1 + D-09-D4 mid-trade ML re-eval.

D-09-D1: hybrid rule + ML re-evaluation. OOD per costruzione (P9): classifier trainato
         su features_at_decision_time (Phase 7 D-17), mid-trade input distribution è OOD.
D-09-D4: prefer cached ml_calibrated_prob da trades_log; fallback recompute via
         decision_context_json + MLFilter.predict (deterministic).

Pattern OOD safe: relative-only signal classification (prob_now vs threshold; prob_now vs prob_at_entry).
MAI absolute interpretation di prob_now (forbidden phrase "probabilità che vinca").
"""
from __future__ import annotations

import json
import logging
import sqlite3
from typing import Any


log = logging.getLogger(__name__)


def evaluate_ml_thesis(state: dict, position_id: int, cfg: Any) -> dict:
    """Mid-trade ML re-eval con OOD caveat surface (P9 + D-09-D1).

    Args:
        state: position state da Phase 6 handle_get_position_state.
        position_id: per cache lookup trades_log.ml_calibrated_prob.
        cfg: Config con ENABLE_ML_FILTER + ML_MODEL_PATH + ML_THRESHOLD_BY_PROFILE.

    Returns:
        dict con calibrated_prob_at_entry, calibrated_prob_now, threshold_profile,
        signal, ood_caveat, ml_filter_disabled. Tutti None se disabled.
    """
    if not getattr(cfg, "ENABLE_ML_FILTER", False):
        # Phase 7 D-07-06-11 fallback: zero-impact rollout
        return {
            "calibrated_prob_at_entry": None,
            "calibrated_prob_now": None,
            "threshold_profile": None,
            "signal": "n/a",
            "ood_caveat": False,
            "ml_filter_disabled": True,
        }

    # D-09-D4: prefer cached prob_at_entry da trades_log
    prob_at_entry = _lookup_cached_prob(position_id, cfg)
    if prob_at_entry is None:
        prob_at_entry = _recompute_prob_at_entry(position_id, cfg)

    # Compute prob NOW via build_feature_vector + MLFilter.predict
    try:
        from ml.inference import get_ml_filter
        from ml.feature_extraction import build_feature_vector

        draft_now, ind_now, ctx_now = _build_inputs_now(state)
        feature_list = _read_feature_list_from_metadata(cfg)
        cat_encodings = _read_cat_encodings_from_metadata(cfg)
        fv = build_feature_vector(draft_now, ind_now, ctx_now, feature_list, cat_encodings)
        ml_filter = get_ml_filter(cfg.ML_MODEL_PATH)
        _, prob_now = ml_filter.predict(fv)
    except (ImportError, FileNotFoundError) as exc:
        log.warning("evaluate_ml_thesis: ML stack non disponibile (%s) → fallback n/a", exc)
        return {
            "calibrated_prob_at_entry": float(prob_at_entry) if prob_at_entry is not None else None,
            "calibrated_prob_now": None,
            "threshold_profile": None,
            "signal": "n/a",
            "ood_caveat": False,
            "ml_filter_disabled": False,
            "warning": "ml_stack_unavailable",
        }

    profile = str(state.get("profile", "MODERATE")).upper()
    threshold = _threshold_for_profile(profile, cfg)

    # Signal classification (D-09 § specifics OOD caveat boilerplate step 5)
    # SAFE: only relative interpretation (prob_now vs threshold; prob_now vs prob_at_entry ratio).
    # NEVER absolute interpretation (P9 forbidden in summary_it).
    if prob_now <= threshold:
        signal = "below_threshold"
    elif prob_at_entry is not None and prob_now < prob_at_entry * 0.9:
        signal = "above_but_decreasing"
    else:
        signal = "above_threshold"

    return {
        "calibrated_prob_at_entry": float(prob_at_entry) if prob_at_entry is not None else None,
        "calibrated_prob_now": float(prob_now),
        "threshold_profile": float(threshold),
        "signal": signal,
        "ood_caveat": True,                 # P9 MANDATORY when ML active
        "ml_filter_disabled": False,
    }


def _lookup_cached_prob(position_id: int, cfg: Any) -> float | None:
    """SELECT ml_calibrated_prob FROM trades_log WHERE position_id=? (D-09-D4)."""
    try:
        from logger import _trades_db_path
        db_path = _trades_db_path(cfg)
        with sqlite3.connect(str(db_path)) as c:
            c.execute("PRAGMA journal_mode=WAL")
            row = c.execute(
                "SELECT ml_calibrated_prob FROM trades_log WHERE id = ?",
                (int(position_id),),
            ).fetchone()
        if row is None or row[0] is None:
            return None
        return float(row[0])
    except Exception as exc:
        log.warning("_lookup_cached_prob: errore lookup pid=%s: %s", position_id, exc)
        return None


def _recompute_prob_at_entry(position_id: int, cfg: Any) -> float | None:
    """Fallback recompute via decision_context_json + MLFilter.predict (D-09-D4)."""
    try:
        from logger import _trades_db_path
        db_path = _trades_db_path(cfg)
        with sqlite3.connect(str(db_path)) as c:
            c.execute("PRAGMA journal_mode=WAL")
            row = c.execute(
                "SELECT decision_context_json FROM trades_log WHERE id = ?",
                (int(position_id),),
            ).fetchone()
        if row is None or row[0] is None:
            return None
        decision_ctx = json.loads(row[0])
        draft = decision_ctx.get("draft", {})
        ind = decision_ctx.get("indicators", {})
        ctx = decision_ctx.get("ctx", {})

        from ml.inference import get_ml_filter
        from ml.feature_extraction import build_feature_vector

        feature_list = _read_feature_list_from_metadata(cfg)
        cat_encodings = _read_cat_encodings_from_metadata(cfg)
        fv = build_feature_vector(draft, ind, ctx, feature_list, cat_encodings)
        ml_filter = get_ml_filter(cfg.ML_MODEL_PATH)
        _, prob = ml_filter.predict(fv)
        return float(prob)
    except Exception as exc:
        log.warning("_recompute_prob_at_entry: pid=%s errore %s", position_id, exc)
        return None


def _build_inputs_now(state: dict) -> tuple[dict, dict, dict]:
    """Costruisce (draft, indicators, ctx) NOW da position state corrente.

    Wave 6 integration: source delle current indicators via BarSource (Phase 6 D-D1).
    Per Wave 4 standalone: draft NOW = legacy draft (state-dict shape), indicators
    placeholder (operator-supplied), ctx con profile/symbol/timeframe.
    """
```

```python
"""position_action.orchestrator — Phase 9 Area D MCP-18 hybrid + structured rationale.

D-09-D1: hybrid rule + ML; D-09-D2: structured rationale + summary_it (CLAUDE.md italiano).
P15 guard: summary_it MUST contain Italian markers, MUST NOT contain absolute prob phrases.
"""
from __future__ import annotations

import logging
from typing import Any


log = logging.getLogger(__name__)


def handle_suggest_position_action(args: dict, mt5_client: Any, cfg: Any, log_handle: Any = None) -> dict:
    """MCP-18 D-09-D — hold/move_sl/partial_close/full_close + structured rationale.

    READ-ONLY: NON muta posizione. Skill chiama modify_position se accetta.
    """
    pid_raw = args.get("position_id")
    try:
        pid = int(pid_raw)
    except (TypeError, ValueError):
        from mcp_tools.errors import ErrorCodes, envelope
        return envelope(ErrorCodes.VALIDATION_FAILED, f"position_id non valido: {pid_raw}")

    # Step 1: stato posizione via Phase 6 handler (NO mt5 direct call)
    from mcp_tools.handlers.position import handle_get_position_state
    state = handle_get_position_state({"position_id": pid}, mt5_client, cfg)
    if not isinstance(state, dict) or state.get("ok") is False:
        return state  # propaga error envelope

    # Step 2: rule-based primary (D-09-D3)
    from position_action.rules import evaluate_rules, modulate_with_ml
    rule_outcome = evaluate_rules(state, cfg)

    # Step 3: ML re-eval (D-09-D1, OOD caveat surface P9)
    from position_action.ml_recheck import evaluate_ml_thesis
    ml_check = evaluate_ml_thesis(state, pid, cfg)

    # Step 4: hybrid combine — ML modula severity
    final_action, final_params, confidence = modulate_with_ml(rule_outcome, ml_check, cfg)

    # Step 5: structured rationale (D-09-D2)
    rationale = {
        "rules_triggered": rule_outcome["rules_triggered"],
        "ml_check": ml_check,
        "summary_it": _format_summary_it(rule_outcome, ml_check, final_action),
    }

    return {
        "position_id": pid,
        "action": final_action,
        "params": final_params,
        "confidence": confidence,
        "rationale": rationale,
    }


def _format_summary_it(rule_outcome: dict, ml_check: dict, final_action: str) -> str:
    """Italiano operator-readable summary — CLAUDE.md + P15.

    Required markers (≥1): suggerisco|confermata|invalidata|raggiunto|caveat.
    Forbidden phrases (P9 OOD guard): "probabilità che vinca", "P(TP_HIT)", "odds of TP".
    """
    rules_part = ", ".join(r["name"] for r in rule_outcome["rules_triggered"]) or "nessuna regola attivata"

    if ml_check.get("ml_filter_disabled"):
        ml_part = "ML filter disabilitato (solo regole)"
    elif ml_check.get("signal") == "n/a":
        ml_part = "ML stack non disponibile"
    elif ml_check.get("signal") == "below_threshold":
        prob_now = ml_check.get("calibrated_prob_now") or 0.0
        threshold = ml_check.get("threshold_profile") or 0.0
        ml_part = (
            f"thesis invalidata (prob {prob_now:.2f} < soglia {threshold:.2f}) — caveat OOD"
        )
    elif ml_check.get("signal") == "above_but_decreasing":
        prob_entry = ml_check.get("calibrated_prob_at_entry") or 0.0
        prob_now = ml_check.get("calibrated_prob_now") or 0.0
        ml_part = (
            f"thesis in calo (entry {prob_entry:.2f} → now {prob_now:.2f}) — caveat OOD"
        )
    else:
        ml_part = "thesis confermata ML — caveat OOD"

    return f"Suggerisco {final_action} ({rules_part}). {ml_part}."
```
