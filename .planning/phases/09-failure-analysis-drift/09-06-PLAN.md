```python
"""drift.report_writer — Phase 9 INT-03 markdown dashboard generator.

Mirror Pattern K `backtest/baseline/report_writer.py` — 6 sezioni:
1. Header (timestamp UTC + git_sha + cli_command + reference_period + current_period)
2. Prediction KS table
3. Calibration ECE table (current vs reference + delta + breach)
4. Feature drift top-K table
5. Alarm status italiano (summary)
6. Appendix (reference_hash + .env thresholds + sample counts)

Determinism: timestamp + git_sha esposti come `meta` arg → caller controlla per byte-identical output.
"""
from __future__ import annotations

import subprocess
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


def _get_git_sha() -> str:
    try:
        result = subprocess.run(
            ["git", "rev-parse", "--short", "HEAD"],
            capture_output=True, text=True, timeout=5,
        )
        return result.stdout.strip() or "unknown"
    except Exception:
        return "unknown"


def write_drift_report(
    metrics: dict,
    out_path: str | Path,
    meta: dict | None = None,
) -> None:
    """Genera markdown report 6-section da drift metrics dict.

    Args:
        metrics: output di `handle_get_drift_metrics` (CONTEXT.md § specifics lines 316-345 shape).
        out_path: percorso file `.md` (es. `.planning/research/drift-report-{date}.md`).
        meta: optional dict con `cli_command`, `git_sha`, `timestamp` per determinism control.
    """
    meta = meta or {}
    out_path = Path(out_path)
    out_path.parent.mkdir(parents=True, exist_ok=True)

    timestamp = meta.get("timestamp") or datetime.now(timezone.utc).isoformat()
    git_sha = meta.get("git_sha") or _get_git_sha()
    cli_command = meta.get("cli_command", "n/a")

    lines: list[str] = []

    # === Section 1 — Header ===
    lines.append("# Drift Report — Phase 9 INT-03\n")
    lines.append("## Header\n")
    lines.append(f"- **Data run**: {timestamp}")
    lines.append(f"- **Comando CLI**: `{cli_command}`")
    lines.append(f"- **git_sha**: `{git_sha}`")
    lines.append(f"- **Reference period**: {metrics.get('reference_period', 'n/a')}")
    lines.append(f"- **Current period**: {metrics.get('current_period', 'n/a')}")
    lines.append(f"- **Model version**: `{metrics.get('model_version', 'n/a')}`")
    lines.append(f"- **Window days**: {metrics.get('window_days', 'n/a')}")
    lines.append(f"- **n_predictions_reference**: {metrics.get('n_predictions_reference', 0)}")
    lines.append(f"- **n_predictions_current**: {metrics.get('n_predictions_current', 0)}")
    lines.append("")

    # === Section 2 — Prediction KS table ===
    lines.append("## Prediction Distribution KS\n")
    lines.append("| Metric | Value | Threshold | Breached |")
    lines.append("|--------|------:|----------:|:--------:|")
    pd = metrics.get("prediction_drift") or {}
    lines.append(
        f"| ks_stat | {pd.get('ks_stat', 0.0):.4f} | n/a | n/a |"
    )
    breached_pred = "yes" if pd.get("breached") else "no"
    lines.append(
        f"| ks_pvalue | {pd.get('ks_pvalue', 0.0):.4f} | "
        f"{pd.get('threshold', 0.0):.4f} | {breached_pred} |"
    )
    lines.append("")

    # === Section 3 — Calibration ECE table ===
    lines.append("## Calibration ECE\n")
    lines.append("| Metric | Value | Threshold | Breached |")
    lines.append("|--------|------:|----------:|:--------:|")
    cd = metrics.get("calibration_drift") or {}
    cur_ece = cd.get("current_ece")
    ref_ece = cd.get("reference_ece")
    delta = cd.get("delta_ece")
    breached_ece = "yes" if cd.get("breached") else "no"
    lines.append(
        f"| reference_ece | {ref_ece if ref_ece is not None else 'n/a'} | n/a | n/a |"
    )
    lines.append(
        f"| current_ece | {cur_ece if cur_ece is not None else 'n/a (cold-start)'} | n/a | n/a |"
    )
    lines.append(
        f"| delta_ece | {delta if delta is not None else 'n/a'} | "
        f"{cd.get('threshold', 0.0):.4f} | {breached_ece} |"
    )
    if cd.get("cold_start"):
        lines.append("\n> ⚠ Cold-start: n_samples_current sotto soglia DRIFT_MIN_SAMPLES_CURRENT.\n")
    lines.append("")

    # === Section 4 — Feature drift top-K table ===
    lines.append("## Feature Drift Top-K\n")
    lines.append("| Feature | ks_stat | ks_pvalue | Breached |")
    lines.append("|---------|--------:|----------:|:--------:|")
    for f in metrics.get("feature_drift") or []:
        breached_f = "yes" if f.get("breached") else "no"
        lines.append(
            f"| `{f.get('feature', '?')}` | {f.get('ks_stat', 0.0):.4f} | "
            f"{f.get('ks_pvalue', 0.0):.4f} | {breached_f} |"
        )
    if not metrics.get("feature_drift"):
        lines.append("| _nessuna feature_ | n/a | n/a | n/a |")
    lines.append("")

    # === Section 5 — Alarm status italiano ===
    lines.append("## Alarm Status\n")
    alarm_status = metrics.get("alarm_status", "ok")
    breaches = metrics.get("breaches", [])
    if alarm_status == "breach":
        lines.append(f"**Stato**: ⚠ BREACH — {len(breaches)} di 3 tier violati.")
        lines.append(f"**Tier violati**: `{', '.join(breaches)}`")
        lines.append(
            "**Azione raccomandata**: investigare drift root-cause; "
            "verificare se retrain scheduled in arrivo (cron) o invocare `trigger_retrain` manuale (MCP-09)."
        )
    else:
        lines.append("**Stato**: ✓ OK — nessun tier violato.")
        lines.append("**Azione raccomandata**: nessuna; continuare monitoraggio.")
    lines.append("")

    # === Section 6 — Appendix ===
    lines.append("## Appendix — Audit Anchors\n")
    lines.append(f"- **reference_hash (sha256 baseline parquet)**: `{metrics.get('reference_hash', 'n/a')}`")
    lines.append(f"- **n_samples_reference**: {metrics.get('n_predictions_reference', 0)}")
    lines.append(f"- **n_samples_current**: {metrics.get('n_predictions_current', 0)}")
    if cd.get("n_with_outcome") is not None:
        lines.append(f"- **n_with_outcome (current ECE input)**: {cd.get('n_with_outcome')}")
    if cd.get("n_pending_outcome") is not None:
        lines.append(f"- **n_pending_outcome**: {cd.get('n_pending_outcome')}")
    lines.append("\n### .env Thresholds (Audit)\n")
    lines.append(f"- `DRIFT_KS_PVALUE_THRESHOLD = {pd.get('threshold', 'n/a')}`")
    lines.append(f"- `DRIFT_ECE_DELTA_THRESHOLD = {cd.get('threshold', 'n/a')}`")
    lines.append("")

    out_path.write_text("\n".join(lines), encoding="utf-8")
```

```python
"""scripts.show_drift — Phase 9 INT-03 CLI dashboard.

Exit codes:
    0 = OK no breach
    1 = drift breach detected (per future cron integration Phase 11+)
    2 = input data missing (ml model bundle, baseline parquet, drift_log.db)
    3 = schema / integrity error (P14 sha256 mismatch)
    4 = unexpected error

Usage:
    python scripts/show_drift.py --window-days 30 --out-path .planning/research/drift-report-{date}.md
    python scripts/show_drift.py --window-days 7 --exit-on-breach
"""
from __future__ import annotations

import argparse
import logging
import sys
from datetime import date
from pathlib import Path


ROOT = Path(__file__).resolve().parent.parent
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))


logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
_log = logging.getLogger("show_drift")


def main() -> int:
    parser = argparse.ArgumentParser(description="Phase 9 INT-03 — Drift dashboard CLI")
    parser.add_argument("--window-days", type=int, default=30, help="Window in days (default 30)")
    parser.add_argument(
        "--out-path", type=Path,
        default=Path(".planning/research") / f"drift-report-{date.today()}.md",
        help="Output markdown path",
    )
    parser.add_argument(
        "--exit-on-breach", action="store_true",
        help="Exit 1 se breach (per cron integration). Default exit 0 anche se breach.",
    )
    args = parser.parse_args()

    try:
        from config import Config
        cfg = Config()
    except Exception as exc:
        _log.error("Config init fallita: %s", exc)
        return 4

    try:
        from logger import _trades_db_path
        drift_db = str(Path(_trades_db_path(cfg)).parent / "drift_log.db")
    except Exception as exc:
        _log.error("trades_db_path lookup fallito: %s", exc)
        return 2

    try:
        from drift.log_db import ensure_drift_log_db
        ensure_drift_log_db(drift_db)
    except Exception as exc:
        _log.error("drift_log.db init fallita: %s", exc)
        return 2

    # Invoca handler MCP-08 (compose pipeline + persist + return metrics)
    try:
        from mcp_tools.handlers.drift import handle_get_drift_metrics
        metrics = handle_get_drift_metrics(
            {"window_days": args.window_days}, cfg, drift_db,
        )
    except Exception as exc:
        _log.exception("handle_get_drift_metrics fallita")
        return 4

    if isinstance(metrics, dict) and metrics.get("ok") is False:
        error_code = metrics.get("error", "")
        if error_code in ("not_found",):
            _log.error("Input mancante: %s", metrics.get("message"))
            return 2
        if error_code in ("internal_error",):
            _log.error("Integrity error: %s", metrics.get("message"))
            return 3
        _log.error("Errore: %s", metrics.get("message"))
        return 4

    # Genera markdown report
    try:
        from drift.report_writer import write_drift_report
        write_drift_report(
            metrics, args.out_path,
            meta={"cli_command": " ".join(sys.argv), "git_sha": None, "timestamp": None},
        )
        _log.info("Drift report scritto: %s", args.out_path)
    except Exception as exc:
        _log.exception("write_drift_report fallita")
        return 4

    breached = bool(metrics.get("alarm_status") == "breach")
    if breached:
        _log.warning("Drift BREACH rilevato: %s", metrics.get("breaches"))
        if args.exit_on_breach:
            return 1
    else:
        _log.info("Drift OK — nessun tier violato.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
