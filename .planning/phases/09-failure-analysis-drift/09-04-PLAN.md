```python
"""retrain.dedup — Phase 9 Area C D-09-C3 idempotent dedup vs JobQueue cap=1.

Riusa `mcp_tools.job_queue.JobQueue.submit` run_in_progress branch — converte shape
ritorno in `{ok: true, status: "already_pending"}` (idempotent MCP contract).
D-09-C1: enqueue Phase 8 `_train_ml_filter_worker` (zero duplicazione code).
"""
from __future__ import annotations

import logging
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


log = logging.getLogger(__name__)


def enqueue_or_dedup(
    job_queue: Any,
    source: str,
    cfg: Any,
    reason: str = "",
) -> dict:
    """3 trigger source (drift_breach, scheduled_cron, manual_mcp) condividono dedup.

    Args:
        job_queue: mcp_tools.job_queue.JobQueue (cap=1 condivisa Phase 6/8).
        source: enum ∈ {"drift_breach", "scheduled_cron", "manual_mcp"}.
        cfg: Config con MCP_TRAINING_DATA_PATH + ML_MODEL_PATH.
        reason: optional human-readable string per audit.

    Returns:
        {ok: bool, status: "started"|"already_pending"|"validation_failed"|...,
         job_id: str, triggered_by_sources: list[str]}.
    """
    # Phase 8 dependency: _train_ml_filter_worker (top-level picklable, D-08-A1)
    from mcp_tools.handlers.ml import _train_ml_filter_worker, _validate_parquet_schema_v2

    parquet_path = Path(cfg.MCP_TRAINING_DATA_PATH)
    ok, msg, parquet_sha = _validate_parquet_schema_v2(parquet_path)
    if not ok:
        from mcp_tools.errors import ErrorCodes, envelope
        return envelope(ErrorCodes.VALIDATION_FAILED, msg)

    run_id = _build_retrain_run_id(source)
    metadata = {
        "source": source,
        "reason": reason,
        "parquet_sha256": parquet_sha,
        "triggered_by_sources": [source],
    }
    result = job_queue.submit(
        run_id, _train_ml_filter_worker, metadata,
        run_id, str(parquet_path), str(Path(cfg.ML_MODEL_PATH).parent),
        "data/configs/ml.yaml", parquet_sha,
    )

    if isinstance(result, dict) and result.get("error") == "run_in_progress":
        # D-09-C3 idempotent: merge source into existing job's sources array
        active_id = result.get("active_run_id")
        _append_source_to_active(job_queue, active_id, source)
        return {
            "ok": True,
            "status": "already_pending",
            "job_id": active_id,
            "triggered_by_sources": _read_sources(job_queue, active_id),
        }

    # Success path: result è dict da JobQueue.submit; aggiungi sources
    if isinstance(result, dict):
        result.setdefault("triggered_by_sources", [source])
        result.setdefault("status", "started")
    return result


def _build_retrain_run_id(source: str) -> str:
    """run_id univoco per audit: mcp_retrain_{utc_ts}_{source}."""
    ts = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
    return f"mcp_retrain_{ts}_{source}"


def _append_source_to_active(job_queue: Any, active_id: str, source: str) -> None:
    """Append source a JobRecord.metadata.triggered_by_sources (audit completeness)."""
    try:
        job = job_queue._jobs.get(active_id) if hasattr(job_queue, "_jobs") else None
        if job is not None and hasattr(job, "metadata"):
            existing = list(job.metadata.get("triggered_by_sources", []))
            if source not in existing:
                existing.append(source)
                job.metadata["triggered_by_sources"] = existing
    except Exception:
        log.exception("dedup: errore append source to active job %s", active_id)


def _read_sources(job_queue: Any, active_id: str) -> list[str]:
    """Read triggered_by_sources from active JobRecord."""
    try:
        job = job_queue._jobs.get(active_id) if hasattr(job_queue, "_jobs") else None
        if job is not None and hasattr(job, "metadata"):
            return list(job.metadata.get("triggered_by_sources", []))
    except Exception:
        log.exception("dedup: errore lettura sources")
    return []
```

```python
"""retrain.scheduler — Phase 9 Area C D-09-C1 APScheduler monthly cron.

Coesiste con scheduler.py BlockingScheduler ordinary cycle (process owner).
BackgroundScheduler thread-based, no shared persistent jobstore (P4 race guard).
"""
from __future__ import annotations

import logging
from typing import Any

from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.cron import CronTrigger


log = logging.getLogger(__name__)


class RetrainScheduler:
    """Monthly retrain via APScheduler BackgroundScheduler + JobQueue cap=1 condivisa.

    P4: MemoryJobStore implicit default (no `jobstore=` arg) — no shared persistent
        race con scheduler.py main loop.
    P5: replace_existing=True per restart safety (no ConflictingIdError).
    P13: misfire_grace_time=3600 (1h) per agent downtime tolerance.
    """

    def __init__(self, job_queue: Any, cfg: Any) -> None:
        # P4: MemoryJobStore implicit (no jobstore= arg)
        self._scheduler = BackgroundScheduler(timezone="UTC")
        self._job_queue = job_queue
        self._cfg = cfg

    def start(self) -> None:
        """Start scheduler con cron expression .env-driven + P5/P13 guards."""
        trigger = CronTrigger.from_crontab(self._cfg.RETRAIN_CRON_SCHEDULE)
        self._scheduler.add_job(
            self._fire_scheduled_retrain,
            trigger=trigger,
            id="retrain_monthly",
            replace_existing=True,        # P5 restart-safe
            max_instances=1,
            coalesce=True,
            misfire_grace_time=3600,      # P13 1h grace
        )
        self._scheduler.start()
        log.info("RetrainScheduler started — cron=%s tz=UTC", self._cfg.RETRAIN_CRON_SCHEDULE)

    def shutdown(self, wait: bool = False) -> None:
        """Stop scheduler (used in test teardown / app shutdown)."""
        if self._scheduler.running:
            self._scheduler.shutdown(wait=wait)

    def _fire_scheduled_retrain(self) -> None:
        """Tick callback — D-09-C3 dedup idempotent."""
        from retrain.dedup import enqueue_or_dedup
        result = enqueue_or_dedup(
            self._job_queue, source="scheduled_cron", cfg=self._cfg,
        )
        log.info(
            "RetrainScheduler tick — status=%s job_id=%s sources=%s",
            result.get("status"), result.get("job_id"),
            result.get("triggered_by_sources"),
        )
```

```python
"""retrain.validation_gate — Phase 9 Area C D-09-C2 auto-validation gate.

3 check pre atomic promote:
1. ECE regression vs metadata.aggregate.ece (D-09-C2 check 1, .env MAX_ECE_REGRESSION_PCT)

_VALIDATION_LOG_DDL = """
CREATE TABLE IF NOT EXISTS validation_log (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  ts_utc TEXT NOT NULL,
  candidate_path TEXT NOT NULL,
  previous_version TEXT,
  check_name TEXT NOT NULL,
  metric_value REAL NOT NULL,
  threshold REAL NOT NULL,
  passed INTEGER NOT NULL,
  reject_reason TEXT,
  triggered_by_sources_json TEXT
)
"""


def ensure_validation_log_db(db_path: str | Path) -> None:
    """Idempotent WAL — mirror drift_log.db pattern."""
    db_path = Path(db_path)
    db_path.parent.mkdir(parents=True, exist_ok=True)
    with sqlite3.connect(str(db_path)) as c:
        c.execute("PRAGMA journal_mode=WAL")
        c.execute(_VALIDATION_LOG_DDL)
        c.execute(
            "CREATE INDEX IF NOT EXISTS idx_validation_log_ts ON validation_log("
            "ts_utc, candidate_path, passed)"
        )
        c.commit()


def run_validation_gate(
    candidate_path: str | Path,
    reference_path: str | Path,
    held_out_df,
    cfg: Any,
    triggered_by_sources: list[str] | None = None,
    log_db_path: str | Path | None = None,
) -> dict:
    """3-check validation gate. Returns dict con `passed` + `checks` + `reject_reason`."""
    from drift.compute import compute_ece

    candidate_path = Path(candidate_path)
    reference_path = Path(reference_path)
    cand_meta_path = candidate_path.with_suffix(".metadata.json")
    ref_meta_path = reference_path.with_suffix(".metadata.json")
    if not cand_meta_path.exists():
        raise FileNotFoundError(f"candidate metadata mancante: {cand_meta_path}")
    if not ref_meta_path.exists():
        raise FileNotFoundError(f"reference metadata mancante: {ref_meta_path}")

    cand_meta = json.loads(cand_meta_path.read_text(encoding="utf-8"))
    ref_meta = json.loads(ref_meta_path.read_text(encoding="utf-8"))

    checks: list[dict] = []

    # CHECK 1 — ECE regression
    cand_ece = float(cand_meta.get("aggregate", {}).get("ece", 0.0))
    ref_ece = float(ref_meta.get("aggregate", {}).get("ece", 0.0))
    regression_pct = ((cand_ece - ref_ece) / ref_ece * 100.0) if ref_ece > 0 else 0.0
    ece_passed = regression_pct <= cfg.MAX_ECE_REGRESSION_PCT
    checks.append({
        "name": "ece_regression",
        "metric_value": float(regression_pct),
        "threshold": float(cfg.MAX_ECE_REGRESSION_PCT),
        "passed": bool(ece_passed),
    })

    # CHECK 2 — KS divergence candidate vs reference predictions
    try:
        import joblib
        cand_bundle = joblib.load(candidate_path)
        ref_bundle = joblib.load(reference_path)
        pred_cand = cand_bundle["model"].predict_proba(held_out_df)[:, 1]
        pred_ref = ref_bundle["model"].predict_proba(held_out_df)[:, 1]
        ks_stat, _ = ks_2samp(pred_ref, pred_cand, method="auto")
        ks_passed = ks_stat <= cfg.KS_CANDIDATE_DIVERGENCE_THRESHOLD
        checks.append({
            "name": "ks_divergence",
            "metric_value": float(ks_stat),
            "threshold": float(cfg.KS_CANDIDATE_DIVERGENCE_THRESHOLD),
            "passed": bool(ks_passed),
        })
    except Exception as exc:
        log.warning("validation_gate KS check skipped: %s", exc)
        checks.append({
            "name": "ks_divergence",
            "metric_value": 0.0,
            "threshold": float(cfg.KS_CANDIDATE_DIVERGENCE_THRESHOLD),
            "passed": True,
            "skipped": True,
            "reason": str(exc),
        })

    # CHECK 3 — Threshold-sweep expectancy regression per profile
    # Phase 7 ml/threshold.py dependency (Plan 07-04). Se Phase 7 non on-disk → skip.
    try:
        from ml.threshold import sweep_expectancy_per_profile  # Phase 7 planned
        expectancy_delta = sweep_expectancy_per_profile(
            cand_bundle, ref_bundle, held_out_df, cfg,
        )
        expectancy_passed = expectancy_delta >= -cfg.EXPECTANCY_REGRESSION_PCT
        checks.append({
            "name": "expectancy_regression",
            "metric_value": float(expectancy_delta),
            "threshold": float(-cfg.EXPECTANCY_REGRESSION_PCT),
            "passed": bool(expectancy_passed),
        })
    except (ImportError, AttributeError) as exc:
        log.info("validation_gate expectancy check skipped (Phase 7 dep mancante): %s", exc)
        checks.append({
            "name": "expectancy_regression",
            "metric_value": 0.0,
            "threshold": float(-cfg.EXPECTANCY_REGRESSION_PCT),
            "passed": True,
            "skipped": True,
        })

    all_passed = all(c["passed"] for c in checks)
    reject_reason = None if all_passed else (
        "validation_failed: " + ",".join(c["name"] for c in checks if not c["passed"])
    )

    # Audit log validation_log.db
    if log_db_path is not None:
        try:
            ensure_validation_log_db(log_db_path)
            _append_validation_rows(
                log_db_path, candidate_path, ref_meta.get("model_version"),
                checks, reject_reason, triggered_by_sources or [],
            )
        except Exception:
            log.exception("validation_gate: errore append validation_log.db")

    return {
        "passed": all_passed,
        "checks": checks,
        "reject_reason": reject_reason,
    }


def _append_validation_rows(
    db_path, candidate_path, previous_version, checks, reject_reason, triggered_by_sources,
):
    ts_utc = datetime.now(timezone.utc).isoformat()
    sources_json = json.dumps(triggered_by_sources)
    with sqlite3.connect(str(db_path)) as c:
        c.execute("PRAGMA journal_mode=WAL")
        for ck in checks:
            c.execute(
                "INSERT INTO validation_log ("
                "ts_utc, candidate_path, previous_version, check_name, metric_value, "
                "threshold, passed, reject_reason, triggered_by_sources_json"
                ") VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
                (
                    ts_utc, str(candidate_path), previous_version, ck["name"],
                    float(ck["metric_value"]), float(ck["threshold"]),
                    int(ck["passed"]), reject_reason if not ck["passed"] else None,
                    sources_json,
                ),
            )
        c.commit()
```

```python
"""retrain.promotion — Phase 9 Area C atomic candidate → final promote.

P6: os.replace (Windows MoveFileEx replace-existing + POSIX rename atomic).
P7: pair-rename order metadata.json FIRST, .pkl SECOND. Crash inter-pair →
    FeatureSchemaMismatchError on next MLFilter.load (Phase 7 D-07-05-D)
    — designed failure mode, NOT a bug.
P8: _invalidate_singleton() post promote — hot-reload MLFilter cache.
"""
from __future__ import annotations

import logging
import os
from pathlib import Path


log = logging.getLogger(__name__)


def atomic_promote(candidate_path: str | Path, final_path: str | Path) -> None:
    """os.replace atomico per pair (P6 + P7) + singleton invalidate (P8).

    Order MANDATORY:
      1. os.replace(cand_metadata, final_metadata) — smaller/faster file first
      2. os.replace(cand_pkl, final_pkl) — model bundle second

    Args:
        candidate_path: percorso candidate .pkl (sibling .metadata.json).
        final_path: percorso final .pkl (sibling .metadata.json).
    """
    candidate_path = Path(candidate_path)
    final_path = Path(final_path)
    cand_meta = candidate_path.with_suffix(".metadata.json")
    final_meta = final_path.with_suffix(".metadata.json")

    if not candidate_path.exists():
        raise FileNotFoundError(f"candidate .pkl mancante: {candidate_path}")
    if not cand_meta.exists():
        raise FileNotFoundError(f"candidate metadata mancante: {cand_meta}")

    # P6 + P7: metadata FIRST, pkl SECOND
    # P6: Windows-atomic via MoveFileEx(MOVEFILE_REPLACE_EXISTING) — NEVER use os.rename.
    os.replace(str(cand_meta), str(final_meta))
    os.replace(str(candidate_path), str(final_path))

    # P8: hot-reload MLFilter singleton (Wave 0 hook in ml/inference.py)
    try:
        from ml.inference import _invalidate_singleton
        _invalidate_singleton()
        log.info("atomic_promote: completato + singleton invalidato. final=%s", final_path)
    except ImportError:
        log.warning("atomic_promote: _invalidate_singleton non importabile (Phase 7 dep?)")
```
