```python
def merge_metadata_source(self, run_id: str, source: str) -> bool:
    """Phase 9 W8 PUBLIC API — append `source` to JobRecord.metadata.triggered_by_sources.

    Atomic vs concurrent get/submit via existing `_lock`. Returns True se merge
    avvenuto (job esistente + source non già presente), False altrimenti.
    """
    with self._lock:
        job = self._jobs.get(run_id)
        if job is None or not hasattr(job, "metadata"):
            return False
        existing = list(job.metadata.get("triggered_by_sources", []))
        if source in existing:
            return False
        existing.append(source)
        job.metadata["triggered_by_sources"] = existing
        return True


def get_active_retrain_job_id(self) -> str | None:
    """Phase 9 W8 PUBLIC API — ritorna run_id del retrain currently running/pending, o None.

    Search criterion: JobRecord con run_id startswith 'mcp_retrain_' (Phase 9 convention)
    AND status in {'running', 'pending'}. Atomic via `_lock`.
    """
    with self._lock:
        for run_id, job in self._jobs.items():
            if not run_id.startswith("mcp_retrain_"):
                continue
            status = getattr(job, "status", None)
            if status in ("running", "pending"):
                return run_id
    return None


def get_triggered_by_sources(self, run_id: str) -> list[str]:
    """Phase 9 W8 PUBLIC API — lettura sources array atomic."""
    with self._lock:
        job = self._jobs.get(run_id)
        if job is None or not hasattr(job, "metadata"):
            return []
        return list(job.metadata.get("triggered_by_sources", []))
```

```python
B2: Phase 8 worker import LAZY (inside enqueue_or_dedup) — graceful failure
    mode con RuntimeError chiaro se Phase 8 non eseguita.

D-09-C1: enqueue Phase 8 `_train_ml_filter_worker` (zero duplicazione code).
"""
from __future__ import annotations

import logging
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


log = logging.getLogger(__name__)


def _import_phase8_worker():
    """B2 lazy import — Phase 8 Plan 08-04 prereq. RuntimeError chiaro se absent."""
    try:
        from mcp_tools.handlers.ml import _train_ml_filter_worker, _validate_parquet_schema_v2
        return _train_ml_filter_worker, _validate_parquet_schema_v2
    except ImportError as e:
        raise RuntimeError(
            "Phase 8 retrain prerequisites missing: mcp_tools.handlers.ml not importable. "
            "Run Phase 8 Plan 08-04 execute before Phase 9 Wave 3."
        ) from e


def enqueue_or_dedup(
    job_queue: Any,
    source: str,
    cfg: Any,
    reason: str = "",
) -> dict:
    """3 trigger source condividono dedup. Returns envelope dict."""
    # B2 lazy import — done INSIDE function, before any submit call
    _train_ml_filter_worker, _validate_parquet_schema_v2 = _import_phase8_worker()

    parquet_path = Path(cfg.MCP_TRAINING_DATA_PATH)
    ok, msg, parquet_sha = _validate_parquet_schema_v2(parquet_path)
    if not ok:
        from mcp_tools.errors import ErrorCodes, envelope
        return envelope(ErrorCodes.VALIDATION_FAILED, msg)

    run_id = _build_retrain_run_id(source)
    metadata = {
        "source": source,
        "reason": reason,
        "parquet_sha256": parquet_sha,
        "triggered_by_sources": [source],
    }
    result = job_queue.submit(
        run_id, _train_ml_filter_worker, metadata,
        run_id, str(parquet_path), str(Path(cfg.ML_MODEL_PATH).parent),
        "data/configs/ml.yaml", parquet_sha,
    )

    if isinstance(result, dict) and result.get("error") == "run_in_progress":
        # D-09-C3 idempotent: W8 public API merge_metadata_source
        active_id = result.get("active_run_id") or job_queue.get_active_retrain_job_id()
        if active_id:
            job_queue.merge_metadata_source(active_id, source)
        return {
            "ok": True,
            "status": "already_pending",
            "job_id": active_id,
            "triggered_by_sources": job_queue.get_triggered_by_sources(active_id) if active_id else [source],
        }

    if isinstance(result, dict):
        result.setdefault("triggered_by_sources", [source])
        result.setdefault("status", "started")
    return result


def _build_retrain_run_id(source: str) -> str:
    """run_id univoco per audit: mcp_retrain_{utc_ts}_{source}."""
    ts = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
    return f"mcp_retrain_{ts}_{source}"
```

```python
def ensure_validation_log_db(db_path: str | Path) -> None:

def _check_ece_regression(cand_meta: dict, ref_meta: dict, cfg: Any) -> ValidationCheckResult:
    cand_ece = float(cand_meta.get("aggregate", {}).get("ece", 0.0))
    ref_ece = float(ref_meta.get("aggregate", {}).get("ece", 0.0))
    regression_pct = ((cand_ece - ref_ece) / ref_ece * 100.0) if ref_ece > 0 else 0.0
    passed = regression_pct <= cfg.MAX_ECE_REGRESSION_PCT
    return ValidationCheckResult(
        check_name="ece_regression",
        passed=bool(passed),
        metric_value=float(regression_pct),
        threshold=float(cfg.MAX_ECE_REGRESSION_PCT),
        reject_reason=None if passed else f"ece_regression: {regression_pct:.2f}% > {cfg.MAX_ECE_REGRESSION_PCT:.2f}%",
    )


def _check_ks_divergence(candidate_path: Path, reference_path: Path, held_out_df, cfg: Any) -> ValidationCheckResult:
    try:
        import joblib
        cand_bundle = joblib.load(candidate_path)
        ref_bundle = joblib.load(reference_path)
        pred_cand = cand_bundle["model"].predict_proba(held_out_df)[:, 1]
        pred_ref = ref_bundle["model"].predict_proba(held_out_df)[:, 1]
        ks_stat, _ = ks_2samp(pred_ref, pred_cand, method="auto")
        passed = ks_stat <= cfg.KS_CANDIDATE_DIVERGENCE_THRESHOLD
        return ValidationCheckResult(
            check_name="ks_divergence",
            passed=bool(passed),
            metric_value=float(ks_stat),
            threshold=float(cfg.KS_CANDIDATE_DIVERGENCE_THRESHOLD),
            reject_reason=None if passed else f"ks_divergence: {ks_stat:.4f} > {cfg.KS_CANDIDATE_DIVERGENCE_THRESHOLD:.4f}",
        )
    except Exception as exc:
        log.exception("KS divergence check error")
        return ValidationCheckResult(
            check_name="ks_divergence", passed=False,
            metric_value=float("nan"),
            threshold=float(cfg.KS_CANDIDATE_DIVERGENCE_THRESHOLD),
            reject_reason=f"ks_divergence_error: {exc}",

def run_validation_gate(
    candidate_path: str | Path,
    reference_path: str | Path,
    held_out_df,
    cfg: Any,
    triggered_by_sources: list[str] | None = None,
    log_db_path: str | Path | None = None,
) -> dict:
    """3-check HARD-FAIL validation gate. Returns dict con `passed`+`checks`+`reject_reason`."""
    candidate_path = Path(candidate_path)
    reference_path = Path(reference_path)
    cand_meta_path = candidate_path.with_suffix(".metadata.json")
    ref_meta_path = reference_path.with_suffix(".metadata.json")
    if not cand_meta_path.exists():
        raise FileNotFoundError(f"candidate metadata mancante: {cand_meta_path}")
    if not ref_meta_path.exists():
        raise FileNotFoundError(f"reference metadata mancante: {ref_meta_path}")

    cand_meta = json.loads(cand_meta_path.read_text(encoding="utf-8"))
    ref_meta = json.loads(ref_meta_path.read_text(encoding="utf-8"))

    checks: list[ValidationCheckResult] = [
        _check_ece_regression(cand_meta, ref_meta, cfg),
        _check_ks_divergence(candidate_path, reference_path, held_out_df, cfg),
        _check_expectancy_regression(candidate_path, reference_path, held_out_df, cfg),
    ]

    all_passed = all(c.passed for c in checks)
    reject_reason = None if all_passed else (
        "validation_failed: " + "; ".join(c.reject_reason for c in checks if not c.passed and c.reject_reason)
    )

    if log_db_path is not None:
        try:
            ensure_validation_log_db(log_db_path)
            _append_validation_rows(
                log_db_path, candidate_path, ref_meta.get("model_version"),
                checks, reject_reason, triggered_by_sources or [],
            )
        except Exception:
            log.exception("validation_gate: errore append validation_log.db")

    return {
        "passed": all_passed,
        "checks": [
            {
                "name": c.check_name,
                "metric_value": c.metric_value,
                "threshold": c.threshold,
                "passed": c.passed,
                "reject_reason": c.reject_reason,
            }
            for c in checks
        ],
        "reject_reason": reject_reason,
    }


def _append_validation_rows(db_path, candidate_path, previous_version, checks, reject_reason, triggered_by_sources):
    ts_utc = datetime.now(timezone.utc).isoformat()
    sources_json = json.dumps(triggered_by_sources)
    with sqlite3.connect(str(db_path)) as c:
        c.execute("PRAGMA journal_mode=WAL")
        for ck in checks:
            c.execute(
                "INSERT INTO validation_log ("
                "ts_utc, candidate_path, previous_version, check_name, metric_value, "
                "threshold, passed, reject_reason, triggered_by_sources_json"
                ") VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
                (
                    ts_utc, str(candidate_path), previous_version, ck.check_name,
                    float(ck.metric_value), float(ck.threshold),
                    int(ck.passed), ck.reject_reason,
                    sources_json,
                ),
            )
        c.commit()
```
