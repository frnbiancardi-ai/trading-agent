"""cluster.preprocessing — Phase 9 Area A D-09-A4 batch wrapper su Phase 7 build_feature_vector."""
from __future__ import annotations

import json

import numpy as np
import pandas as pd

from ml.feature_extraction import build_feature_vector


def build_feature_matrix(
    losing_trades_df: pd.DataFrame,
    feature_list: list[str],
    cat_encodings: dict[str, dict],
) -> tuple[np.ndarray, list[int]]:
    """Costruisce matrice (N×D) iterando build_feature_vector Phase 7 — D-09-A4.

    Args:
        losing_trades_df: subset di baseline_decisions filtrato y=0 (exit_reason != 'TP_HIT' per D-09-A2).
        feature_list: lista feature da metadata.json bundle Phase 7 D-15.
        cat_encodings: encoder categorical da metadata.json bundle Phase 7 D-10.
    Returns:
        (X: ndarray (N, D) float32, trade_ids aligned list[int]).
    """
    rows: list[np.ndarray] = []
    trade_ids: list[int] = []
    for _, row in losing_trades_df.iterrows():
        draft, indicators, ctx = _row_to_phase7_inputs(row)
        fv = build_feature_vector(draft, indicators, ctx, feature_list, cat_encodings)
        rows.append(fv.values[0])
        trade_ids.append(int(row["trade_id"]))
    return np.asarray(rows, dtype=np.float32), trade_ids


def _row_to_phase7_inputs(row: pd.Series) -> tuple[dict, dict, dict]:
    """Decodifica row parquet schema-v2 in (draft, indicators, ctx) Phase 7-compatible."""
    decision_ctx = json.loads(row.get("decision_context_json") or "{}")
    draft = decision_ctx.get("draft", {})
    indicators = decision_ctx.get("indicators", {})
    ctx = decision_ctx.get("ctx", {
        "symbol": row.get("symbol"),
        "timeframe": row.get("timeframe"),
        "profile": row.get("profile"),
        "regime": row.get("regime_state"),
    })
    return draft, indicators, ctx

```python
"""cluster.hdbscan_runner — Phase 9 Area A D-09-A3 HDBSCAN failure-mode clustering.

P1: prediction_data=True at fit time (post-restart approximate_predict).
P2: import `from hdbscan import HDBSCAN`, NOT `from sklearn.cluster import HDBSCAN`.
P3: contrib package min_samples semantics differ from sklearn (1 sample).
"""
from __future__ import annotations

import logging
from typing import Any

import numpy as np
import pandas as pd

from hdbscan import HDBSCAN  # P2 — NOT sklearn.cluster.HDBSCAN
from sklearn.metrics import davies_bouldin_score, silhouette_score


log = logging.getLogger(__name__)


def fit_clusters(X: np.ndarray, cfg: Any) -> HDBSCAN:
    """Fit HDBSCAN su feature matrix con P1 + determinism contract."""
    clusterer = HDBSCAN(
        min_cluster_size=cfg.HDBSCAN_MIN_CLUSTER_SIZE,
        min_samples=cfg.HDBSCAN_MIN_SAMPLES,
        cluster_selection_method=cfg.HDBSCAN_CLUSTER_SELECTION_METHOD,
        metric="euclidean",
        prediction_data=True,
        core_dist_n_jobs=1,
    )
    clusterer.fit(X)
    return clusterer


def cluster_quality(X: np.ndarray, labels: np.ndarray) -> dict:
    """Validazione qualità cluster: Davies-Bouldin + Silhouette.

    Quality enum:
      - "good": DB < 1.5 AND silhouette > 0.3
      - "acceptable": DB < 2.0 AND silhouette > 0.15
      - "poor": altrimenti (warning + fallback PCA recommendation Wave 2)
      - "insufficient": meno di 2 cluster non-noise OR meno di 10 sample non-noise
    """
    mask = labels != -1
    if mask.sum() < 10 or len(set(labels[mask])) < 2:
        return {"davies_bouldin": float("nan"), "silhouette": float("nan"), "quality": "insufficient"}
    db = davies_bouldin_score(X[mask], labels[mask])
    sil = silhouette_score(X[mask], labels[mask])
    if db < 1.5 and sil > 0.3:
        quality = "good"
    elif db < 2.0 and sil > 0.15:
        quality = "acceptable"
    else:
        quality = "poor"
        log.warning("cluster_quality poor: DB=%.3f sil=%.3f. Recommend PCA fallback Wave 2.", db, sil)
    return {"davies_bouldin": float(db), "silhouette": float(sil), "quality": quality}


def summarize_clusters(
    clusterer: HDBSCAN,
    trade_ids: list[int],
    losing_trades_df: pd.DataFrame,
    top_n_exemplars: int = 3,
) -> list[dict]:
    """Per ogni cluster ≠ -1: exemplar IDs + feature_distribution_summary + summary_it.

    Output shape MCP-07 response (CONTEXT.md § specifics lines 348-379).
    """
    out: list[dict] = []
    labels = clusterer.labels_
    probs = clusterer.probabilities_

    for cid in sorted(set(labels) - {-1}):
        mask = labels == cid
        cluster_ids = [trade_ids[i] for i in np.where(mask)[0]]
        cluster_probs = probs[mask]
        order = np.argsort(-cluster_probs)[:top_n_exemplars]
        exemplars = [cluster_ids[i] for i in order]
        sub = losing_trades_df.loc[losing_trades_df["trade_id"].isin(cluster_ids)]
        out.append({
            "cluster_id": int(cid),
            "n_trades": int(mask.sum()),
            "exemplar_trade_ids": exemplars,
            "feature_distribution_summary": _summarize_features(sub),
            "avg_pnl_pips": float(sub["pnl_pips"].mean()) if "pnl_pips" in sub.columns else 0.0,
            "hit_rate_within_cluster": float((sub["exit_reason"] == "TP_HIT").mean()) if "exit_reason" in sub.columns else 0.0,
            "summary_it": _summary_it(sub),
        })
    return out


def _summarize_features(sub: pd.DataFrame) -> dict:
    """Median + IQR per numeric, mode + freq per categorical."""
    summary: dict = {}
    NUMERIC_FEATURES = ["atr", "adx", "rsi", "ema_distance", "spread_pips", "r_to_r"]
    CATEGORICAL_FEATURES = ["regime_state", "setup_name", "symbol", "timeframe", "profile"]
    for col in NUMERIC_FEATURES:
        if col not in sub.columns:
            continue
        vals = sub[col].dropna()
        if len(vals) == 0:
            continue
        q1, q3 = float(vals.quantile(0.25)), float(vals.quantile(0.75))
        summary[col] = {"median": float(vals.median()), "iqr": [q1, q3]}
    for col in CATEGORICAL_FEATURES:
        if col not in sub.columns:
            continue
        vc = sub[col].value_counts(normalize=True)
        if len(vc) == 0:
            continue
        key = "regime" if col == "regime_state" else col
        summary[key] = {"mode": str(vc.index[0]), "freq": float(vc.iloc[0])}
    return summary


def _summary_it(sub: pd.DataFrame) -> str:
    """Italian operator-readable cluster summary (P15 + CLAUDE.md italiano)."""
    setup_mode = sub["setup_name"].mode().iloc[0] if "setup_name" in sub.columns and len(sub["setup_name"].mode()) else "?"
    regime_mode = sub["regime_state"].mode().iloc[0] if "regime_state" in sub.columns and len(sub["regime_state"].mode()) else "?"
    sym_mode = sub["symbol"].mode().iloc[0] if "symbol" in sub.columns and len(sub["symbol"].mode()) else "?"
    hit_rate = float((sub["exit_reason"] == "TP_HIT").mean()) if "exit_reason" in sub.columns else 0.0
    return f"Cluster setup={setup_mode} regime={regime_mode} sym={sym_mode} hit_rate={hit_rate*100:.1f}%"
```

"""cluster.artifact — Phase 9 Area A joblib bundle + sidecar.json persistence."""
from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import joblib


def dump_cluster_bundle(
    clusterer: Any,
    trade_ids: list[int],
    summaries: list[dict],
    dataset_hash: str,
    quality_metrics: dict,
    out_path: str | Path,
    model_version: str = "cluster_v1",
) -> None:
    """Persiste clusterer + sidecar metadata.json + dataset_hash audit anchor."""
    out_path = Path(out_path)
    out_path.parent.mkdir(parents=True, exist_ok=True)

    bundle = {"clusterer": clusterer, "trade_ids": trade_ids, "summaries": summaries}
    joblib.dump(bundle, out_path)

    import numpy as np
    n_noise = int((np.asarray(clusterer.labels_) == -1).sum())
    meta = {
        "model_version": model_version,
        "computed_at": datetime.now(timezone.utc).isoformat(),
        "dataset_hash": dataset_hash,
        "n_losing_trades_total": len(trade_ids),
        "n_clusters": len(summaries),
        "n_noise": n_noise,
        "quality_metrics": quality_metrics,
    }
    meta_path = out_path.with_suffix(".metadata.json")
    meta_path.write_text(json.dumps(meta, indent=2), encoding="utf-8")


def load_cluster_bundle(in_path: str | Path) -> tuple[dict, dict]:
    """Carica bundle joblib + metadata.json sidecar.

    Returns:
        (bundle_dict, metadata_dict). bundle_dict["clusterer"].prediction_data_
        deve essere not-None (P1).
    """
    in_path = Path(in_path)
    if not in_path.exists():
        raise FileNotFoundError(f"cluster bundle non trovato: {in_path}")
    bundle = joblib.load(in_path)
    meta_path = in_path.with_suffix(".metadata.json")
    if not meta_path.exists():
        raise FileNotFoundError(f"metadata sidecar non trovato: {meta_path}")
    meta = json.loads(meta_path.read_text(encoding="utf-8"))
    return bundle, meta
```

"""mcp_tools.handlers.cluster — MCP-07 get_failure_clusters (D-09-A read-only handler)."""
from __future__ import annotations

import logging
from pathlib import Path
from typing import Any

from mcp.types import Tool

from mcp_tools.errors import ErrorCodes, envelope


log = logging.getLogger(__name__)


GET_FAILURE_CLUSTERS_TOOL = Tool(
    name="get_failure_clusters",
    description=(
        "Top-N failure-mode cluster (HDBSCAN su losing trades baseline). "
        "Per ogni cluster: exemplar trade IDs, feature-distribution summary "
        "(median+IQR per numeric, mode+freq per categorical), avg_pnl_pips, "
        "hit_rate intra-cluster, summary_it italiano. Quality metrics "
        "(Davies-Bouldin index, silhouette, quality enum)."
    ),
    inputSchema={
        "type": "object",
        "properties": {"top_n": {"type": "integer", "minimum": 1, "maximum": 20, "default": 5}},
        "required": [],
    },
)


def handle_get_failure_clusters(args: dict, cfg: Any) -> dict:
    """MCP-07 read-only — carica cluster artifact + top_n summary."""
    try:
        top_n = int(args.get("top_n", 5))
    except (TypeError, ValueError):
        return envelope(ErrorCodes.VALIDATION_FAILED, "top_n deve essere int")
    if top_n < 1 or top_n > 20:
        return envelope(ErrorCodes.VALIDATION_FAILED, f"top_n fuori range [1,20]: {top_n}")

    bundle_path = Path(cfg.CLUSTER_ARTIFACT_PATH)
    if not bundle_path.exists():
        return envelope(
            ErrorCodes.NOT_FOUND,
            f"cluster artifact mancante: {bundle_path}",
        )

    from cluster.artifact import load_cluster_bundle
    try:
        bundle, meta = load_cluster_bundle(bundle_path)
    except (FileNotFoundError, OSError) as exc:
        return envelope(ErrorCodes.NOT_FOUND, f"errore lettura artifact: {exc}")
    except Exception as exc:
        log.exception("get_failure_clusters: errore inatteso")
        return envelope(ErrorCodes.INTERNAL_ERROR, str(exc))

    summaries = bundle.get("summaries", [])
    top_summaries = summaries[:top_n]

    import numpy as np
    labels = bundle["clusterer"].labels_
    trade_ids = bundle["trade_ids"]
    noise_trade_ids = [trade_ids[i] for i in np.where(np.asarray(labels) == -1)[0]]

    return {
        "model_version": meta.get("model_version"),
        "computed_at": meta.get("computed_at"),
        "n_losing_trades_total": meta.get("n_losing_trades_total"),
        "n_noise_points": meta.get("n_noise"),
        "clusters": top_summaries,
        "noise_trade_ids": noise_trade_ids,
        "quality_metrics": meta.get("quality_metrics"),
    }
```
