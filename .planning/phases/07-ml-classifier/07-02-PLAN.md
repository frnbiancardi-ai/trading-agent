```python
def build_ml_folds(
    df: pd.DataFrame,
    n_folds: int,
    embargo_bars_by_tf: dict[str, int],
) -> list[tuple[np.ndarray, np.ndarray, np.ndarray]]:
    df_sorted = df.sort_values("decision_ts_utc").reset_index(drop=True)
    total = len(df_sorted)
    slot = total // (n_folds + 1)
    embargo_rows = max(embargo_bars_by_tf.values())  # D-07 upper-bound conservativo

    folds = []
    for k in range(n_folds):
        train_end = slot + k * slot
        test_start = train_end + embargo_rows
        test_end = min(test_start + slot, total)
        if test_end <= test_start:
            continue
        val_start = int(train_end * 0.80)  # D-08
        folds.append((
            np.arange(0, val_start),
            np.arange(val_start, train_end),
            np.arange(test_start, test_end),
        ))
    return folds
```

```python
# --- Temporal order invariants -------------------------------------------


def test_ml_folds_temporal_order_on_fixture(fixture_df):
    """Per ogni fold: max(train) < max(val) < min(test). Fixture 200 rows (post-iter-2 _N_ROWS bump)."""
    folds = build_ml_folds(fixture_df, n_folds=5, embargo_bars_by_tf={"M15": 10, "M30": 10, "H1": 10})

def test_ml_folds_embargo_respected_on_fixture(fixture_df):

def test_ml_folds_no_overlap_between_folds(fixture_df):

# --- AST guard: train_test_split(shuffle=True) forbidden -----------------


def test_no_shuffle_split_ast_guard():
    """ML-03 SC#2: `train_test_split(shuffle=True)` NON deve apparire in ml/*.py.

    AST walk: cerca ast.Call dove func.attr == 'train_test_split'; se trova
    keyword 'shuffle' con value True → fail. Se keyword 'shuffle' assente,
    sklearn default e' True → fail anche in quel caso (esplicito > implicit).
    """
    violations: list[str] = []
    for mod_path in Path("ml").glob("*.py"):
        tree = ast.parse(mod_path.read_text(encoding="utf-8"))
        for node in ast.walk(tree):
            if not isinstance(node, ast.Call):
                continue
            func = node.func
            attr = getattr(func, "attr", None) or getattr(func, "id", None)
            if attr != "train_test_split":
                continue
            shuffle_kw = next((kw for kw in node.keywords if kw.arg == "shuffle"), None)
            if shuffle_kw is None:
                violations.append(f"{mod_path}:{node.lineno}: train_test_split() senza shuffle=False esplicito (default True vietato)")
                continue
            try:
                val = ast.literal_eval(shuffle_kw.value)
            except (ValueError, SyntaxError):
                violations.append(f"{mod_path}:{node.lineno}: train_test_split(shuffle=<non-literal>) — usare costanti")
                continue
            if val is not False:
                violations.append(f"{mod_path}:{node.lineno}: train_test_split(shuffle={val}) — forbidden, deve essere False")
    assert not violations, "ml/ contiene shuffle violations:\n  " + "\n  ".join(violations)


# --- Sort-before-slice invariant -----------------------------------------


def test_ml_folds_dataframe_is_sorted_before_slicing(fixture_df):
    """Anche su df shuffled, build_ml_folds produce fold con timestamps crescenti."""
    shuffled = fixture_df.sample(frac=1, random_state=7).reset_index(drop=True)
    folds = build_ml_folds(shuffled, n_folds=5, embargo_bars_by_tf={"M15": 10, "M30": 10, "H1": 10})
    # Estrai timestamp colonna ordinata; assert che gli indici train precedono
    # temporalmente val che precede test (l'invariante e' VERA solo se build_ml_folds
    # ha ordinato internamente).
    sorted_df = shuffled.sort_values("decision_ts_utc").reset_index(drop=True)
    for k, (train_idx, val_idx, test_idx) in enumerate(folds):
        if len(val_idx) == 0 or len(test_idx) == 0:
            continue
        train_max_ts = sorted_df.iloc[max(train_idx)]["decision_ts_utc"]
        val_min_ts = sorted_df.iloc[min(val_idx)]["decision_ts_utc"]
        test_min_ts = sorted_df.iloc[min(test_idx)]["decision_ts_utc"]
        assert train_max_ts <= val_min_ts, f"fold {k}: train ts > val ts (post-sort)"
        assert val_min_ts <= test_min_ts, f"fold {k}: val ts > test ts (post-sort)"
```

```python
"""Walk-forward fold builder Phase 7 (ML-03, D-05/06/07/08).

Estende `backtest.walk_forward.walk_forward_slices` (Phase 1 BACK-05, expanding
mode) con:
  - operazione su DataFrame ordinato per decision_ts_utc (vs sequenza bar Phase 1)
  - embargo per-TF uniforme = max(embargo_bars_by_tf) (D-07 upper-bound
    conservativo; per-row embargo riservato Phase 9)
  - train/val split temporale last-20% del train fold (D-08, no shuffle)
  - returns list of (train_idx, val_idx, test_idx) — 3-tupla vs 2-tupla Phase 1

D-17 no-future-leakage by construction:
  - df.sort_values("decision_ts_utc") rimuove ogni accidental shuffling caller-side
  - embargo_rows separa train_end da test_start
  - val_start interno al train range (caso degenere: tutto coperto)

Plan 07-02 D-07-02-A: embargo uniforme = 120 righe (M15+M30+H1 max). Per-row
embargo basato su row.timeframe e' refinement Phase 9 (PATTERNS.md §Notes-for-planner #7).

Modulo puro: nessun broker, nessun logging, nessun datetime.now(). AST purity
gate (tests/test_ml_purity.py) verifica invariante.
"""
from __future__ import annotations

import numpy as np
import pandas as pd

_FOLD_CAP = 10  # D-06 cap PROJECT.md (mirror backtest/walk_forward.py:11)


def build_ml_folds(
    df: pd.DataFrame,
    n_folds: int,
    embargo_bars_by_tf: dict[str, int],
) -> list[tuple[np.ndarray, np.ndarray, np.ndarray]]:
    """Expanding-window walk-forward folds con embargo + val 20% del train.

    Args:
        df: DataFrame con colonna `decision_ts_utc` (sortable ISO 8601).
        n_folds: numero di fold, clampato a 1..10 (_FOLD_CAP D-06).
        embargo_bars_by_tf: mappa TF -> bar embargo (e.g., {"M15": 96, "H1": 120}).
                            Uniform max() usato come embargo conservativo D-07-02-A.

    Returns:
        Lista di (train_idx, val_idx, test_idx). Ogni elemento e' np.ndarray
        di indici posizionali nel DataFrame ordinato. Fold senza test slot
        sufficiente (test_end <= test_start) sono OMESSI dalla lista.

    Raises:
        ValueError: n_folds fuori [1, 10] o df senza colonna decision_ts_utc.
    """
    if n_folds < 1 or n_folds > _FOLD_CAP:
        raise ValueError(f"n_folds deve essere 1..{_FOLD_CAP}, got {n_folds}")
    if "decision_ts_utc" not in df.columns:
        raise ValueError("df deve avere colonna 'decision_ts_utc'")
    if not embargo_bars_by_tf:
        raise ValueError("embargo_bars_by_tf non puo' essere vuoto")

    df_sorted = df.sort_values("decision_ts_utc").reset_index(drop=True)
    total = len(df_sorted)
    if total < n_folds + 1:
        # Dataset troppo piccolo per n_folds — ritorna lista vuota safely
        return []

    slot = total // (n_folds + 1)
    embargo_rows = max(embargo_bars_by_tf.values())  # D-07 upper-bound

    folds: list[tuple[np.ndarray, np.ndarray, np.ndarray]] = []
    for k in range(n_folds):
        train_end = slot + k * slot
        test_start = train_end + embargo_rows
        test_end = min(test_start + slot, total)
        if test_end <= test_start:
            # Fold non producibile (test slot 0 o negativo dopo embargo) — skip
            continue
        # D-08: ultimi 20% del train fold come val (no shuffle)
        val_start = int(train_end * 0.80)
        train_idx = np.arange(0, val_start, dtype=np.int64)
        val_idx = np.arange(val_start, train_end, dtype=np.int64)
        test_idx = np.arange(test_start, test_end, dtype=np.int64)
        folds.append((train_idx, val_idx, test_idx))

    return folds


__all__ = ["build_ml_folds"]
```
