```python
    # Populated at bootstrap by load_ml_thresholds_from_metadata
    ML_THRESHOLD_BY_PROFILE: dict[str, float] = {}


def load_ml_thresholds_from_metadata(cfg) -> None:
    """Bootstrap loader: legge metadata.json e popola cfg.ML_THRESHOLD_BY_PROFILE.

    Idempotent. Fallback {0.5, 0.5, 0.5} se metadata.json mancante e
    ENABLE_ML_FILTER=true (warning).
    """
    if not cfg.ML_MODEL_PATH:
        return
    from pathlib import Path
    import json
    meta_path = Path(cfg.ML_MODEL_PATH).with_suffix(".metadata.json")
    if not meta_path.exists():
        if cfg.ENABLE_ML_FILTER:
            # Logger top-level (config.py uses logging already? verify; if not, print)
            import logging
            logging.getLogger(__name__).warning(
                "ML_MODEL_PATH set but metadata.json missing: %s", meta_path,
            )
        cfg.ML_THRESHOLD_BY_PROFILE = {"CONSERVATIVE": 0.5, "MODERATE": 0.5, "AGGRESSIVE": 0.5}
        return
    metadata = json.loads(meta_path.read_text(encoding="utf-8"))
    cfg.ML_THRESHOLD_BY_PROFILE = dict(metadata.get("threshold_by_profile", {}))
```

```python
def init_ml_inference_logger(cfg) -> logging.Logger:
    """Inference telemetry logger (D-07-06-G).

    Output: logs/ml_inference.log, RotatingFileHandler, weekly rotation,
    8 backups (Phase 9 drift monitor consumer).
    """
    import logging
    from logging.handlers import RotatingFileHandler
    from pathlib import Path

    log = logging.getLogger("ml_inference")
    if log.handlers:
        return log  # idempotent
    log_path = Path(getattr(cfg, "LOG_FILE", "logs/agent.log")).parent / "ml_inference.log"
    log_path.parent.mkdir(parents=True, exist_ok=True)
    handler = RotatingFileHandler(log_path, maxBytes=5_000_000, backupCount=8)
    handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
    log.addHandler(handler)
    log.setLevel(logging.INFO)
    return log
```

```python
def load_ml_thresholds_from_metadata(cfg) -> None:
    """Bootstrap loader: legge metadata.json e popola cfg.ML_THRESHOLD_BY_PROFILE."""
    if not cfg.ML_MODEL_PATH:
        return
    from pathlib import Path
    import json
    meta_path = Path(cfg.ML_MODEL_PATH).with_suffix(".metadata.json")
    fallback = {"CONSERVATIVE": 0.5, "MODERATE": 0.5, "AGGRESSIVE": 0.5}
    if not meta_path.exists():
        if cfg.ENABLE_ML_FILTER:
            import logging
            logging.getLogger(__name__).warning(
                "ML_MODEL_PATH=%s ma metadata.json mancante: fallback {0.5,0.5,0.5}",
                cfg.ML_MODEL_PATH,
            )
        cfg.ML_THRESHOLD_BY_PROFILE = fallback
        return
    try:
        metadata = json.loads(meta_path.read_text(encoding="utf-8"))
        thresholds = dict(metadata.get("threshold_by_profile", {}))
        cfg.ML_THRESHOLD_BY_PROFILE = {**fallback, **thresholds}
    except (json.JSONDecodeError, OSError) as exc:
        import logging
        logging.getLogger(__name__).warning("metadata.json parse fail: %s — fallback", exc)
        cfg.ML_THRESHOLD_BY_PROFILE = fallback
```

```python
def init_ml_inference_logger(cfg) -> logging.Logger:
    """Logger dedicato per ML inference telemetry (D-07-06-G).

    Output: logs/ml_inference.log, RotatingFileHandler weekly, 8 backup.
    Phase 9 drift monitor consumer.
    """
    import logging
    from logging.handlers import RotatingFileHandler
    from pathlib import Path

    log = logging.getLogger("ml_inference")
    if log.handlers:
        return log  # idempotent
    log_path = Path(getattr(cfg, "LOG_FILE", "logs/agent.log")).parent / "ml_inference.log"
    log_path.parent.mkdir(parents=True, exist_ok=True)
    handler = RotatingFileHandler(log_path, maxBytes=5_000_000, backupCount=8)
    handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
    log.addHandler(handler)
    log.setLevel(logging.INFO)
    log.propagate = False  # no double-log to root
    return log
```

```python
@pytest.mark.integration
def test_held_out_e2e_rejection_rate(real_train_artifacts):
    """SC#5: rejection rate per profile ∈ [0.20, 0.80] (sanity range)."""
    last_fold_path, held_out, _ = real_train_artifacts
    flt = MLFilter.load(last_fold_path)
    thresholds = flt.threshold_by_profile

    per_profile_counts: dict[str, dict[str, int]] = {
        "CONSERVATIVE": {"approve": 0, "reject": 0},
        "MODERATE":     {"approve": 0, "reject": 0},
        "AGGRESSIVE":   {"approve": 0, "reject": 0},
    }

    # Rev iter1: categorical mapping coverage guard. Tracciamo per ogni categorical feature
    # quante righe held-out hanno valore mappato vs -1. Soglia 70%: sotto, il held-out segment
    # e' out-of-distribution per quella categorica → test rejection-rate non e' informativo.
    cat_features_in_held_out = [f for f in flt.features if f in flt.categorical_encodings]
    cat_mapped_count: dict[str, int] = {f: 0 for f in cat_features_in_held_out}
    cat_total_count: dict[str, int] = {f: 0 for f in cat_features_in_held_out}

    for _, row in held_out.iterrows():
        profile = row.get("profile")
        if profile not in per_profile_counts:
            continue
        # Build single-row DataFrame with feature_list order from bundle
        sample = {}
        for f in flt.features:
            val = row.get(f)
            if f in flt.categorical_encodings:
                cat_total_count[f] += 1
                mapped = flt.categorical_encodings[f].get(val, -1)
                if mapped != -1:
                    cat_mapped_count[f] += 1
                val = mapped
            sample[f] = val
        X = pd.DataFrame([sample])
        try:
            _raw, calibrated = flt.predict(X)
        except Exception:
            continue
        threshold = thresholds.get(profile, 0.5)
        if calibrated < threshold:
            per_profile_counts[profile]["reject"] += 1
        else:
            per_profile_counts[profile]["approve"] += 1

    # Rev iter1: categorical mapping coverage — assert that >=70% of held-out values per cat feature
    # sono mappati (non -1). Sotto soglia → held-out e' out-of-distribution e il test e' rumore.
    for f in cat_features_in_held_out:
        if cat_total_count[f] == 0:
            continue  # feature non popolata nel held-out
        coverage = cat_mapped_count[f] / cat_total_count[f]
        assert coverage >= 0.70, (
            f"categorical '{f}' mapping coverage {coverage:.2%} < 70% — held-out potrebbe essere "
            f"out-of-distribution ({cat_mapped_count[f]}/{cat_total_count[f]} valori mappati)"
        )

    # Rev iter1: per-profile count guard — se un profile ha <=5 decisioni totali nel held-out, il test
    # diventa vacuamente passing/failing (n=3 → reject_rate ∈ {0%, 33%, 66%, 100%}). Assert min count
    # PRIMA del rate check.
    for profile in ("CONSERVATIVE", "MODERATE", "AGGRESSIVE"):
        total = per_profile_counts[profile]["approve"] + per_profile_counts[profile]["reject"]
        assert total > 5, (
            f"profile {profile} ha solo {total} decisioni held-out → test rejection-rate vacuo. "
            f"Indica probabile drift train/held-out (profile assente in last n_test_last righe)."
        )

    # Tightening rev iter1: 0.20 <= reject_rate <= 0.80 (era 0.10..0.90 vacuamente largo).
    # Range allineato con manual section di 07-VALIDATION (tipico 30-60%, sanity 20-80%).
    for profile, counts in per_profile_counts.items():
        total = counts["approve"] + counts["reject"]
        if total == 0:
            continue  # gia' coperto da count guard sopra ma defensive
        reject_rate = counts["reject"] / total
        assert 0.20 <= reject_rate <= 0.80, (
            f"{profile}: reject_rate={reject_rate:.2%} (counts={counts}, threshold={thresholds.get(profile)}) "
            f"fuori range sanity [0.20, 0.80]"
        )
        print(f"  {profile}: {counts['approve']} approve / {counts['reject']} reject = {reject_rate:.1%} reject_rate")
```
