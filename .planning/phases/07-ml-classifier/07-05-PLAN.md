```python
from __future__ import annotations
from pathlib import Path
from threading import Lock
from typing import Any

import numpy as np
import pandas as pd

from ml.artifact import load_bundle  # ALL file I/O delegated here (D-07-05-B)

_MODEL_CACHE: "MLFilter | None" = None
_CACHE_LOCK = Lock()


def get_ml_filter(model_path: str | Path) -> "MLFilter":
    global _MODEL_CACHE
    if _MODEL_CACHE is None:
        with _CACHE_LOCK:
            if _MODEL_CACHE is None:
                _MODEL_CACHE = MLFilter.load(model_path)
    return _MODEL_CACHE


class MLFilter:
    def __init__(self, bundle, metadata):
        self.model = bundle["model"]
        self.platt = bundle.get("platt")  # may be None if winner=isotonic
        self.isotonic = bundle.get("isotonic")
        self.calibrator_winner = bundle.get("calibrator_winner")  # "sigmoid" | "isotonic"
        self.features: list[str] = bundle["features"]
        self.categorical_features: list[str] = bundle["categorical_features"]
        self.categorical_encodings: dict = bundle["categorical_encodings"]
        self.threshold_by_profile: dict[str, float] = bundle["threshold_by_profile"]
        self.version: str = bundle["version"]
        self.metadata = metadata

    @classmethod
    def load(cls, path) -> "MLFilter":
        bundle, metadata = load_bundle(path)  # delegates to ml.artifact
        if metadata["n_features"] != len(bundle["features"]):
            raise FeatureSchemaMismatchError(
                f"n_features mismatch: meta={metadata['n_features']} bundle={len(bundle['features'])}"
            )
        return cls(bundle, metadata)

    def predict(self, X: pd.DataFrame) -> tuple[float, float]:
        # Plan 07-05 D-07-05-G: DataFrame input, NOT np.ndarray
        raw_arr = self.model.predict(X, raw_score=True)  # LightGBM raw logit (pre-calibration)
        raw = float(raw_arr[0]) if hasattr(raw_arr, "__len__") else float(raw_arr)
        if self.calibrator_winner == "sigmoid":
            calibrated = float(self.platt.predict_proba(np.array([[raw]]))[0, 1])
        else:
            calibrated = float(self.isotonic.transform(np.array([raw]))[0])
        if not (0.0 <= calibrated <= 1.0):
            raise ValueError(f"calibrated_prob out of range [0,1]: {calibrated}")
        return raw, calibrated


class FeatureSchemaMismatchError(Exception):
    """Inference receives feature schema diverso da training (RESEARCH §Pitfall 6)."""
```

NB: in PATTERNS.md line 535 the code uses `self.model.predict(X, raw_score=True)` — this is the LightGBM Booster API, but `LGBMClassifier` (sklearn wrapper) doesn't have `raw_score=True` directly. We can get raw via `model.predict_proba(X)[:, 1]` (probability) OR access `model.booster_.predict(X, raw_score=True)` (booster method). For Phase 7, we use `predict_proba[:,1]` as the "raw" — pre-calibration prob — and call THAT raw_score for API consistency. The calibrator then transforms it. This matches Plan 07-03 `_train_one_fold` raw_test computation.

```python
# ml/artifact.py — joblib bundle + sidecar json
from __future__ import annotations
import hashlib
import json
import subprocess
from pathlib import Path

import joblib
import sklearn  # for __version__
import lightgbm as lgb


def dump_bundle(bundle: dict, metadata: dict, out_path: Path) -> Path:
    out_path.parent.mkdir(parents=True, exist_ok=True)
    joblib.dump(bundle, out_path, compress=3)
    meta_path = out_path.with_suffix(".metadata.json")
    meta_path.write_text(json.dumps(metadata, indent=2), encoding="utf-8")
    return out_path


def load_bundle(path: Path | str) -> tuple[dict, dict]:
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"Bundle missing: {path}")
    meta_path = path.with_suffix(".metadata.json")
    if not meta_path.exists():
        raise FileNotFoundError(f"Metadata sidecar missing: {meta_path}")
    bundle = joblib.load(path)
    metadata = json.loads(meta_path.read_text(encoding="utf-8"))
    return bundle, metadata


def _git_sha() -> str:
    try:
        return subprocess.check_output(
            ["git", "rev-parse", "HEAD"], stderr=subprocess.DEVNULL,
        ).decode().strip()
    except Exception:
        return "unknown"


def _file_sha256(path: Path) -> str:
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()
```

```python
        self.threshold_by_profile: dict[str, float] = bundle["threshold_by_profile"]
        self.version: str = bundle["version"]
        self.metadata: dict = metadata

    @classmethod
    def load(cls, path: str | Path) -> "MLFilter":
        """Carica bundle + metadata; valida schema (D-07-05-D)."""
        bundle, metadata = load_bundle(path)
        n_meta = int(metadata.get("n_features", -1))
        n_bundle = len(bundle["features"])
        if n_meta != n_bundle:
            raise FeatureSchemaMismatchError(
                f"n_features mismatch: meta={n_meta} bundle={n_bundle}"
            )
        return cls(bundle, metadata)

    def predict(self, X: pd.DataFrame) -> tuple[float, float]:
        """Single-sample predict: raw_score (pre-calibration) + calibrated_prob ∈ [0,1].

        Args:
            X: pd.DataFrame single-row con feature_list nell'ordine training
               (RESEARCH §Code Examples lines 686-692 — DataFrame preserva feature names).

        Returns:
            (raw, calibrated): raw e' la probabilita' grezza pre-calibrazione (LightGBM
            predict_proba[:, 1]); calibrated e' la probabilita' calibrata (Platt o
            Isotonic vincitore). Entrambi sono Python float per JSON-friendly downstream.

        Raises:
            ValueError: calibrated_prob fuori [0, 1] (sanity check finale, V5 ASVS).
        """
        proba_arr = self.model.predict_proba(X)
        raw = float(proba_arr[0, 1])
        if self.calibrator_winner == "sigmoid":
            calibrated = float(self.platt.predict_proba(np.array([[raw]]))[0, 1])
        else:
            calibrated = float(self.isotonic.transform(np.array([raw]))[0])
        if not (0.0 <= calibrated <= 1.0):
            raise ValueError(
                f"calibrated_prob fuori range [0,1]: {calibrated} (raw={raw}, winner={self.calibrator_winner})"
            )
        return raw, calibrated
```
