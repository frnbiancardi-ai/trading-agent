```python
def profit_curve_optimal_threshold(
    prob_val: np.ndarray,
    pnl_pips: np.ndarray,
    sweep: np.ndarray,
) -> float:
    """Threshold ottimale = argmax expectancy_pips sul val set."""
    best_t, best_exp = float(sweep[0]), -np.inf

    return best_t
```

```python
"""Profit-curve threshold optimizer per profile (ML-04, D-14, RESEARCH §Pattern 6).

Sweep threshold ∈ [0.20, 0.70] step 0.01 sul VALIDATION set (D-07-04-A: NO test
fold per evitare reverse-leak — Plan 07-04 AST guard enforce in tests/test_ml_threshold.py).
Max expectancy_pips per profile (CONSERVATIVE, MODERATE, AGGRESSIVE).

Final threshold per profile = mediana 10-fold (D-07-04-B robust a outlier).
Fallback 0.5 se tutti i fold producono None per quel profile (raro: solo se
nessuna riga di quel profile e' presente in nessun val set).

Modulo PURO: nessun logging, nessun broker, nessun I/O. Pattern verbatim
RESEARCH §Pattern 6 + PATTERNS.md ml/threshold.py.
"""
from __future__ import annotations

from typing import Sequence

import numpy as np

# Mirror risk_engine.py:15-19 — names tuple (no profile-specific risk params here)
PROFILES: tuple[str, str, str] = ("CONSERVATIVE", "MODERATE", "AGGRESSIVE")


def profit_curve_optimal_threshold(
    prob_val: np.ndarray,
    pnl_pips: np.ndarray,
    sweep: np.ndarray,
) -> float:
    """Threshold ottimale = argmax expectancy_pips sul val set.

    Strategia: per ogni `t` in sweep, computa mean pnl_pips dove prob_val >= t.
    Ritorna t con expectancy massima. Se nessun threshold ha mask non-empty
    (es. prob_val < min(sweep)), ritorna sweep[0] (best_exp resta -inf).
    """
    best_t = float(sweep[0])
    best_exp = -np.inf
    for t in sweep:
        mask = prob_val >= t
        if mask.sum() == 0:
            continue
        exp = float(pnl_pips[mask].mean())
        if exp > best_exp:
            best_exp = exp
            best_t = float(t)
    return best_t


def optimize_thresholds_for_fold(
    prob_val: np.ndarray,
    pnl_val: np.ndarray,
    profile_val: np.ndarray,
    sweep: np.ndarray,
) -> dict[str, float | None]:
    """Threshold ottimale per ogni profile sul val set di un fold.

    Args:
        prob_val: calibrated_prob array (post Brier-winner) sul val set.
        pnl_val: pnl_pips array sul val set.
        profile_val: array di string profile labels sul val set.
        sweep: array di candidate thresholds (e.g., np.arange(0.20, 0.70, 0.01)).

    Returns:
        Dict con chiavi CONSERVATIVE/MODERATE/AGGRESSIVE; value = best threshold
        oppure None se profile assente nel val set (mask vuoto).
    """
    out: dict[str, float | None] = {}
    for profile in PROFILES:
        mask = profile_val == profile
        if mask.sum() == 0:
            out[profile] = None
            continue
        out[profile] = profit_curve_optimal_threshold(
            prob_val[mask], pnl_val[mask], sweep
        )
    return out


def aggregate_thresholds_across_folds(
    fold_thresholds: Sequence[dict[str, float | None]],
) -> dict[str, float]:
    """Mediana cross-fold per profile, ignorando None (D-07-04-B).

    Se TUTTI i fold producono None per un profile → fallback 0.5 (D-07-04-E
    safety: risk_engine usa la mappa per gate; 0.5 = nessun bias).
    """
    result: dict[str, float] = {}
    for profile in PROFILES:
        values = [
            t for ft in fold_thresholds
            if (t := ft.get(profile)) is not None
        ]
        if not values:
            result[profile] = 0.5
        else:
            result[profile] = float(np.median(values))
    return result


__all__ = [
```
