From RESEARCH §Pattern 4 (ECE, VERIFIED):

```python
def expected_calibration_error(y_true, y_prob, n_bins=10):
    bin_edges = np.linspace(0, 1, n_bins + 1)
    ece = 0.0
    for i in range(n_bins):
        if i < n_bins - 1:
            mask = (y_prob >= bin_edges[i]) & (y_prob < bin_edges[i + 1])
        else:
            mask = (y_prob >= bin_edges[i]) & (y_prob <= bin_edges[i + 1])  # <= last bin
        if mask.sum() == 0:
            continue
        bin_acc = float(y_true[mask].mean())
        bin_conf = float(y_prob[mask].mean())
        ece += (mask.sum() / len(y_true)) * abs(bin_acc - bin_conf)
    return ece
```

```python
def pick_brier_winner(
    raw_test: np.ndarray, y_test: np.ndarray,
    platt: LogisticRegression, iso: IsotonicRegression,
    val_size: int, val_size_threshold: int = 50,
) -> tuple[str, np.ndarray]:
    if val_size < val_size_threshold:
        return "sigmoid", platt.predict_proba(raw_test.reshape(-1, 1))[:, 1]
    prob_platt = platt.predict_proba(raw_test.reshape(-1, 1))[:, 1]
    prob_iso   = iso.transform(raw_test)
    brier_platt = brier_score_loss(y_test, prob_platt)
    brier_iso   = brier_score_loss(y_test, prob_iso)
    return ("isotonic", prob_iso) if brier_iso <= brier_platt else ("sigmoid", prob_platt)
```

"""Manual Platt + Isotonic calibration + ECE (ML-04, RESEARCH §Pattern 2/4).

sklearn 1.8.0 ha RIMOSSO `CalibratedClassifierCV(cv='prefit')` — CONTEXT skeleton
lines 316-319 e' obsoleto (InvalidParameterError a runtime). Approccio:
fit LogisticRegression (Platt) o IsotonicRegression direttamente su raw
predict_proba del LightGBM model pre-fittato. NO refit del base model.

Brier winner picking per fold (D-18); val<50 → Platt-only forzato (HANDOFF
blocker #3). ECE last-bin <= per includere y_prob == 1.0 (Pitfall 5).

Modulo PURO: nessun logging, nessun broker, nessun I/O (test_ml_purity.py gating).
"""
from __future__ import annotations

import numpy as np
from sklearn.isotonic import IsotonicRegression
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import brier_score_loss


def platt_fit(raw_val: np.ndarray, y_val: np.ndarray) -> LogisticRegression:
    """Platt scaling: LogisticRegression su raw proba.

    Pre-condizione: raw_val e' l'output di `model.predict_proba(X_val)[:, 1]`
    (raw, NON calibrato) del LightGBM gia' fittato con early stopping.
    """
    cal = LogisticRegression(C=1.0)
    cal.fit(raw_val.reshape(-1, 1), y_val)
    return cal


def isotonic_fit(raw_val: np.ndarray, y_val: np.ndarray) -> IsotonicRegression:
    """Isotonic monotonic non-parametric. out_of_bounds='clip' per safety OOD."""
    cal = IsotonicRegression(out_of_bounds="clip")
    cal.fit(raw_val, y_val)
    return cal


def pick_brier_winner(
    raw_test: np.ndarray,
    y_test: np.ndarray,
    platt: LogisticRegression,
    iso: IsotonicRegression,
    val_size: int,
    val_size_threshold: int = 50,
) -> tuple[str, np.ndarray]:
    """Sceglie il calibratore vincente per Brier su test fold.

    val < val_size_threshold → Platt-only forzato (HANDOFF blocker #3 — Isotonic
    unreliable su < 50 sample). Altrimenti: argmin Brier.

    Returns:
        ("sigmoid" | "isotonic", calibrated_probs_test).
    """
    if val_size < val_size_threshold:
        return "sigmoid", platt.predict_proba(raw_test.reshape(-1, 1))[:, 1]
    prob_platt = platt.predict_proba(raw_test.reshape(-1, 1))[:, 1]
    prob_iso = iso.transform(raw_test)
    brier_platt = brier_score_loss(y_test, prob_platt)
    brier_iso = brier_score_loss(y_test, prob_iso)
    return ("isotonic", prob_iso) if brier_iso <= brier_platt else ("sigmoid", prob_platt)


def expected_calibration_error(
    y_true: np.ndarray,
    y_prob: np.ndarray,
    n_bins: int = 10,
) -> float:
    """ECE = Σ_b (|B_b|/N) × |acc(B_b) − conf(B_b)| (RESEARCH §Pattern 4).

    Last bin usa `<=` (NON `<`) per includere `y_prob == 1.0`. CONTEXT.md skeleton
    riga 582-585 e' off-by-one (Pitfall 5); questa versione e' corretta.
    """
    y_true = np.asarray(y_true)
    y_prob = np.asarray(y_prob)
    bin_edges = np.linspace(0, 1, n_bins + 1)
    ece = 0.0
    for i in range(n_bins):
        if i < n_bins - 1:
            mask = (y_prob >= bin_edges[i]) & (y_prob < bin_edges[i + 1])
        else:
            mask = (y_prob >= bin_edges[i]) & (y_prob <= bin_edges[i + 1])  # FIX last bin
        if mask.sum() == 0:
            continue
        bin_acc = float(y_true[mask].mean())
        bin_conf = float(y_prob[mask].mean())
        ece += (mask.sum() / len(y_true)) * abs(bin_acc - bin_conf)
    return float(ece)

    "pick_brier_winner",
    "expected_calibration_error",
]

def _build_feature_list(df: pd.DataFrame) -> tuple[list[str], list[str]]:
    """Costruisce feature_list (numerical + categorical encoded) escludendo y/meta.

    Returns (feature_list, cat_cols_present): cat_cols filtrato a colonne realmente
    presenti nel df (es. fixture potrebbe non avere "profile").
    """
    cat_cols_present = [c for c in _CAT_COLS if c in df.columns]
    features = [
        c for c in df.columns
        if c not in _EXCLUDE_FROM_FEATURES and c not in cat_cols_present
    ]
    # Numerical features prima, poi categoricals → ordine stabile cross-run
    return features + cat_cols_present, cat_cols_present

      - y_test: np.ndarray | None — labels effettivi sul test fold
    """
    result: "FoldResult | None"

def _train_one_fold(
    df_sorted: pd.DataFrame,
    train_idx: np.ndarray,
    val_idx: np.ndarray,
    test_idx: np.ndarray,
    feature_list: list[str],
    cat_cols: list[str],
    cfg: MLTrainingConfig,
    fold_num: int,
) -> "FoldArtifacts":
    """Trains LightGBM + manual calibration per fold.

    Returns FoldArtifacts (NamedTuple) per cross-plan contract stabilita' — Plan
    07-04 e 07-05 unpack via attribute access (fa.result, fa.model, ...). Vedi
    FoldArtifacts dataclass sotto. Rev iter1: eliminato il 6-tuple opaco per
    prevenire silent break su rename downstream.
    """
    if len(val_idx) == 0 or len(test_idx) == 0:
        _log.warning("fold %d: val_idx o test_idx vuoto, skip", fold_num)
        return FoldArtifacts(result=None, model=None, platt=None, iso=None, prob_test=None, y_test=None)

    # Rev iter1: y_full computato OUT-OF-FOLD da build_dataset_xy (Plan 07-01 D-01 single source of truth).
    # Il caller (train_classifier) passa df_sorted GIA' con cat-encoding applicato + "regime" rinominato;
    # qui basta lo slice. NB: nel design corrente _train_one_fold non riceve y_full → ricomputa una volta sola.
    # Refactor full: train_classifier passera' y_full per arg (vedi Step seguente nel caller).
    y_full_local = (df_sorted["exit_reason"] == "TP_HIT").astype(int).to_numpy()
    X_train = df_sorted.iloc[train_idx][feature_list]
    y_train = y_full_local[train_idx]
    X_val = df_sorted.iloc[val_idx][feature_list]
    y_val = y_full_local[val_idx]
    X_test = df_sorted.iloc[test_idx][feature_list]
    y_test = y_full_local[test_idx]
    # D-01 label semantica vive in ml.feature_extraction.build_dataset_xy; questo blocco
    # ricalcola SOLO con la stessa formula. Drift di semantica catturato da
    # test_ml_feature_extraction::test_build_dataset_xy_label_encoding.

    # D-04: scale_pos_weight per fold
    n_pos = int(y_train.sum())
    n_neg = int(len(y_train) - n_pos)
    spw = (n_neg / max(n_pos, 1)) if n_pos > 0 else 1.0

    lgb_params = dict(cfg.lightgbm)
    early_stop_rounds = int(lgb_params.pop("early_stopping_rounds", 50))

    model = lgb.LGBMClassifier(
        **lgb_params,
        scale_pos_weight=spw,
        objective="binary",
    )
    model.fit(
        X_train, y_train,
        eval_set=[(X_val, y_val)] if len(y_val) > 0 else None,
        callbacks=[lgb.early_stopping(early_stop_rounds, verbose=False), lgb.log_evaluation(-1)] if len(y_val) > 0 else None,
        categorical_feature=cat_cols if cat_cols else "auto",
    )

    # Manual calibration su val
    raw_val = model.predict_proba(X_val)[:, 1]
    platt = platt_fit(raw_val, y_val)
    iso = isotonic_fit(raw_val, y_val)

    # Brier winner su test
    raw_test = model.predict_proba(X_test)[:, 1]
    winner, prob_test = pick_brier_winner(
        raw_test, y_test, platt, iso,
        val_size=len(y_val),
        val_size_threshold=cfg.val_size_threshold,
    )

    brier = float(brier_score_loss(y_test, prob_test))
    ece = float(expected_calibration_error(y_test, prob_test, n_bins=10))
    precision, recall, _ = precision_recall_curve(y_test, prob_test)
    auc_pr = float(auc(recall, precision))

    result = FoldResult(
        fold=fold_num,
        n_train=int(len(y_train)),
        n_val=int(len(y_val)),
        n_test=int(len(y_test)),
        scale_pos_weight=float(spw),
        brier_test=brier,
        ece_test=ece,
        auc_pr_test=auc_pr,
        calibrator_winner=winner,
    )
    _log.info(
        "fold %d: train=%d val=%d test=%d spw=%.4f Brier=%.4f ECE=%.4f AUC-PR=%.4f winner=%s",
        fold_num, len(y_train), len(y_val), len(y_test), spw, brier, ece, auc_pr, winner,
    )
    return FoldArtifacts(result=result, model=model, platt=platt, iso=iso, prob_test=prob_test, y_test=y_test)


def plot_reliability_diagram(
    y_true: np.ndarray,
    y_prob: np.ndarray,
    fold_num: int,
    out_path: Path,
    n_bins: int = 10,
) -> Path:
    """Reliability diagram (matplotlib Agg backend; pattern Plan 05-07)."""
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt

    out_path.parent.mkdir(parents=True, exist_ok=True)
    bin_edges = np.linspace(0, 1, n_bins + 1)
    bin_lowers = bin_edges[:-1]
    frac_pos, mean_pred = [], []
    for i in range(n_bins):
        lo, hi = bin_edges[i], bin_edges[i + 1]
        if i < n_bins - 1:
            mask = (y_prob >= lo) & (y_prob < hi)
        else:
            mask = (y_prob >= lo) & (y_prob <= hi)
        if mask.sum() == 0:
            frac_pos.append(np.nan)
            mean_pred.append(np.nan)
        else:
            frac_pos.append(float(y_true[mask].mean()))
            mean_pred.append(float(y_prob[mask].mean()))

    fig, ax = plt.subplots(figsize=(6, 6))
    ax.plot([0, 1], [0, 1], "k--", label="Calibrazione perfetta")
    ax.bar(bin_lowers, frac_pos, width=1 / n_bins, alpha=0.5, color="steelblue", label="Frazione positivi")
    ax.plot(mean_pred, frac_pos, "ro-", label="Modello")
    ax.set_xlabel("Probabilita predetta media (bin)")
    ax.set_ylabel("Frazione positivi reali")
    ax.set_title(f"Reliability Diagram — Fold {fold_num}")
    ax.legend()
    ax.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.savefig(out_path, dpi=100)
    plt.close(fig)
    return out_path

def write_training_report(
    folds: list[FoldResult],
    cfg: MLTrainingConfig,
) -> Path:
    """Markdown report con fold metrics table + aggregate stats (SC#3)."""
    today = date.today().isoformat()
    report_dir = Path(cfg.research_dir)
    report_dir.mkdir(parents=True, exist_ok=True)
    report_path = report_dir / f"{cfg.training_report_prefix}-{today}.md"

    lines = [
        f"# ML Training Report — {today}",
        "",
        f"**Dataset:** `{cfg.dataset_path}`",
        f"**LightGBM:** {lgb.__version__}",
        f"**scikit-learn:** {sklearn.__version__}",
        f"**Git SHA:** `{_git_sha()}`",
        f"**N folds produced:** {len(folds)}",
        "",
        "## Per-Fold Metrics",
        "",
        "| Fold | n_train | n_val | n_test | scale_pos_weight | Brier | ECE | AUC-PR | calibrator |",
        "|------|---------|-------|--------|------------------|-------|-----|--------|------------|",
    ]
    for f in folds:
        lines.append(
            f"| {f.fold} | {f.n_train} | {f.n_val} | {f.n_test} | {f.scale_pos_weight:.4f} | "
            f"{f.brier_test:.4f} | {f.ece_test:.4f} | {f.auc_pr_test:.4f} | {f.calibrator_winner} |"
        )
    if folds:
        avg_brier = sum(f.brier_test for f in folds) / len(folds)
        avg_ece = sum(f.ece_test for f in folds) / len(folds)
        avg_auc = sum(f.auc_pr_test for f in folds) / len(folds)
        winners = [f.calibrator_winner for f in folds]
        winner_mode = max(set(winners), key=winners.count)
        lines += [
            "",
            "## Aggregate",
            "",
            f"- **Mean Brier:** {avg_brier:.4f}",
            f"- **Mean ECE:** {avg_ece:.4f}",
            f"- **Mean AUC-PR:** {avg_auc:.4f}",
            f"- **Calibrator winner mode:** {winner_mode} ({winners.count(winner_mode)}/{len(folds)} folds)",
            "",
        ]
    report_path.write_text("\n".join(lines), encoding="utf-8")
    return report_path


def train_classifier(cfg: MLTrainingConfig) -> list[FoldResult]:
    """Pipeline completa: carica → fold → train+calibrate per fold → save artifacts.

    Plan 07-03 scope:
      - Output: list[FoldResult] + metadata.json per fold + reliability PNG + report MD
      - NON output: joblib bundle (Plan 07-05) + threshold per profile (Plan 07-04)
    """
    dataset_path = Path(cfg.dataset_path)
    if not dataset_path.exists():
        raise FileNotFoundError(f"dataset path mancante: {dataset_path}")
    _log.info("Carico dataset %s", dataset_path)
    df = pd.read_parquet(dataset_path)
    _log.info("Dataset: %d rows × %d cols", len(df), len(df.columns))

    # dataset_hash per audit trail
    import hashlib
    dataset_hash = hashlib.sha256(dataset_path.read_bytes()).hexdigest()

    # Sort + reset index (mirror walk_forward internal sort, ma serve qui anche
    # per coerenza tra df_sorted e fold indices)
    df_sorted = df.sort_values("decision_ts_utc").reset_index(drop=True)

    # D-07-03-X (rev iter1): canonical naming — inference contract usa "regime", train pipeline rinomina
    # df["regime_state"] -> df["regime"] PRIMA di build_categorical_encodings, cosi' la mappa
    # metadata.json["categorical_encodings"]["regime"] e' coerente con _resolve_categorical("regime") di Plan 07-01.
    if "regime_state" in df_sorted.columns and "regime" not in df_sorted.columns:
        df_sorted = df_sorted.rename(columns={"regime_state": "regime"})

    feature_list, cat_cols = _build_feature_list(df_sorted)
    _log.info("Features: %d numerical+categorical (cat: %s)", len(feature_list), cat_cols)

    # Build folds
    folds_idx = build_ml_folds(df_sorted, n_folds=cfg.n_folds, embargo_bars_by_tf=cfg.embargo_bars_by_tf)
    _log.info("Walk-forward: %d fold producibili", len(folds_idx))

    # Build categorical encodings su train globale (NB: tutti i train fold combinati =
    # tutto il df_sorted in expanding mode → encoding stabile cross-fold)
    cat_encodings = build_categorical_encodings(df_sorted, cat_cols)
    for col in cat_cols:
        df_sorted[col] = df_sorted[col].map(cat_encodings[col]).fillna(-1).astype(int)

    today = date.today().isoformat()
    fold_dir = Path(cfg.output_dir) / cfg.fold_subdir / f"v1_{today}"
    reliability_dir = Path(cfg.research_dir) / cfg.reliability_subdir

    fold_results: list[FoldResult] = []
    for k, (train_idx, val_idx, test_idx) in enumerate(folds_idx, start=1):
        try:
            fa = _train_one_fold(
                df_sorted, train_idx, val_idx, test_idx,
                feature_list, cat_cols, cfg, fold_num=k,
            )
            if fa.result is None:
                continue
            # Rev iter1: unpack via attribute access — cross-plan contract via FoldArtifacts NamedTuple.
            result, prob_test, y_test = fa.result, fa.prob_test, fa.y_test
            fold_results.append(result)
            _write_fold_metadata(result, cfg, fold_dir, dataset_hash)
            plot_reliability_diagram(
                y_test, prob_test, fold_num=k,
                out_path=reliability_dir / f"fold_{k}.png",
            )
        except Exception as exc:  # noqa: BLE001
            _log.exception("fold %d fallito: %s", k, exc)

    write_training_report(fold_results, cfg)
    _log.info("Train pipeline completata: %d fold riusciti", len(fold_results))
    return fold_results
