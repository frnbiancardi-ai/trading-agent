            ctx = {}
    elif not isinstance(ctx, dict):

"""Feature extraction Phase 7 (ML-01, D-01, D-10, D-11, D-09-G).

Pure-function module:
  - derive_d_09_g_fields(row): deriva 8 fields D-09-G deferred da Plan 05-09
    (setup_name, pattern_name, confluence_factors_json, bias, sl_pips, tp_pips,
    r_to_r, bars_to_outcome) da decision_context_json + pnl_pips + timestamps.
  - build_categorical_encodings(df, cat_cols): mappa stabile (sorted-value order)
    per inference-time consistency (RESEARCH §Pitfall 2).
  - build_feature_vector(draft, indicators, ctx, feature_list, cat_encodings):
    single-row pd.DataFrame per inference (preserva nomi colonna → no sklearn
    UserWarning).
  - build_dataset_xy(parquet_path, feature_list, cat_cols): batch X/y per
    training. Etichetta D-01: y=1 ⟺ exit_reason==TP_HIT.

D-19 italiano: docstring/commenti/rationale italiani, identifier inglesi.
D-07-01-F: questo modulo e' OPEN_EXEMPT (usa pd.read_parquet che chiama open()).
D-17 no-future-leakage: legge SOLO il parquet baseline (Plan 05-09 D-09-B
strict-< slice gia' applicato a write-time); nessun recompute indicatori.
"""
from __future__ import annotations

import json
from pathlib import Path
from typing import Any

import numpy as np
import pandas as pd

# Mapping TF -> bar/ora per derive_d_09_g_fields bars_to_outcome calculation.
# M15 = 60/15 = 4, M30 = 60/30 = 2, H1 = 1, H4 = 0.25 (riservato future).
TF_BARS_PER_HOUR: dict[str, float] = {
    "M15": 4.0,
    "M30": 2.0,
    "H1": 1.0,
    "H4": 0.25,
    "D1": 1.0 / 24.0,
}


def _pip_size(symbol: str) -> float:
    """0.01 per coppie JPY, 0.0001 altrimenti (convenzione FX)."""
    return 0.01 if "JPY" in symbol.upper() else 0.0001


def _safe_parse_ctx(ctx_value: Any) -> dict:
    """Defensive parse di decision_context_json — fallback empty dict."""
    if isinstance(ctx_value, dict):
        return ctx_value
    if isinstance(ctx_value, str):
        try:
            return json.loads(ctx_value)
        except (json.JSONDecodeError, ValueError):
            return {}
    return {}


def derive_d_09_g_fields(row: pd.Series) -> dict:
    """Deriva 8 campi D-09-G da row del parquet baseline_decisions.

    Plan 05-09 ha deferito by-design 8 campi che richiedono ctx + timestamps
    + price math. Plan 07-01 li deriva qui (read-only sul parquet).

    Returns dict con chiavi:
      - setup_name: ctx['setup_name'] o None
      - pattern_name: ctx.get('pattern_name') o None
      - confluence_factors_json: json.dumps(ctx.get('factors', {}))
      - bias: ctx.get('direction')  # BUY/SELL
      - sl_pips: abs(entry_price - sl) / pip_size(symbol)
      - tp_pips: abs(tp - entry_price) / pip_size(symbol)
      - r_to_r: tp_pips / sl_pips se sl_pips > 0 else None
      - bars_to_outcome: int((exit_ts - entry_ts) / bar_duration_for(timeframe))

    Resiliente: ctx parse fail → empty dict, fields string = None, fields
    derivabili da prezzi/timestamps comunque calcolati.
    """
    ctx = _safe_parse_ctx(row.get("decision_context_json"))

    # String fields da ctx
    setup_name = ctx.get("setup_name")
    pattern_name = ctx.get("pattern_name")
    bias = ctx.get("direction")
    factors = ctx.get("factors", {})
    confluence_factors_json = json.dumps(factors) if isinstance(factors, dict) else json.dumps({})

    # Numeric fields da prezzi
    symbol = row.get("symbol")
    pip = _pip_size(symbol) if symbol else 0.0001
    entry = float(row.get("entry_price", 0.0))
    sl = float(row.get("sl", entry))
    tp = float(row.get("tp", entry))
    sl_pips = abs(entry - sl) / pip if pip > 0 else None
    tp_pips = abs(tp - entry) / pip if pip > 0 else None
    r_to_r = (tp_pips / sl_pips) if sl_pips and sl_pips > 0 else None

    # bars_to_outcome da timestamps
    tf = row.get("timeframe")
    bars_to_outcome = None
    try:
        entry_ts = pd.Timestamp(row.get("entry_ts_utc"))
        exit_ts = pd.Timestamp(row.get("exit_ts_utc"))
        bars_per_h = TF_BARS_PER_HOUR.get(tf, 1.0)
        hours = (exit_ts - entry_ts).total_seconds() / 3600.0
        bars_to_outcome = int(round(hours * bars_per_h))
    except (TypeError, ValueError):
        pass

    return {
        "setup_name": setup_name,
        "pattern_name": pattern_name,
        "confluence_factors_json": confluence_factors_json,
        "bias": bias,
        "sl_pips": sl_pips,
        "tp_pips": tp_pips,
        "r_to_r": r_to_r,
        "bars_to_outcome": bars_to_outcome,
    }


def build_categorical_encodings(df: pd.DataFrame, cat_cols: list[str]) -> dict[str, dict[str, int]]:
    """Mappa value → int code per ogni colonna categorica.

    Codici assegnati in sorted-value order (lexicographic). Stable cross-run
    e cross-DataFrame-order (RESEARCH §Pitfall 2: pd.Categorical.cat.codes
    e' instable su insertion order, da NON usare).

    Persistito da Plan 07-05 in metadata.json; inference fa lookup con
    fallback -1 per unknown.
    """
    encodings: dict[str, dict[str, int]] = {}
    for col in cat_cols:
        if col not in df.columns:
            encodings[col] = {}
            continue
        unique_vals = sorted(v for v in df[col].dropna().unique())
        encodings[col] = {val: idx for idx, val in enumerate(unique_vals)}
    return encodings


def build_feature_vector(
    draft,
    indicators,
    ctx,
    feature_list: list[str],
    cat_encodings: dict[str, dict[str, int]],
) -> pd.DataFrame:
    """Costruisce single-row DataFrame ordinato per inference.

    Lookup field-by-field nei 3 source object (draft, indicators, ctx); ordine
    finale = feature_list. Categoricals encodati via cat_encodings.get(value, -1).

    Output: pd.DataFrame con 1 riga, colonne = feature_list. NON np.ndarray:
    LightGBM emette UserWarning su numpy se il modello e' stato fittato con
    DataFrame (RESEARCH §Code Examples lines 686-692).
    """
    row: dict[str, Any] = {}
    cat_cols = set(cat_encodings.keys())

    for fname in feature_list:
        if fname in cat_cols:
            val = _resolve_categorical(fname, draft, ctx)
            row[fname] = cat_encodings[fname].get(val, -1)
        else:
            row[fname] = _resolve_numeric(fname, draft, indicators, ctx)

    return pd.DataFrame([{f: row.get(f) for f in feature_list}])


def _resolve_categorical(name: str, draft, ctx) -> Any:
    """Risolve un valore categorico da draft o ctx.

    Cross-plan naming contract (D-07-01-G + D-07-03-X):
      - Inference-side canonical column name = "regime" (NON "regime_state").
      - Plan 07-03 train pipeline rinomina df['regime_state'] -> df['regime']
        PRIMA di costruire build_categorical_encodings, cosi' la mappa
        metadata.json["categorical_encodings"]["regime"] e' coerente con il
        lookup di inference qui sotto.
      - Backward-compat: si accetta anche ctx.regime_state se ctx non espone
        il campo "regime" (es. StrategyContext attuale Phase 4).
    """
    if name == "symbol":
        return getattr(ctx, "symbol", None)
    if name == "timeframe":
        return getattr(ctx, "timeframe", None)
    if name == "profile":
        return getattr(ctx, "profile", None)
    if name == "setup_name":
        return getattr(draft, "setup_name", None)
    if name == "regime":
        # Canonico "regime"; fallback su "regime_state" per StrategyContext legacy.
        return getattr(ctx, "regime", None) or getattr(ctx, "regime_state", None)
    # Fallback generico
    return getattr(draft, name, None) or getattr(ctx, name, None)


def _resolve_numeric(name: str, draft, indicators, ctx):
    """Risolve un valore numerico da draft.factors / indicators / ctx."""
    # Factors dict (Phase 4 D-08)
    if name.startswith("factors_"):
        factor_key = name[len("factors_"):]
        factors = getattr(draft, "factors", {}) or {}
        return int(bool(factors.get(factor_key)))
    # ProposalDraft numerics
    if name in ("confidence", "entry_price", "stop_loss_price", "take_profit_price"):
        return getattr(draft, name, None)
    if name == "rr":
        entry = getattr(draft, "entry_price", None)
        sl = getattr(draft, "stop_loss_price", None)
        tp = getattr(draft, "take_profit_price", None)
        if entry and sl and tp:
            risk = abs(entry - sl)
            reward = abs(tp - entry)
            return reward / risk if risk > 0 else None
        return None
    # Indicators (Phase 2 ExtendedIndicators)
    val = getattr(indicators, name, None)
    if val is not None:
        return val
    # Ctx fallback
    return getattr(ctx, name, None)


def build_dataset_xy(
    parquet_path: Path | str,
    feature_list: list[str],
    cat_cols: list[str],
) -> tuple[pd.DataFrame, np.ndarray, dict[str, dict[str, int]]]:
    """Carica parquet baseline_decisions e costruisce (X, y, cat_encodings).

    Returns:
        X: pd.DataFrame con colonne feature_list, categoricals encodati come int.
        y: np.ndarray binary (D-01: TP_HIT → 1, altro → 0).
        cat_encodings: mappa value → int code per inference-time replay.

    D-09-G: se feature_list include uno dei 8 fields deferred, viene calcolato
    on-the-fly via derive_d_09_g_fields applicato row-wise.
    """
    df = pd.read_parquet(parquet_path)

    # Etichetta D-01
    y = (df["exit_reason"] == "TP_HIT").astype(int).to_numpy()

    # D-09-G derivation se richiesto
    d_09_g_fields = {"setup_name", "pattern_name", "confluence_factors_json",
                     "bias", "sl_pips", "tp_pips", "r_to_r", "bars_to_outcome"}
    needed = d_09_g_fields & set(feature_list)
    if needed:
        derived_rows = df.apply(derive_d_09_g_fields, axis=1)
        derived_df = pd.DataFrame(list(derived_rows))
        for field in needed:
            if field not in df.columns and field in derived_df.columns:
                df[field] = derived_df[field]

    # Encoding categorical
    cat_encodings = build_categorical_encodings(df, cat_cols)
    for col in cat_cols:
        if col in df.columns:
            df[col] = df[col].map(cat_encodings[col]).fillna(-1).astype(int)

    # X = sottoinsieme colonne feature_list (drop y/exit_reason)
    available = [f for f in feature_list if f in df.columns]
    X = df[available].copy()
    return X, y, cat_encodings
